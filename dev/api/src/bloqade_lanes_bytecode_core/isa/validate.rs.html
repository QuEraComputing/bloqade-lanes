1//! Architecture-dependent validation for the vihaco-backed ISA.
2//!
3//! These checks gate a [`Program`] against an [`ArchSpec`]. They are
4//! **arch-dependent by design**: when no arch spec is supplied ([`validate`]
5//! called with `None`), every check here is skipped and an empty error list is
6//! returned — an arch-free program is considered fine at this layer.
7//!
8//! [`validate`] runs the arch-dependent capability + address checks below.
9//! [`validate_structure`] adds arch-independent structural checks, and
10//! [`simulate_stack`] adds optional stack-type simulation (underflow, type
11//! mismatches, and lane/location group validation).
12//!
13//! ## Capability checks
14//!
15//! - **`feed_forward` → multiple measurements.** Without mid-circuit classical
16//!   feedback the hardware can only run straight-line code, so at most one
17//!   `measure` may appear.
18//!
19//!   The companion rule against branch/call instructions
20//!   ([`ControlFlowRequiresFeedForward`](ValidationError::ControlFlowRequiresFeedForward))
21//!   rejects control flow for the same reason. It was unreachable for as long
22//!   as the ISA had no such instructions; adopting the calling convention
23//!   brought it back.
24//! - **`atom_reloading` → `fill`.** Without atom reloading, refilling atoms
25//!   after the initial fill is unsupported.
26//!
27//! ## What the stack simulation does not cover
28//!
29//! [`simulate_stack`] is a linear pass over each function from an empty stack,
30//! so it stops at the first branch or call and skips any function a `call`
31//! passes operands to. Both are conservative: it reports nothing rather than
32//! something it cannot justify from a state it does not have. Restoring that
33//! coverage needs a CFG walk that merges state at join points and models calls
34//! against the frame — see
35//! <https://github.com/QuEraComputing/bloqade-lanes/issues/1042>.
36//!
37//! ## Address checks
38//!
39//! Every `const_loc` / `const_lane` / `const_zone` operand is decoded and
40//! checked against the architecture's topology via
41//! [`ArchSpec::check_location`] / [`check_lane`](ArchSpec::check_lane) /
42//! [`check_zone`](ArchSpec::check_zone) — invalid zones, words, sites, lanes,
43//! and AOD constraints are reported with the arch layer's own message.
44

72/// An arch-dependent validation failure, tagged with the offending
73/// instruction's program counter.
74#[derive(Debug, Clone, PartialEq, Eq)]
75pub enum ValidationError {
76    /// A `br`, `cond_br`, `call` or `call_indirect` appears but
77    /// `feed_forward` is disabled.
78    ///
79    /// Without mid-circuit classical feedback the hardware can only run
80    /// straight-line code. This was documented as unreachable for as long as
81    /// the ISA had no control flow to reject; adopting the calling convention
82    /// brought it back.
83    ControlFlowRequiresFeedForward {
84        pc: usize,
85        /// The offending mnemonic (e.g. `"cond_br"`).
86        mnemonic: &'static str,
87    },
88    /// More than one `measure` appears but `feed_forward` is disabled.
89    MultipleMeasuresRequireFeedForward { pc: usize },
90    /// A `fill` appears but `atom_reloading` is disabled.
91    FillRequiresAtomReloading { pc: usize },
92    /// A `const_loc` operand does not name a valid location in the arch spec.
93    InvalidLocation { pc: usize, message: String },
94    /// A `const_lane` operand does not name a valid lane in the arch spec.
95    InvalidLane { pc: usize, message: String },
96    /// A `const_zone` operand does not name a valid zone in the arch spec.
97    InvalidZone { pc: usize, message: String },
98
99    // ---- structural (arch-independent) ----
100    /// `new_array` dim0 must be greater than zero.
101    NewArrayZeroDim0 { pc: usize },
102    /// `new_array` type_tag exceeds the maximum value tag.
103    NewArrayInvalidTypeTag { pc: usize, type_tag: u32 },
104    /// `new_array` declares more elements than [`MAX_ARRAY_ELEMENTS`].
105    NewArrayTooManyElements { pc: usize, count: u64 },
106    /// `get_item` takes an index count outside `1..=`[`MAX_GET_ITEM_DIMS`].
107    GetItemInvalidDims { pc: usize, ndims: u32 },
108    /// `load`/`store` names a local past [`MAX_LOCAL_INDEX`].
109    LocalIndexOutOfRange {
110        pc: usize,
111        /// The offending mnemonic (`"load"` or `"store"`).
112        mnemonic: &'static str,
113        index: u32,
114    },
115    /// An instruction sits outside every function's extent — after the last
116    /// `func_end`, or in a stream with no markers at all. Unreachable, and
117    /// unrenderable: [`super::text::to_text`] emits it after the closing brace.
118    CodeOutsideFunction { pc: usize },
119    /// A `br`/`cond_br` names an address outside the code, or a `call` names
120    /// one that does not begin a function.
121    InvalidControlFlowTarget {
122        pc: usize,
123        target: u32,
124        /// What the target should have been: `"an address in the code"` or
125        /// `"a function entry"`.
126        expected: &'static str,
127    },
128    /// A `call`'s arity operand disagrees with the callee's declared
129    /// parameter count.
130    ///
131    /// The operand is not a hint: it sets `base = stack.len() - arity`, so an
132    /// unchecked one redefines the callee's frame per call site.
133    CallArityMismatch {
134        pc: usize,
135        target: u32,
136        declared: u32,
137        got: u32,
138    },
139    /// A `ret`'s keep count disagrees with the enclosing function's declared
140    /// return count.
141    ///
142    /// Two `ret`s that disagree make the caller's post-call stack depth
143    /// path-dependent, which is the same defect as a branch whose arms leave
144    /// different depths. Checking both against the declaration catches it
145    /// without having to compare them to each other.
146    ReturnCountMismatch { pc: usize, declared: u32, got: u32 },
147    /// `initial_fill` is not the first non-constant instruction.
148    InitialFillNotFirst { pc: usize },
149    /// The program has no instructions (and therefore no terminator).
150    EmptyProgram,
151    /// Some path through a function runs off its end instead of reaching
152    /// `return` or `halt`. Per function, not per program: `func_end` is a
153    /// no-op, so falling off it runs whatever was laid out next.
154    MissingTerminator { pc: usize },
155    /// An instruction follows a `return`/`halt` and is unreachable.
156    UnreachableInstruction { pc: usize },
157
158    // ---- stack-type simulation (only via `simulate_stack`) ----
159    /// An instruction popped from an empty stack.
160    StackUnderflow { pc: usize },
161    /// A popped value had the wrong type tag (see [`tag`]).
162    TypeMismatch { pc: usize, expected: u8, got: u8 },
163    /// A `local_r`/`local_rz`/`fill`/`initial_fill` location group is invalid.
164    LocationGroupValidation {
165        pc: usize,
166        error: LocationGroupError,
167    },
168    /// A `move` lane group is invalid (duplicate, inconsistent, AOD, …).
169    LaneGroupValidation { pc: usize, error: LaneGroupError },
170}

172/// Maximum valid `new_array` element type tag ([`tag::MEASUREMENT_RESULT`]).
173const MAX_TYPE_TAG: u32 = tag::MEASUREMENT_RESULT as u32;
174
175/// Maximum number of elements a `new_array` may declare.
176///
177/// `dim0` and `dim1` are attacker-controlled `u32`s read straight out of the
178/// instruction word, and their product drives a pop loop. Without a bound, a
179/// 28-byte program can make the validator pop four billion times, and the
180/// product itself overflows `u32`. An array holds one element per measured
181/// site, so a million is already orders of magnitude past any physical
182/// architecture — the bound exists to make a malformed word a diagnosis
183/// rather than a hang.
184pub const MAX_ARRAY_ELEMENTS: u64 = 1 << 20;
185
186/// Maximum number of indices a `get_item` may take.
187///
188/// `new_array` carries exactly two dimension fields, so an array is at most
189/// 2-D and one or two indices is the only well-formed shape. The Python
190/// `stack_move.GetItem` documents the same invariant and defers enforcement
191/// here.
192pub const MAX_GET_ITEM_DIMS: u32 = 2;
193
194/// Highest local index a `load` or `store` may name, so a frame addresses at
195/// most 1024 locals.
196///
197/// vihaco's locals are a window into the operand stack starting at the current
198/// frame's base, and `store` *grows the stack to reach its index*: `op_store`
199/// calls `get_local_mut(index)`, which `resize`s to `base + index + 1` and
200/// writes `Undefined` into every new slot. The index is therefore a memory
201/// request read straight out of the instruction word — at the 16 bytes per slot
202/// measured in #1032, `store u64, 4294967295` touches about 68 GB from a
203/// 12-byte program. And because `resize` writes rather than reserves, the pages
204/// are resident on every platform, so this does not hide behind macOS's lazy
205/// commit the way the earlier `Vec::with_capacity` bounds did.
206///
207/// A lanes function's locals are its arguments, and the pipeline emits no
208/// `load`/`store` at all, so every legitimate index is a handful. The bound is
209/// set well above that rather than at it: its job is to keep a malformed
210/// operand from becoming an allocation, not to impose a calling convention on a
211/// hand-written program. The ceiling costs about 16 KB of operand stack, which
212/// is nothing, while still being three orders of magnitude past any function
213/// the compiler will emit.
214pub const MAX_LOCAL_INDEX: u32 = 1023;
215/// Record a `br`/`cond_br` target that leaves the branch's own function.
216///
217/// Testing only `target < code.len()` let a branch land anywhere in the
218/// stream: a trailing `cpu::cpu.label` resolves to the function's own
219/// `func_end`, and a target past it enters the *next* function's body without
220/// a frame — the caller's locals still in place, no `ret` to come back to.
221/// A branch is intra-procedural; leaving the function is what `call` is for.
222fn check_branch_target(
223    errors: &mut Vec<ValidationError>,
224    owner: &[Option<usize>],
225    pc: usize,
226    target: u32,
227) {
228    let same_function = owner
229        .get(target as usize)
230        .copied()
231        .flatten()
232        .is_some_and(|t| owner.get(pc).copied().flatten() == Some(t));
233    if !same_function {
234        errors.push(ValidationError::InvalidControlFlowTarget {
235            pc,
236            target,
237            expected: "an address in the same function",
238        });
239    }
240}

242/// Element count of a `new_array`, in `u64` so the product cannot overflow.
243/// `dim1 == 0` means a 1-D array.
244pub(crate) fn array_element_count(dim0: u32, dim1: u32) -> u64 {
245    dim0 as u64 * if dim1 == 0 { 1 } else { dim1 as u64 }
246}

248/// Record a `load`/`store` local index past [`MAX_LOCAL_INDEX`].
249fn check_local_index(
250    errors: &mut Vec<ValidationError>,
251    pc: usize,
252    mnemonic: &'static str,
253    index: u32,
254) {
255    if index > MAX_LOCAL_INDEX {
256        errors.push(ValidationError::LocalIndexOutOfRange {
257            pc,
258            mnemonic,
259            index,
260        });
261    }

366/// Validate a program's arch-dependent constraints (capabilities + addresses).
367///
368/// When `arch` is `None`, all checks are skipped and an empty list is returned.
369/// Otherwise every violation is collected in program order (the validator does
370/// not stop at the first error), so callers can report them all at once.
371pub fn validate(program: &Program, arch: Option<&ArchSpec>) -> Vec<ValidationError> {
372    let Some(arch) = arch else {
373        return Vec::new();
374    };
375
376    let mut errors = Vec::new();
377    let mut measure_count = 0u32;
378    let mut initial_fill_count = 0u32;
379
380    // The capability rules describe *the run*, so they are scoped to the code
381    // that runs. With `feed_forward` off the control-flow arm below rejects
382    // every `call`, so only the entry function can execute — counting within
383    // its span is exact wherever the measure rule is active, and stops an
384    // uncallable function's `measure` from condemning a program that measures
385    // once.
386    let entry = program
387        .main_function
388        .and_then(|i| program.functions.get(i as usize))
389        .map(|f| (f.start_address as usize, f.end_address as usize));
390    let runs = |pc: usize| entry.is_none_or(|(start, end)| pc >= start && pc < end);
391
392    for (pc, inst) in program.code.iter().enumerate() {
393        match inst {
394            // ---- capability checks ----
395            // Without mid-circuit classical feedback the hardware runs
396            // straight-line code only, so a branch or a call is unrunnable.
397            // This arm did not exist while the ISA had no such instructions;
398            // now that a program can declare functions and branch between
399            // them, its absence let every one of them through.
400            //
401            // Guarded on [`is_control_flow`] rather than re-listing the
402            // variants, so a new control-flow op is gated here the moment it
403            // joins that predicate, instead of silently passing the rule that
404            // exists to reject it.
405            inst if is_control_flow(inst) && !arch.feed_forward => {
406                errors.push(ValidationError::ControlFlowRequiresFeedForward {
407                    pc,
408                    mnemonic: super::machine::op_name(inst),
409                });
410            }
411            M::Lanes(L::Measure(_)) => {
412                // A textual count over the code stream, which is exact only
413                // because the arm above rejects every branch and call when
414                // `feed_forward` is false: without control flow the code runs
415                // once, top to bottom. With `feed_forward` on, repeats are
416                // allowed anyway, so the count does not have to be exact.
417                if runs(pc) {
418                    measure_count += 1;
419                    if !arch.feed_forward && measure_count > 1 {
420                        errors.push(ValidationError::MultipleMeasuresRequireFeedForward { pc });
421                    }
422                }
423            }
424            M::Lanes(L::Fill(_)) if !arch.atom_reloading => {
425                errors.push(ValidationError::FillRequiresAtomReloading { pc });
426            }
427            // A *second* `initial_fill` is a refill, whatever it is spelled.
428            // The whole-program "must be first" ordering rule used to make
429            // that unreachable; scoping it to a function did not, so a helper
430            // could reload atoms on hardware that cannot.
431            //
432            // Counted across the whole program, unlike `measure` above. The
433            // measure rule is only active when `feed_forward` is off, which
434            // bans `call` and leaves the entry function as the only code that
435            // runs — so entry-scoping is exact there. This rule is active with
436            // `feed_forward` *on*, where a called function does run, and
437            // deciding which functions are callable needs a call-graph walk.
438            // Counting all of them can only over-report, which is the right
439            // way for a capability gate to be wrong.
440            M::Lanes(L::InitialFill(_)) => {
441                initial_fill_count += 1;
442                if !arch.atom_reloading && initial_fill_count > 1 {
443                    errors.push(ValidationError::FillRequiresAtomReloading { pc });
444                }
445            }
446
447            // ---- address checks ----
448            M::Lanes(L::ConstLoc(bits)) => {
449                if let Some(message) = arch.check_location(&LocationAddr::decode(*bits)) {
450                    errors.push(ValidationError::InvalidLocation { pc, message });
451                }
452            }
453            M::Lanes(L::ConstLane(bits)) => {
454                for message in arch.check_lane(&LaneAddr::decode_u64(*bits)) {
455                    errors.push(ValidationError::InvalidLane { pc, message });
456                }
457            }
458            M::Lanes(L::ConstZone(bits)) => {
459                if let Some(message) = arch.check_zone(&ZoneAddr::decode(*bits)) {
460                    errors.push(ValidationError::InvalidZone { pc, message });
461                }
462            }
463
464            _ => {}
465        }
466    }
467
468    errors
469}

471/// True if `inst` transfers control somewhere the linear walk cannot follow.
472fn is_control_flow(inst: &M) -> bool {
473    matches!(
474        inst,
475        M::Cpu(C::Branch(_) | C::ConditionalBranch(_, _) | C::Call(_, _) | C::IndirectCall)
476    )
477}

479/// True if `inst` terminates execution (`return` or `halt`).
480fn is_terminator(inst: &M) -> bool {
481    matches!(inst, M::Cpu(C::Return(_)) | M::Cpu(C::Halt))
482}

484/// True if `inst` only pushes a constant (and so may precede `initial_fill`).
485fn is_constant_push(inst: &M) -> bool {
486    matches!(
487        inst,
488        M::Lanes(L::ConstLoc(_))
489            | M::Lanes(L::ConstLane(_))
490            | M::Lanes(L::ConstZone(_))
491            | M::Cpu(C::Const(Type::F64, Value::F64(_)))
492            | M::Cpu(C::Const(Type::I64, Value::I64(_)))
493    )
494}

496/// Validate a program's arch-independent structural rules: `new_array` operand
497/// bounds, `initial_fill` ordering, and terminator/reachability. These never
498/// consult an arch spec, so they always run.
499pub fn validate_structure(program: &Program) -> Vec<ValidationError> {
500    let mut errors = Vec::new();
501    let mut seen_non_constant = false;
502    let spans = function_spans(&program.code);
503
504    // Which function owns each address, so a branch can be constrained to its
505    // own. `pc -> span index`; `None` for code outside every function.
506    let mut owner = vec![None; program.code.len()];
507    for (i, span) in spans.iter().enumerate() {
508        for slot in owner.iter_mut().take(span.end).skip(span.start) {
509            *slot = Some(i);
510        }
511    }
512
513    for (pc, inst) in program.code.iter().enumerate() {
514        match inst {
515            M::Lanes(L::NewArray(type_tag, dim0, dim1)) => {
516                if *dim0 == 0 {
517                    errors.push(ValidationError::NewArrayZeroDim0 { pc });
518                }
519                if *type_tag > MAX_TYPE_TAG {
520                    errors.push(ValidationError::NewArrayInvalidTypeTag {
521                        pc,
522                        type_tag: *type_tag,
523                    });
524                }
525                // Bound the element count here, where the operands are read,
526                // so neither the stack simulator nor the machine has to guard
527                // its own pop loop against a four-billion-element array.
528                let count = array_element_count(*dim0, *dim1);
529                if count > MAX_ARRAY_ELEMENTS {
530                    errors.push(ValidationError::NewArrayTooManyElements { pc, count });
531                }
532                seen_non_constant = true;
533            }
534            M::Lanes(L::GetItem(ndims)) => {
535                if *ndims == 0 || *ndims > MAX_GET_ITEM_DIMS {
536                    errors.push(ValidationError::GetItemInvalidDims { pc, ndims: *ndims });
537                }
538                seen_non_constant = true;
539            }
540            // Bound the local index here, where the operand is read, so a
541            // `store` cannot turn a one-word instruction into a multi-gigabyte
542            // stack `resize` inside vihaco (see [`MAX_LOCAL_INDEX`]). `load`
543            // only reads, so it fails cleanly either way — it is checked too
544            // because a program naming a local that far out is malformed
545            // regardless of which end of it is reached first.
546            M::Cpu(C::Load(_, index)) => {
547                check_local_index(&mut errors, pc, "load", *index);
548                seen_non_constant = true;
549            }
550            M::Cpu(C::Store(_, index)) => {
551                check_local_index(&mut errors, pc, "store", *index);
552                seen_non_constant = true;
553            }
554            M::Lanes(L::InitialFill(_)) => {
555                if seen_non_constant {
556                    errors.push(ValidationError::InitialFillNotFirst { pc });
557                }
558                seen_non_constant = true;
559            }
560            // Branch and call targets are addresses, and nothing has checked
561            // them: a `br` past the end of the code, or a `call` to an address
562            // that begins no function, is a program the renderer cannot even
563            // name — it has to invent `@L9` / `@F1`, producing text that will
564            // not re-read.
565            M::Cpu(C::Branch(t)) => {
566                check_branch_target(&mut errors, &owner, pc, *t);
567                seen_non_constant = true;
568            }
569            M::Cpu(C::ConditionalBranch(t, f)) => {
570                check_branch_target(&mut errors, &owner, pc, *t);
571                check_branch_target(&mut errors, &owner, pc, *f);
572                seen_non_constant = true;
573            }
574            M::Cpu(C::Call(arity, target)) => {
575                if !matches!(
576                    program.code.get(*target as usize),
577                    Some(M::Cpu(C::FunctionStart))
578                ) {
579                    errors.push(ValidationError::InvalidControlFlowTarget {
580                        pc,
581                        target: *target,
582                        expected: "a function entry",
583                    });
584                } else if let Some(callee) = program
585                    .functions
586                    .iter()
587                    .find(|f| f.start_address == *target)
588                {
589                    // `call <arity>` sets `base = stack.len() - arity`, so the
590                    // operand decides where the callee's frame begins. Checked
591                    // against the declaration, that is a claim about the
592                    // callee; unchecked, it silently *redefines* the callee's
593                    // shape per call site, and two sites could disagree with
594                    // nothing to notice.
595                    let declared = callee.signature.params.len() as u32;
596                    if *arity != declared {
597                        errors.push(ValidationError::CallArityMismatch {
598                            pc,
599                            target: *target,
600                            declared,
601                            got: *arity,
602                        });
603                    }
604                }
605                seen_non_constant = true;
606            }
607            M::Cpu(C::Return(keep)) => {
608                // Checked against the enclosing function's declaration rather
609                // than against its other `ret`s. Same outcome when they
610                // disagree, but it names the offender: a `ret` that does not
611                // match what the function promised, rather than a pair that
612                // happens to differ.
613                if let Some(declared) = owner[pc]
614                    .and_then(|i| spans.get(i))
615                    .and_then(|span| {
616                        program
617                            .functions
618                            .iter()
619                            .find(|f| f.start_address == span.start as u32)
620                    })
621                    .map(|f| f.signature.ret.len() as u32)
622                    && *keep != declared
623                {
624                    errors.push(ValidationError::ReturnCountMismatch {
625                        pc,
626                        declared,
627                        got: *keep,
628                    });
629                }
630                seen_non_constant = true;
631            }
632            // A function boundary resets the ordering rule: `initial_fill`
633            // must lead its own function, not the whole code stream.
634            M::Cpu(C::FunctionStart) => seen_non_constant = false,
635            M::Cpu(C::FunctionEnd) => {}
636            inst if is_constant_push(inst) => {}
637            _ => seen_non_constant = true,
638        }
639    }
640
641    // Reachability and terminators are per *function*, and reachability
642    // follows branches rather than latching on the first terminator.
643    //
644    // The latch this replaces was right while a program was one flat `@main`.
645    // It condemned `cond_br`'s second arm — the `halt` ending the first arm
646    // set the latch — and it could not see code a `br` jumped over. Following
647    // the edges gets both right, and the edge extraction already exists for
648    // the target checks above.
649    let spans = function_spans(&program.code);
650
651    // An instruction outside every function is unreachable *and* unrenderable:
652    // `to_text` emits it after the last `}`, producing text that will not
653    // re-read.
654    let mut covered = vec![false; program.code.len()];
655    for span in &spans {
656        for slot in covered.iter_mut().take(span.end).skip(span.start) {
657            *slot = true;
658        }
659    }
660    for (pc, inside) in covered.iter().enumerate() {
661        if !inside {
662            errors.push(ValidationError::CodeOutsideFunction { pc });
663        }
664    }
665
666    // A label names a position in a function's *body*. One sitting on a
667    // `func_start` cannot be written down: the only spot before a function's
668    // first instruction is the header line, so `to_text` emits it as the first
669    // body label and it re-reads one address later — a different binary, no
670    // error. One on a `func_end`, or outside every function, has no
671    // instruction to mark at all.
672    for label in &program.labels {
673        let body = spans.iter().any(|span| {
674            let body_end = if span.closed { span.end - 1 } else { span.end };
675            (label.address as usize) > span.start && (label.address as usize) < body_end
676        });
677        if !body {
678            errors.push(ValidationError::InvalidControlFlowTarget {
679                pc: label.address as usize,
680                target: label.address,
681                expected: "an instruction inside a function body",
682            });
683        }
684    }
685
686    // "Empty" means no *body*: a program that is nothing but markers has
687    // nothing to run, however many functions it declares.
688    if !program.code.iter().any(|i| !is_marker(i)) {
689        errors.push(ValidationError::EmptyProgram);
690        return errors;
691    }
692
693    for span in &spans {
694        walk_function(program, span, &mut errors);
695    }
696
697    errors
698}

700/// One function's extent in the code stream.
701struct Span {
702    /// Address of the `func_start`.
703    start: usize,
704    /// One past the last address the function owns.
705    end: usize,
706    /// Whether a `func_end` closed it. An unclosed final function still owns
707    /// the rest of the code, and falls off its own end.
708    closed: bool,
709}
710
711fn is_marker(inst: &M) -> bool {
712    matches!(inst, M::Cpu(C::FunctionStart) | M::Cpu(C::FunctionEnd))
713}

715/// Entry addresses of every function some `call` passes operands to.
716///
717/// The arity lives at the call site, not on the callee — `resolve` records an
718/// empty [`vihaco::module::Signature`] for every function — so the only way to
719/// learn that a function receives operands is to read the calls that reach it.
720/// A function called from several sites with different arities appears here if
721/// *any* of them is nonzero, because one caller passing operands is enough to
722/// make the empty-stack simulation wrong.
723fn callee_arities(code: &[M]) -> HashSet<u32> {
724    code.iter()
725        .filter_map(|inst| match inst {
726            M::Cpu(C::Call(arity, target)) if *arity > 0 => Some(*target),
727            _ => None,
728        })
729        .collect()
730}

732/// Split the code into function extents, delimited by the markers.
733fn function_spans(code: &[M]) -> Vec<Span> {
734    let mut spans = Vec::new();
735    let mut open: Option<usize> = None;
736    for (pc, inst) in code.iter().enumerate() {
737        match inst {
738            M::Cpu(C::FunctionStart) => {
739                // A nested `func_start` closes nothing; the outer function
740                // runs up to it and `resolve` rejects the construct anyway.
741                if let Some(start) = open.replace(pc) {
742                    spans.push(Span {
743                        start,
744                        end: pc,
745                        closed: false,
746                    });
747                }
748            }
749            M::Cpu(C::FunctionEnd) => {
750                if let Some(start) = open.take() {
751                    spans.push(Span {
752                        start,
753                        end: pc + 1,
754                        closed: true,
755                    });
756                }
757            }
758            _ => {}
759        }
760    }
761    if let Some(start) = open {
762        spans.push(Span {
763            start,
764            end: code.len(),
765            closed: false,
766        });
767    }
768    spans
769}

771/// Walk one function from its entry, following branch edges.
772///
773/// Reports every body address no path reaches, and a missing terminator when
774/// any path runs off the end instead of hitting `ret`/`halt`. Both are
775/// per-function: a second function's dead code no longer hides a first
776/// function's missing terminator, which the old single either/or gate did.
777fn walk_function(program: &Program, span: &Span, errors: &mut Vec<ValidationError>) {
778    let body_end = if span.closed { span.end - 1 } else { span.end };
779    let mut seen = vec![false; span.end - span.start];
780    let mut work = vec![span.start];
781    let mut falls_off = false;
782
783    while let Some(pc) = work.pop() {
784        // A target outside this function is reported by the target checks;
785        // following it would attribute the callee's code to the caller.
786        if pc < span.start || pc >= span.end || seen[pc - span.start] {
787            continue;
788        }
789        seen[pc - span.start] = true;
790
791        let inst = &program.code[pc];
792        let fall_through = |work: &mut Vec<usize>, falls_off: &mut bool| {
793            if pc + 1 >= body_end {
794                *falls_off = true;
795            }
796            if pc + 1 < span.end {
797                work.push(pc + 1);
798            }
799        };
800        match inst {
801            M::Cpu(C::FunctionEnd) => falls_off = true,
802            M::Cpu(C::Branch(t)) => work.push(*t as usize),
803            M::Cpu(C::ConditionalBranch(t, f)) => {
804                work.push(*t as usize);
805                work.push(*f as usize);
806            }
807            inst if is_terminator(inst) => {}
808            _ => fall_through(&mut work, &mut falls_off),
809        }
810    }
811
812    for pc in span.start + 1..body_end {
813        if !seen[pc - span.start] {
814            errors.push(ValidationError::UnreachableInstruction { pc });
815        }
816    }
817    if falls_off {
818        // Point at the last body instruction where there is one; an empty
819        // function has only its marker to name. An empty function is not
820        // exempt: `func_end` is a no-op, so falling off it runs whatever was
821        // laid out next rather than returning.
822        errors.push(ValidationError::MissingTerminator {
823            pc: body_end.saturating_sub(1).max(span.start),
824        });
825    }
826}

830/// One tracked stack value: its type tag and (when known) concrete bits.
831#[derive(Debug, Clone)]
832struct SimEntry {
833    tag: u8,
834    value: Option<u64>,
835}

837/// Type-level stack simulator: walks the instruction stream tracking value
838/// types, reporting underflow and type mismatches, and — when given an
839/// [`ArchSpec`] — validating `move` lane groups and `fill`/`local_*` location
840/// groups (duplicates only without an arch).
841struct StackSimulator<'a> {
842    stack: Vec<SimEntry>,
843    errors: Vec<ValidationError>,
844    arch: Option<&'a ArchSpec>,
845    pc: usize,
846}
847
848impl<'a> StackSimulator<'a> {
849    fn new(arch: Option<&'a ArchSpec>) -> Self {
850        Self {
851            stack: Vec::new(),
852            errors: Vec::new(),
853            arch,
854            pc: 0,
855        }
856    }
857
858    /// Record that a pop found the stack empty.
859    ///
860    /// `StackUnderflow` carries nothing but the `pc`, so a second one for the
861    /// same instruction is a literally identical value and says nothing new.
862    /// Group pops (`fill 40`, `new_array`) would otherwise emit one per
863    /// missing operand — a million lines of `pc 0: stack underflow` for a
864    /// single malformed word. All pops for one instruction are consecutive, so
865    /// checking the last error is enough.
866    fn underflow(&mut self) {
867        let already = matches!(
868            self.errors.last(),
869            Some(ValidationError::StackUnderflow { pc }) if *pc == self.pc
870        );
871        if !already {
872            self.errors
873                .push(ValidationError::StackUnderflow { pc: self.pc });
874        }
875    }
876
877    fn pop_any(&mut self) {
878        if self.stack.pop().is_none() {
879            self.underflow();
880        }
881    }
882
883    fn pop_typed(&mut self, expected: u8) {
884        match self.stack.pop() {
885            Some(entry) if entry.tag != expected => {
886                self.errors.push(ValidationError::TypeMismatch {
887                    pc: self.pc,
888                    expected,
889                    got: entry.tag,
890                })
891            }
892            Some(_) => {}
893            None => self.underflow(),
894        }
895    }
896
897    fn pop_typed_n(&mut self, expected: u8, count: u32) {
898        for _ in 0..count {
899            self.pop_typed(expected);
900        }
901    }
902
903    /// Pop one value expected to have `expected` tag, returning its concrete
904    /// bits when the type matches.
905    fn pop_addr(&mut self, expected: u8) -> Option<u64> {
906        match self.stack.pop() {
907            Some(entry) if entry.tag == expected => entry.value,
908            Some(entry) => {
909                self.errors.push(ValidationError::TypeMismatch {
910                    pc: self.pc,
911                    expected,
912                    got: entry.tag,
913                });
914                None
915            }
916            None => {
917                self.underflow();
918                None
919            }
920        }
921    }
922
923    fn push(&mut self, tag: u8, value: Option<u64>) {
924        self.stack.push(SimEntry { tag, value });
925    }
926
927    fn sim_dup(&mut self) {
928        if let Some(top) = self.stack.last().cloned() {
929            self.stack.push(top);
930        } else {
931            self.underflow();
932        }
933    }
934
935    fn sim_swap(&mut self) {
936        let len = self.stack.len();
937        if len >= 2 {
938            self.stack.swap(len - 1, len - 2);
939        } else {
940            self.underflow();
941        }
942    }
943
944    /// Report each uniquely-duplicated location once (no-arch fallback).
945    fn check_duplicate_locations(&mut self, locations: &[LocationAddr]) {
946        let mut seen = HashSet::new();
947        let mut reported = HashSet::new();
948        for loc in locations {
949            let bits = loc.encode();
950            if !seen.insert(bits) && reported.insert(bits) {
951                self.errors.push(ValidationError::LocationGroupValidation {
952                    pc: self.pc,
953                    error: LocationGroupError::DuplicateAddress { address: bits },
954                });
955            }
956        }
957    }
958
959    /// Report each uniquely-duplicated lane once (no-arch fallback).
960    fn check_duplicate_lanes(&mut self, lanes: &[LaneAddr]) {
961        let mut seen = HashSet::new();
962        let mut reported = HashSet::new();
963        for lane in lanes {
964            let (d0, d1) = lane.encode();
965            let combined = (d0 as u64) | ((d1 as u64) << 32);
966            if !seen.insert(combined) && reported.insert(combined) {
967                self.errors.push(ValidationError::LaneGroupValidation {
968                    pc: self.pc,
969                    error: LaneGroupError::DuplicateAddress { address: (d0, d1) },
970                });
971            }
972        }
973    }
974
975    /// Pop `arity` locations and validate them as a group.
976    fn pop_and_validate_locations(&mut self, arity: u32) {
977        let bits: Vec<Option<u64>> = (0..arity).map(|_| self.pop_addr(tag::LOCATION)).collect();
978        let locations: Vec<LocationAddr> = bits
979            .iter()
980            .filter_map(|v| v.map(LocationAddr::decode))
981            .collect();
982        let pc = self.pc;
983        if let Some(arch) = self.arch {
984            for error in arch.check_locations(&locations) {
985                self.errors
986                    .push(ValidationError::LocationGroupValidation { pc, error });
987            }
988        } else {
989            self.check_duplicate_locations(&locations);
990        }
991    }
992
993    /// Pop `arity` lanes and validate them as a group.
994    fn sim_move(&mut self, arity: u32) {
995        let bits: Vec<Option<u64>> = (0..arity).map(|_| self.pop_addr(tag::LANE)).collect();
996        let lanes: Vec<LaneAddr> = bits
997            .iter()
998            .filter_map(|v| v.map(LaneAddr::decode_u64))
999            .collect();
1000        let pc = self.pc;
1001        if let Some(arch) = self.arch {
1002            for error in arch.check_lanes(&lanes) {
1003                self.errors
1004                    .push(ValidationError::LaneGroupValidation { pc, error });
1005            }
1006        } else {
1007            self.check_duplicate_lanes(&lanes);
1008        }
1009    }
1010
1011    fn dispatch(&mut self, inst: &M) {
1012        match inst {
1013            // constants push a typed value
1014            M::Cpu(C::Const(Type::F64, Value::F64(v))) => self.push(tag::FLOAT, Some(v.to_bits())),
1015            M::Cpu(C::Const(Type::I64, Value::I64(v))) => self.push(tag::INT, Some(*v as u64)),
1016            M::Lanes(L::ConstLoc(v)) => self.push(tag::LOCATION, Some(*v)),
1017            M::Lanes(L::ConstLane(v)) => self.push(tag::LANE, Some(*v)),
1018            M::Lanes(L::ConstZone(v)) => self.push(tag::ZONE, Some(*v as u64)),
1019
1020            // stack manipulation
1021            M::Lanes(L::Pop) => self.pop_any(),
1022            M::Cpu(C::Dup) => self.sim_dup(),
1023            M::Lanes(L::Swap) => self.sim_swap(),
1024
1025            // atom arrangement
1026            M::Lanes(L::InitialFill(arity)) | M::Lanes(L::Fill(arity)) => {
1027                self.pop_and_validate_locations(*arity)
1028            }
1029            M::Lanes(L::Move(arity)) => self.sim_move(*arity),
1030
1031            // gates
1032            M::Lanes(L::LocalR(arity)) => {
1033                self.pop_typed_n(tag::FLOAT, 2);
1034                self.pop_and_validate_locations(*arity);
1035            }
1036            M::Lanes(L::LocalRz(arity)) => {
1037                self.pop_typed_n(tag::FLOAT, 1);
1038                self.pop_and_validate_locations(*arity);
1039            }
1040            M::Lanes(L::GlobalR) => self.pop_typed_n(tag::FLOAT, 2),
1041            M::Lanes(L::GlobalRz) => self.pop_typed_n(tag::FLOAT, 1),
1042            M::Lanes(L::Cz) => self.pop_typed(tag::ZONE),
1043
1044            // measurement
1045            M::Lanes(L::Measure(arity)) => {
1046                self.pop_typed_n(tag::ZONE, *arity);
1047                for _ in 0..*arity {
1048                    self.push(tag::MEASURE_FUTURE, None);
1049                }
1050            }
1051            // `await_measure` yields an *array* of measurement results — the
1052            // same value `set_detector`/`set_observable` consume, and the type
1053            // `stack_move.AwaitMeasure` is declared to produce. The distinct
1054            // `MEASUREMENT_RESULT` tag is that array's *element* type, which a
1055            // one-tag-per-slot simulator cannot express; it exists so
1056            // `new_array` can name the element type (#547). Giving the whole
1057            // array that tag broke `measure -> await_measure -> set_detector`.
1058            M::Lanes(L::AwaitMeasure) => {
1059                self.pop_typed(tag::MEASURE_FUTURE);
1060                self.push(tag::ARRAY_REF, None);
1061            }
1062
1063            // arrays
1064            M::Lanes(L::NewArray(_type_tag, dim0, dim1)) => {
1065                // `validate_structure` rejects counts past the cap, so the
1066                // clamp here only keeps a rejected program from also driving
1067                // an unbounded loop before its diagnosis is reported.
1068                let count = array_element_count(*dim0, *dim1).min(MAX_ARRAY_ELEMENTS);
1069                for _ in 0..count {
1070                    self.pop_any();
1071                }
1072                self.push(tag::ARRAY_REF, None);
1073            }
1074            M::Lanes(L::GetItem(ndims)) => {
1075                // Clamped for the same reason as `new_array`: an out-of-range
1076                // index count is already reported by `validate_structure`.
1077                self.pop_typed_n(tag::INT, (*ndims).min(MAX_GET_ITEM_DIMS));
1078                self.pop_typed(tag::ARRAY_REF);
1079                // Element type is not tracked; assume float.
1080                self.push(tag::FLOAT, None);
1081            }
1082
1083            // detectors / observables
1084            M::Lanes(L::SetDetector) => {
1085                self.pop_typed(tag::ARRAY_REF);
1086                self.push(tag::DETECTOR_REF, None);
1087            }
1088            M::Lanes(L::SetObservable) => {
1089                self.pop_typed(tag::ARRAY_REF);
1090                self.push(tag::OBSERVABLE_REF, None);
1091            }
1092
1093            // control
1094            // `ret <keep>` keeps the top `keep` values as the function's
1095            // return values; it does not pop exactly one. Modelling it as one
1096            // made `ret 0` — the spelling every function the compiler emits
1097            // uses — report a spurious underflow on an empty stack.
1098            //
1099            // Bounded by the stack depth plus one: `keep` is an operand, and
1100            // the extra pop is what records the single underflow when it
1101            // exceeds what the function has.
1102            M::Cpu(C::Return(keep)) => {
1103                for _ in 0..(*keep as usize).min(self.stack.len() + 1) {
1104                    self.pop_any();
1105                }
1106            }
1107            M::Cpu(C::Halt) => {}
1108
1109            // Every other vihaco-cpu op — arithmetic, comparisons, control
1110            // flow, the heap ops — is reachable in a decoded program but is
1111            // never emitted by the lanes pipeline, so its stack effect is not
1112            // modelled here. Treating it as a no-op means the simulator does
1113            // not invent underflows for code it does not understand.
1114            M::Cpu(_) => {}
1115        }
1116    }
1117
1118    fn run(mut self, program: &Program) -> Vec<ValidationError> {
1119        // Simulate each function from an empty stack, independently.
1120        //
1121        // This used to walk the whole stream and `break` at the first control
1122        // flow, which lost two things. A `call` is the *only* way to reach a
1123        // second function, so breaking there meant every multi-function
1124        // program went unchecked from its first call onwards — a callee's
1125        // unconditional underflow, behind no branch at all, was never
1126        // reported. And with no control flow anywhere the markers fell through
1127        // as no-ops, so one function's leftover operands were still on the
1128        // stack when the next began, and its errors surfaced an instruction
1129        // late or not at all.
1130        //
1131        // A branch still stops the *containing* function: past it the stack
1132        // state depends on which edge was taken, and merging those needs the
1133        // CFG walk tracked in
1134        // <https://github.com/QuEraComputing/bloqade-lanes/issues/1042>. The
1135        // linear prefix of every function is checked, which is all of every
1136        // function the compiler emits today.
1137        //
1138        // Starting from empty is the part that is only right for a function
1139        // nothing passes operands to. `call <arity>` does not clear the stack:
1140        // it sets `base = stack.len() - arity`, and the callee's locals *alias*
1141        // the operand stack from there up. So a callee with nonzero arity
1142        // legitimately consumes values this simulation cannot see, and every
1143        // such function reported a `StackUnderflow` that execution disproves.
1144        //
1145        // Skipping them trades coverage for correctness. Checking them properly
1146        // means modelling the frame rather than absolute depth — a callee owns
1147        // exactly `arity` values and popping below `base` corrupts its caller —
1148        // which is the frame-aware walk in #1042, not something this linear
1149        // pass can express. This is a placeholder for that, not the end state.
1150        let takes_operands = callee_arities(&program.code);
1151        for span in function_spans(&program.code) {
1152            if takes_operands.contains(&(span.start as u32)) {
1153                continue;
1154            }
1155            self.stack.clear();
1156            for (pc, inst) in program.code[span.start..span.end]
1157                .iter()
1158                .enumerate()
1159                .map(|(i, inst)| (span.start + i, inst))
1160            {
1161                if is_control_flow(inst) {
1162                    break;
1163                }
1164                self.pc = pc;
1165                self.dispatch(inst);
1166            }
1167        }
1168        self.errors
1169    }
1170}

1172/// Run the type-level stack simulation over a program. Collects underflow and
1173/// type-mismatch errors, plus lane/location group errors (validated against
1174/// `arch` when provided, else duplicate-only).
1175///
1176/// Conservative around control flow: each function is simulated from an empty
1177/// stack up to its first branch or call, and a function some `call` passes
1178/// operands to is skipped entirely. An empty result therefore means "nothing
1179/// this pass can see is wrong", not "the stack discipline is sound" — see the
1180/// module docs.
1181pub fn simulate_stack(program: &Program, arch: Option<&ArchSpec>) -> Vec<ValidationError> {
1182    StackSimulator::new(arch).run(program)
1183}

1754    /// Each function is simulated from an empty stack, independently.

1756    /// Breaking at the first control flow lost every multi-function program's

1765        let src = "sst v1\n\n.section(root):\n.header(root):\nversion 1.0\n.header(root).\n                   .text(root):\nfn @main() {\n  lanes::lanes.const_zone 0x00000000\n                     cpu::cpu.call 0, helper\n  cpu::cpu.halt\n}\n                   fn @helper() {\n  lanes::lanes.pop\n  cpu::cpu.ret 0\n}\n                   .text(root).\n.section(root).\n";

1776        let src = "sst v1\n\n.section(root):\n.header(root):\nversion 1.0\n.header(root).\n                   .text(root):\nfn @main() {\n  lanes::lanes.const_zone 0x00000000\n                     cpu::cpu.halt\n}\n                   fn @helper() {\n  lanes::lanes.cz\n  cpu::cpu.ret 0\n}\n                   .text(root).\n.section(root).\n";
