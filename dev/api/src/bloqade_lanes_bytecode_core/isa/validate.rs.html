1//! Architecture-dependent validation for the vihaco-backed ISA.
2//!
3//! These checks gate a [`Program`] against an [`ArchSpec`]. They are
4//! **arch-dependent by design**: when no arch spec is supplied ([`validate`]
5//! called with `None`), every check here is skipped and an empty error list is
6//! returned — an arch-free program is considered fine at this layer.
7//!
8//! [`validate`] runs the arch-dependent capability + address checks below.
9//! [`validate_structure`] adds arch-independent structural checks, and
10//! [`simulate_stack`] adds optional stack-type simulation (underflow, type
11//! mismatches, and lane/location group validation).
12//!
13//! ## Capability checks
14//!
15//! - **`feed_forward` → multiple measurements.** Without mid-circuit classical
16//!   feedback the hardware can only run straight-line code, so at most one
17//!   `measure` may appear.
18//!
19//!   The companion rule against branch/call instructions
20//!   ([`ControlFlowRequiresFeedForward`](ValidationError::ControlFlowRequiresFeedForward))
21//!   is currently unreachable — the ISA has no such instructions to reject. See
22//!   that variant's docs.
23//! - **`atom_reloading` → `fill`.** Without atom reloading, refilling atoms
24//!   after the initial fill is unsupported.
25//!
26//! ## Address checks
27//!
28//! Every `const_loc` / `const_lane` / `const_zone` operand is decoded and
29//! checked against the architecture's topology via
30//! [`ArchSpec::check_location`] / [`check_lane`](ArchSpec::check_lane) /
31//! [`check_zone`](ArchSpec::check_zone) — invalid zones, words, sites, lanes,
32//! and AOD constraints are reported with the arch layer's own message.
33

61/// An arch-dependent validation failure, tagged with the offending
62/// instruction's program counter.
63#[derive(Debug, Clone, PartialEq, Eq)]
64pub enum ValidationError {
65    /// A control-flow instruction appears but `feed_forward` is disabled.
66    ///
67    /// Currently unreachable: the ISA has no branch/call instructions to
68    /// trigger it. It reached programs while the ISA nested vihaco-cpu's
69    /// instruction set wholesale, which brought `br`/`cond_br`/`call` along;
70    /// vihaco-cpu 0.4 dropped its binary codec and the nesting went with it.
71    /// Retained because `feed_forward` is a real capability and control flow is
72    /// expected back — and because it is mapped to a Python exception, so
73    /// removing it would be a breaking API change.
74    ControlFlowRequiresFeedForward {
75        pc: usize,
76        /// The offending mnemonic (e.g. `"cond_br"`).
77        mnemonic: &'static str,
78    },
79    /// More than one `measure` appears but `feed_forward` is disabled.
80    MultipleMeasuresRequireFeedForward { pc: usize },
81    /// A `fill` appears but `atom_reloading` is disabled.
82    FillRequiresAtomReloading { pc: usize },
83    /// A `const_loc` operand does not name a valid location in the arch spec.
84    InvalidLocation { pc: usize, message: String },
85    /// A `const_lane` operand does not name a valid lane in the arch spec.
86    InvalidLane { pc: usize, message: String },
87    /// A `const_zone` operand does not name a valid zone in the arch spec.
88    InvalidZone { pc: usize, message: String },
89
90    // ---- structural (arch-independent) ----
91    /// `new_array` dim0 must be greater than zero.
92    NewArrayZeroDim0 { pc: usize },
93    /// `new_array` type_tag exceeds the maximum value tag.
94    NewArrayInvalidTypeTag { pc: usize, type_tag: u32 },
95    /// `new_array` declares more elements than [`MAX_ARRAY_ELEMENTS`].
96    NewArrayTooManyElements { pc: usize, count: u64 },
97    /// `get_item` takes an index count outside `1..=`[`MAX_GET_ITEM_DIMS`].
98    GetItemInvalidDims { pc: usize, ndims: u32 },
99    /// `load`/`store` names a local past [`MAX_LOCAL_INDEX`].
100    LocalIndexOutOfRange {
101        pc: usize,
102        /// The offending mnemonic (`"load"` or `"store"`).
103        mnemonic: &'static str,
104        index: u32,
105    },
106    /// `initial_fill` is not the first non-constant instruction.
107    InitialFillNotFirst { pc: usize },
108    /// The program has no instructions (and therefore no terminator).
109    EmptyProgram,
110    /// The final instruction is neither `return` nor `halt`.
111    MissingTerminator { pc: usize },
112    /// An instruction follows a `return`/`halt` and is unreachable.
113    UnreachableInstruction { pc: usize },
114
115    // ---- stack-type simulation (only via `simulate_stack`) ----
116    /// An instruction popped from an empty stack.
117    StackUnderflow { pc: usize },
118    /// A popped value had the wrong type tag (see [`tag`]).
119    TypeMismatch { pc: usize, expected: u8, got: u8 },
120    /// A `local_r`/`local_rz`/`fill`/`initial_fill` location group is invalid.
121    LocationGroupValidation {
122        pc: usize,
123        error: LocationGroupError,
124    },
125    /// A `move` lane group is invalid (duplicate, inconsistent, AOD, …).
126    LaneGroupValidation { pc: usize, error: LaneGroupError },
127}
128
129/// Maximum valid `new_array` element type tag ([`tag::MEASUREMENT_RESULT`]).
130const MAX_TYPE_TAG: u32 = tag::MEASUREMENT_RESULT as u32;
131
132/// Maximum number of elements a `new_array` may declare.
133///
134/// `dim0` and `dim1` are attacker-controlled `u32`s read straight out of the
135/// instruction word, and their product drives a pop loop. Without a bound, a
136/// 28-byte program can make the validator pop four billion times, and the
137/// product itself overflows `u32`. An array holds one element per measured
138/// site, so a million is already orders of magnitude past any physical
139/// architecture — the bound exists to make a malformed word a diagnosis
140/// rather than a hang.
141pub const MAX_ARRAY_ELEMENTS: u64 = 1 << 20;
142
143/// Maximum number of indices a `get_item` may take.
144///
145/// `new_array` carries exactly two dimension fields, so an array is at most
146/// 2-D and one or two indices is the only well-formed shape. The Python
147/// `stack_move.GetItem` documents the same invariant and defers enforcement
148/// here.
149pub const MAX_GET_ITEM_DIMS: u32 = 2;
150
151/// Highest local index a `load` or `store` may name, so a frame addresses at
152/// most 1024 locals.
153///
154/// vihaco's locals are a window into the operand stack starting at the current
155/// frame's base, and `store` *grows the stack to reach its index*: `op_store`
156/// calls `get_local_mut(index)`, which `resize`s to `base + index + 1` and
157/// writes `Undefined` into every new slot. The index is therefore a memory
158/// request read straight out of the instruction word — at the 16 bytes per slot
159/// measured in #1032, `store u64, 4294967295` touches about 68 GB from a
160/// 12-byte program. And because `resize` writes rather than reserves, the pages
161/// are resident on every platform, so this does not hide behind macOS's lazy
162/// commit the way the earlier `Vec::with_capacity` bounds did.
163///
164/// A lanes function's locals are its arguments, and the pipeline emits no
165/// `load`/`store` at all, so every legitimate index is a handful. The bound is
166/// set well above that rather than at it: its job is to keep a malformed
167/// operand from becoming an allocation, not to impose a calling convention on a
168/// hand-written program. The ceiling costs about 16 KB of operand stack, which
169/// is nothing, while still being three orders of magnitude past any function
170/// the compiler will emit.
171pub const MAX_LOCAL_INDEX: u32 = 1023;
172
173/// Element count of a `new_array`, in `u64` so the product cannot overflow.
174/// `dim1 == 0` means a 1-D array.
175pub(crate) fn array_element_count(dim0: u32, dim1: u32) -> u64 {
176    dim0 as u64 * if dim1 == 0 { 1 } else { dim1 as u64 }
177}
178
179/// Record a `load`/`store` local index past [`MAX_LOCAL_INDEX`].
180fn check_local_index(
181    errors: &mut Vec<ValidationError>,
182    pc: usize,
183    mnemonic: &'static str,
184    index: u32,
185) {
186    if index > MAX_LOCAL_INDEX {
187        errors.push(ValidationError::LocalIndexOutOfRange {
188            pc,
189            mnemonic,
190            index,
191        });
192    }
193}

275/// Validate a program's arch-dependent constraints (capabilities + addresses).
276///
277/// When `arch` is `None`, all checks are skipped and an empty list is returned.
278/// Otherwise every violation is collected in program order (the validator does
279/// not stop at the first error), so callers can report them all at once.
280pub fn validate(program: &Program, arch: Option<&ArchSpec>) -> Vec<ValidationError> {
281    let Some(arch) = arch else {
282        return Vec::new();
283    };
284
285    let mut errors = Vec::new();
286    let mut measure_count = 0u32;
287
288    for (pc, inst) in program.code.iter().enumerate() {
289        match inst {
290            // ---- capability checks ----
291            // No `ControlFlowRequiresFeedForward` arm: the ISA has no
292            // branch/call instructions to check for (see the variant's docs).
293            M::Lanes(L::Measure(_)) => {
294                measure_count += 1;
295                if !arch.feed_forward && measure_count > 1 {
296                    errors.push(ValidationError::MultipleMeasuresRequireFeedForward { pc });
297                }
298            }
299            M::Lanes(L::Fill(_)) if !arch.atom_reloading => {
300                errors.push(ValidationError::FillRequiresAtomReloading { pc });
301            }
302
303            // ---- address checks ----
304            M::Lanes(L::ConstLoc(bits)) => {
305                if let Some(message) = arch.check_location(&LocationAddr::decode(*bits)) {
306                    errors.push(ValidationError::InvalidLocation { pc, message });
307                }
308            }
309            M::Lanes(L::ConstLane(bits)) => {
310                for message in arch.check_lane(&LaneAddr::decode_u64(*bits)) {
311                    errors.push(ValidationError::InvalidLane { pc, message });
312                }
313            }
314            M::Lanes(L::ConstZone(bits)) => {
315                if let Some(message) = arch.check_zone(&ZoneAddr::decode(*bits)) {
316                    errors.push(ValidationError::InvalidZone { pc, message });
317                }
318            }
319
320            _ => {}
321        }
322    }
323
324    errors
325}

327/// True if `inst` terminates execution (`return` or `halt`).
328fn is_terminator(inst: &M) -> bool {
329    matches!(inst, M::Cpu(C::Return(_)) | M::Cpu(C::Halt))
330}

332/// True if `inst` only pushes a constant (and so may precede `initial_fill`).
333fn is_constant_push(inst: &M) -> bool {
334    matches!(
335        inst,
336        M::Lanes(L::ConstLoc(_))
337            | M::Lanes(L::ConstLane(_))
338            | M::Lanes(L::ConstZone(_))
339            | M::Cpu(C::Const(Type::F64, Value::F64(_)))
340            | M::Cpu(C::Const(Type::I64, Value::I64(_)))
341    )
342}

344/// Validate a program's arch-independent structural rules: `new_array` operand
345/// bounds, `initial_fill` ordering, and terminator/reachability. These never
346/// consult an arch spec, so they always run.
347pub fn validate_structure(program: &Program) -> Vec<ValidationError> {
348    let mut errors = Vec::new();
349    let mut seen_non_constant = false;
350
351    for (pc, inst) in program.code.iter().enumerate() {
352        match inst {
353            M::Lanes(L::NewArray(type_tag, dim0, dim1)) => {
354                if *dim0 == 0 {
355                    errors.push(ValidationError::NewArrayZeroDim0 { pc });
356                }
357                if *type_tag > MAX_TYPE_TAG {
358                    errors.push(ValidationError::NewArrayInvalidTypeTag {
359                        pc,
360                        type_tag: *type_tag,
361                    });
362                }
363                // Bound the element count here, where the operands are read,
364                // so neither the stack simulator nor the machine has to guard
365                // its own pop loop against a four-billion-element array.
366                let count = array_element_count(*dim0, *dim1);
367                if count > MAX_ARRAY_ELEMENTS {
368                    errors.push(ValidationError::NewArrayTooManyElements { pc, count });
369                }
370                seen_non_constant = true;
371            }
372            M::Lanes(L::GetItem(ndims)) => {
373                if *ndims == 0 || *ndims > MAX_GET_ITEM_DIMS {
374                    errors.push(ValidationError::GetItemInvalidDims { pc, ndims: *ndims });
375                }
376                seen_non_constant = true;
377            }
378            // Bound the local index here, where the operand is read, so a
379            // `store` cannot turn a one-word instruction into a multi-gigabyte
380            // stack `resize` inside vihaco (see [`MAX_LOCAL_INDEX`]). `load`
381            // only reads, so it fails cleanly either way — it is checked too
382            // because a program naming a local that far out is malformed
383            // regardless of which end of it is reached first.
384            M::Cpu(C::Load(_, index)) => {
385                check_local_index(&mut errors, pc, "load", *index);
386                seen_non_constant = true;
387            }
388            M::Cpu(C::Store(_, index)) => {
389                check_local_index(&mut errors, pc, "store", *index);
390                seen_non_constant = true;
391            }
392            M::Lanes(L::InitialFill(_)) => {
393                if seen_non_constant {
394                    errors.push(ValidationError::InitialFillNotFirst { pc });
395                }
396                seen_non_constant = true;
397            }
398            inst if is_constant_push(inst) => {}
399            _ => seen_non_constant = true,
400        }
401    }
402
403    // Any instruction after the first terminator is unreachable. If there are
404    // unreachable instructions they explain a non-terminal last instruction, so
405    // `MissingTerminator` would be a redundant second error.
406    let mut found_terminator = false;
407    let mut unreachable = Vec::new();
408    for (pc, inst) in program.code.iter().enumerate() {
409        if found_terminator {
410            unreachable.push(ValidationError::UnreachableInstruction { pc });
411        }
412        if is_terminator(inst) {
413            found_terminator = true;
414        }
415    }
416
417    if unreachable.is_empty() {
418        match program.code.last() {
419            None => errors.push(ValidationError::EmptyProgram),
420            Some(last) if !is_terminator(last) => errors.push(ValidationError::MissingTerminator {
421                pc: program.code.len() - 1,
422            }),
423            Some(_) => {}
424        }
425    } else {
426        errors.extend(unreachable);
427    }
428
429    errors
430}

434/// One tracked stack value: its type tag and (when known) concrete bits.
435#[derive(Debug, Clone)]
436struct SimEntry {
437    tag: u8,
438    value: Option<u64>,
439}

441/// Type-level stack simulator: walks the instruction stream tracking value
442/// types, reporting underflow and type mismatches, and — when given an
443/// [`ArchSpec`] — validating `move` lane groups and `fill`/`local_*` location
444/// groups (duplicates only without an arch).
445struct StackSimulator<'a> {
446    stack: Vec<SimEntry>,
447    errors: Vec<ValidationError>,
448    arch: Option<&'a ArchSpec>,
449    pc: usize,
450}
451
452impl<'a> StackSimulator<'a> {
453    fn new(arch: Option<&'a ArchSpec>) -> Self {
454        Self {
455            stack: Vec::new(),
456            errors: Vec::new(),
457            arch,
458            pc: 0,
459        }
460    }
461
462    /// Record that a pop found the stack empty.
463    ///
464    /// `StackUnderflow` carries nothing but the `pc`, so a second one for the
465    /// same instruction is a literally identical value and says nothing new.
466    /// Group pops (`fill 40`, `new_array`) would otherwise emit one per
467    /// missing operand — a million lines of `pc 0: stack underflow` for a
468    /// single malformed word. All pops for one instruction are consecutive, so
469    /// checking the last error is enough.
470    fn underflow(&mut self) {
471        let already = matches!(
472            self.errors.last(),
473            Some(ValidationError::StackUnderflow { pc }) if *pc == self.pc
474        );
475        if !already {
476            self.errors
477                .push(ValidationError::StackUnderflow { pc: self.pc });
478        }
479    }
480
481    fn pop_any(&mut self) {
482        if self.stack.pop().is_none() {
483            self.underflow();
484        }
485    }
486
487    fn pop_typed(&mut self, expected: u8) {
488        match self.stack.pop() {
489            Some(entry) if entry.tag != expected => {
490                self.errors.push(ValidationError::TypeMismatch {
491                    pc: self.pc,
492                    expected,
493                    got: entry.tag,
494                })
495            }
496            Some(_) => {}
497            None => self.underflow(),
498        }
499    }
500
501    fn pop_typed_n(&mut self, expected: u8, count: u32) {
502        for _ in 0..count {
503            self.pop_typed(expected);
504        }
505    }
506
507    /// Pop one value expected to have `expected` tag, returning its concrete
508    /// bits when the type matches.
509    fn pop_addr(&mut self, expected: u8) -> Option<u64> {
510        match self.stack.pop() {
511            Some(entry) if entry.tag == expected => entry.value,
512            Some(entry) => {
513                self.errors.push(ValidationError::TypeMismatch {
514                    pc: self.pc,
515                    expected,
516                    got: entry.tag,
517                });
518                None
519            }
520            None => {
521                self.underflow();
522                None
523            }
524        }
525    }
526
527    fn push(&mut self, tag: u8, value: Option<u64>) {
528        self.stack.push(SimEntry { tag, value });
529    }
530
531    fn sim_dup(&mut self) {
532        if let Some(top) = self.stack.last().cloned() {
533            self.stack.push(top);
534        } else {
535            self.underflow();
536        }
537    }
538
539    fn sim_swap(&mut self) {
540        let len = self.stack.len();
541        if len >= 2 {
542            self.stack.swap(len - 1, len - 2);
543        } else {
544            self.underflow();
545        }
546    }
547
548    /// Report each uniquely-duplicated location once (no-arch fallback).
549    fn check_duplicate_locations(&mut self, locations: &[LocationAddr]) {
550        let mut seen = HashSet::new();
551        let mut reported = HashSet::new();
552        for loc in locations {
553            let bits = loc.encode();
554            if !seen.insert(bits) && reported.insert(bits) {
555                self.errors.push(ValidationError::LocationGroupValidation {
556                    pc: self.pc,
557                    error: LocationGroupError::DuplicateAddress { address: bits },
558                });
559            }
560        }
561    }
562
563    /// Report each uniquely-duplicated lane once (no-arch fallback).
564    fn check_duplicate_lanes(&mut self, lanes: &[LaneAddr]) {
565        let mut seen = HashSet::new();
566        let mut reported = HashSet::new();
567        for lane in lanes {
568            let (d0, d1) = lane.encode();
569            let combined = (d0 as u64) | ((d1 as u64) << 32);
570            if !seen.insert(combined) && reported.insert(combined) {
571                self.errors.push(ValidationError::LaneGroupValidation {
572                    pc: self.pc,
573                    error: LaneGroupError::DuplicateAddress { address: (d0, d1) },
574                });
575            }
576        }
577    }
578
579    /// Pop `arity` locations and validate them as a group.
580    fn pop_and_validate_locations(&mut self, arity: u32) {
581        let bits: Vec<Option<u64>> = (0..arity).map(|_| self.pop_addr(tag::LOCATION)).collect();
582        let locations: Vec<LocationAddr> = bits
583            .iter()
584            .filter_map(|v| v.map(LocationAddr::decode))
585            .collect();
586        let pc = self.pc;
587        if let Some(arch) = self.arch {
588            for error in arch.check_locations(&locations) {
589                self.errors
590                    .push(ValidationError::LocationGroupValidation { pc, error });
591            }
592        } else {
593            self.check_duplicate_locations(&locations);
594        }
595    }
596
597    /// Pop `arity` lanes and validate them as a group.
598    fn sim_move(&mut self, arity: u32) {
599        let bits: Vec<Option<u64>> = (0..arity).map(|_| self.pop_addr(tag::LANE)).collect();
600        let lanes: Vec<LaneAddr> = bits
601            .iter()
602            .filter_map(|v| v.map(LaneAddr::decode_u64))
603            .collect();
604        let pc = self.pc;
605        if let Some(arch) = self.arch {
606            for error in arch.check_lanes(&lanes) {
607                self.errors
608                    .push(ValidationError::LaneGroupValidation { pc, error });
609            }
610        } else {
611            self.check_duplicate_lanes(&lanes);
612        }
613    }
614
615    fn dispatch(&mut self, inst: &M) {
616        match inst {
617            // constants push a typed value
618            M::Cpu(C::Const(Type::F64, Value::F64(v))) => self.push(tag::FLOAT, Some(v.to_bits())),
619            M::Cpu(C::Const(Type::I64, Value::I64(v))) => self.push(tag::INT, Some(*v as u64)),
620            M::Lanes(L::ConstLoc(v)) => self.push(tag::LOCATION, Some(*v)),
621            M::Lanes(L::ConstLane(v)) => self.push(tag::LANE, Some(*v)),
622            M::Lanes(L::ConstZone(v)) => self.push(tag::ZONE, Some(*v as u64)),
623
624            // stack manipulation
625            M::Lanes(L::Pop) => self.pop_any(),
626            M::Cpu(C::Dup) => self.sim_dup(),
627            M::Lanes(L::Swap) => self.sim_swap(),
628
629            // atom arrangement
630            M::Lanes(L::InitialFill(arity)) | M::Lanes(L::Fill(arity)) => {
631                self.pop_and_validate_locations(*arity)
632            }
633            M::Lanes(L::Move(arity)) => self.sim_move(*arity),
634
635            // gates
636            M::Lanes(L::LocalR(arity)) => {
637                self.pop_typed_n(tag::FLOAT, 2);
638                self.pop_and_validate_locations(*arity);
639            }
640            M::Lanes(L::LocalRz(arity)) => {
641                self.pop_typed_n(tag::FLOAT, 1);
642                self.pop_and_validate_locations(*arity);
643            }
644            M::Lanes(L::GlobalR) => self.pop_typed_n(tag::FLOAT, 2),
645            M::Lanes(L::GlobalRz) => self.pop_typed_n(tag::FLOAT, 1),
646            M::Lanes(L::Cz) => self.pop_typed(tag::ZONE),
647
648            // measurement
649            M::Lanes(L::Measure(arity)) => {
650                self.pop_typed_n(tag::ZONE, *arity);
651                for _ in 0..*arity {
652                    self.push(tag::MEASURE_FUTURE, None);
653                }
654            }
655            // `await_measure` yields an *array* of measurement results — the
656            // same value `set_detector`/`set_observable` consume, and the type
657            // `stack_move.AwaitMeasure` is declared to produce. The distinct
658            // `MEASUREMENT_RESULT` tag is that array's *element* type, which a
659            // one-tag-per-slot simulator cannot express; it exists so
660            // `new_array` can name the element type (#547). Giving the whole
661            // array that tag broke `measure -> await_measure -> set_detector`.
662            M::Lanes(L::AwaitMeasure) => {
663                self.pop_typed(tag::MEASURE_FUTURE);
664                self.push(tag::ARRAY_REF, None);
665            }
666
667            // arrays
668            M::Lanes(L::NewArray(_type_tag, dim0, dim1)) => {
669                // `validate_structure` rejects counts past the cap, so the
670                // clamp here only keeps a rejected program from also driving
671                // an unbounded loop before its diagnosis is reported.
672                let count = array_element_count(*dim0, *dim1).min(MAX_ARRAY_ELEMENTS);
673                for _ in 0..count {
674                    self.pop_any();
675                }
676                self.push(tag::ARRAY_REF, None);
677            }
678            M::Lanes(L::GetItem(ndims)) => {
679                // Clamped for the same reason as `new_array`: an out-of-range
680                // index count is already reported by `validate_structure`.
681                self.pop_typed_n(tag::INT, (*ndims).min(MAX_GET_ITEM_DIMS));
682                self.pop_typed(tag::ARRAY_REF);
683                // Element type is not tracked; assume float.
684                self.push(tag::FLOAT, None);
685            }
686
687            // detectors / observables
688            M::Lanes(L::SetDetector) => {
689                self.pop_typed(tag::ARRAY_REF);
690                self.push(tag::DETECTOR_REF, None);
691            }
692            M::Lanes(L::SetObservable) => {
693                self.pop_typed(tag::ARRAY_REF);
694                self.push(tag::OBSERVABLE_REF, None);
695            }
696
697            // control
698            M::Cpu(C::Return(_)) => self.pop_any(),
699            M::Cpu(C::Halt) => {}
700
701            // Every other vihaco-cpu op — arithmetic, comparisons, control
702            // flow, the heap ops — is reachable in a decoded program but is
703            // never emitted by the lanes pipeline, so its stack effect is not
704            // modelled here. Treating it as a no-op means the simulator does
705            // not invent underflows for code it does not understand.
706            M::Cpu(_) => {}
707        }
708    }
709
710    fn run(mut self, program: &Program) -> Vec<ValidationError> {
711        for (pc, inst) in program.code.iter().enumerate() {
712            self.pc = pc;
713            self.dispatch(inst);
714        }
715        self.errors
716    }
717}

1270    /// new slot, so `store u64, 4294967295` touches ~68 GB from a three-

1272    /// validator names the operand — and not on the allocator refusing, because
1273    /// a `resize` this size succeeds on a lazily-committing platform and the
1274    /// two earlier bounds of this class only failed on Linux CI.

1281            // The first index past the bound, so the boundary is pinned from
1282            // both sides rather than only at an absurd operand.
