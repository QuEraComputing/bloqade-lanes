1//! The Bloqade Lanes machine: vihaco-cpu's `CPU` plus our atom-movement device.
2//!
3//! This is the idiomatic vihaco composition, and the same shape PPVM uses: each
4//! device owns its instruction set, and `#[composite]` generates the combined
5//! program type, the text parser (an alternation over the devices' parsers) and
6//! the section loaders.
7//!
8//! ```text
9//! cpu::cpu.const u64, 0      <- vihaco-cpu's device
10//! lanes::lanes.move 2        <- ours
11//! ```
12//!
13//! The `<field>::` prefix is the device field name; the second half is that
14//! device's own dialect head.
15//!
16//! ## Who does what
17//!
18//! Instruction operands live on the CPU stack, so a device never reads them
19//! directly. `resolve_lanes` pops them and packs them into a [`LanesMessage`];
20//! the device executes; the machine then applies any [`LanesEffect::Push`]
21//! back onto the stack. That round trip is the reason address constants can
22//! stay lanes instructions while still behaving like stack pushes.
23//!
24//! [`super::validate::simulate_stack`] is the static half of this: it models
25//! the same pops *and the same pushes*, so a program it accepts does not
26//! underflow on any path it checks. That is every path except those past a
27//! `call_indirect`, whose arity only the run knows, and any path merging with
28//! one (TODO(vihaco#110): close with a declared signature on `call_indirect`).
29//! The ops the device does not interpret hold up their end by pushing
30//! [`Value::Undefined`] placeholders — the depth the simulator predicts, with
31//! a value nothing can mistake for a result.

54/// `#[composite]` derives only `Debug` and `Clone` on its runtime enum, but both
55/// device instruction sets *are* `PartialEq`, so comparing programs only needs
56/// the two arms spelled out. Round-trip tests and the Python `__eq__` both rely
57/// on it.
58impl PartialEq for MachineInstruction {
59    fn eq(&self, other: &Self) -> bool {
60        match (self, other) {
61            (Self::Cpu(a), Self::Cpu(b)) => a == b,
62            (Self::Lanes(a), Self::Lanes(b)) => a == b,
63            _ => false,
64        }
65    }
66}

84/// Why [`LanesMachine::run`] stopped.
85#[derive(Debug, Clone, Copy, PartialEq, Eq)]
86pub enum Stopped {
87    /// The program executed `halt`.
88    Halted,
89    /// `@main` returned.
90    Returned,
91    /// Execution ran off the end of the code without a terminator.
92    ///
93    /// [`super::validate::validate_structure`] rejects such a program, so this
94    /// only happens when running one that was never validated.
95    RanOff,
96    /// The step budget was exhausted. A program with a backward branch can
97    /// run forever, and a machine with no budget would hang rather than say
98    /// so.
99    OutOfSteps,
100}

147    /// Run a program: dispatch each instruction to the device that owns it,
148    /// until it halts, returns, or runs out of code or budget.
149    ///
150    /// This is the piece that makes the composite a machine rather than two
151    /// devices: [`step_lanes`](Self::step_lanes) handles one lanes
152    /// instruction, `CPU::execute_instruction` one CPU instruction, and
153    /// nothing until now walked a `MachineInstruction` stream across both.
154    ///
155    /// `max_steps` bounds execution. A program with a backward branch can run
156    /// forever, and the ISA carries vihaco-cpu's control flow whether or not
157    /// the lanes compiler emits it, so the budget is a parameter rather than
158    /// an assumption.
159    ///
160    /// The entry point is entered with no arguments; one that declares
161    /// parameters needs [`run_with_args`](Self::run_with_args).
162    pub fn run(&mut self, program: &Program, max_steps: u64) -> eyre::Result<Run> {
163        self.run_with_args(program, &[], max_steps)
164    }
165
166    /// [`run`](Self::run) an entry point that takes arguments.
167    ///
168    /// `args` are pushed before the first instruction, in declaration order,
169    /// so they sit at the bottom of the entry frame as locals `0..args.len()`
170    /// — where a `call` leaves a callee's. That is the frame `simulate_stack`
171    /// assumes for every function, the entry included.
172    ///
173    /// They are checked against the entry point's declaration, count and
174    /// type, the way `validate` checks a `call` against its callee's. The host
175    /// is the entry point's only caller, and without the check a missing
176    /// argument surfaces mid-run as an out-of-bounds `load`, far from its
177    /// cause.
178    pub fn run_with_args(
179        &mut self,
180        program: &Program,
181        args: &[Value],
182        max_steps: u64,
183    ) -> eyre::Result<Run> {
184        use vihaco_cpu::StepOutcome;
185
186        let mut effects = Vec::new();
187        let mut steps = 0u64;
188        let mut pc = self.enter(program, args)?;
189
190        let stopped = loop {
191            let Some(inst) = program.code.get(pc) else {
192                break Stopped::RanOff;
193            };
194            if steps == max_steps {
195                break Stopped::OutOfSteps;
196            }
197            steps += 1;
198
199            // `op_call` reads this to work out where to return to, so it has
200            // to be set before the instruction runs, not after.
201            self.cpu.set_current_pc(pc as u32);
202
203            match inst {
204                MachineInstruction::Lanes(inst) => {
205                    collect(self.step_lanes(inst.clone())?, &mut effects);
206                }
207                MachineInstruction::Cpu(inst) => {
208                    self.guard_cpu(inst)?;
209                    match self.cpu.execute_instruction(inst.clone())? {
210                        StepOutcome::Halt => break Stopped::Halted,
211                        // `op_return` reports `Return` only when it pops the
212                        // last frame; an inner return sets the resume address
213                        // and reports `Continue`.
214                        StepOutcome::Return => break Stopped::Returned,
215                        StepOutcome::Continue | StepOutcome::Breakpoint => {}
216                    }
217                }
218            }
219
220            // A branch or call leaves its destination here; anything else
221            // falls through to the next instruction.
222            pc = match self.cpu.take_pending_pc() {
223                Some(target) => target as usize,
224                None => pc + 1,
225            };
226        };
227
228        Ok(Run {
229            effects,
230            stopped,
231            steps,
232        })
233    }

235    /// Refuse the two CPU instructions whose operands reach an allocation or a
236    /// subtraction before vihaco-cpu 0.4.1 bounds-checks them.
237    ///
238    /// Both belong upstream, and both are caught here for the same reason
239    /// [`pop_capacity`](Self::pop_capacity) clamps its reservation: `run` has
240    /// no validation gate — `validate` is its own subcommand, and a program you
241    /// have not validated is still one you may want to execute — so the
242    /// execution path has to be safe on the operands alone.
243    ///
244    /// - `store` ([#1032]): `op_store` calls `get_local_mut`, which `resize`s
245    ///   the operand stack to `base + index + 1`. The bound is the validator's
246    ///   [`MAX_LOCAL_INDEX`], so a program the validator accepts still runs.
247    /// - `ret` ([#1033]): `op_return` computes `stack.len() - frame.base`
248    ///   before comparing it. `op_call` establishes `base <= len`, but a callee
249    ///   that pops more than it pushed breaks that and the subtraction
250    ///   underflows — a panic (debug) or an out-of-range `drain` (release)
251    ///   rather than the `Err` an embedder can handle.
252    ///
253    /// Both are one comparison against state this loop already reads, and both
254    /// can go when the `=0.4.1` pin moves past an upstream fix.
255    ///
256    /// [#1032]: https://github.com/QuEraComputing/bloqade-lanes/issues/1032
257    /// [#1033]: https://github.com/QuEraComputing/bloqade-lanes/issues/1033
258    fn guard_cpu(&self, inst: &vihaco_cpu::RuntimeInstruction) -> eyre::Result<()> {
259        use vihaco_cpu::RuntimeInstruction as C;
260        match inst {
261            C::Store(_, index) if *index > MAX_LOCAL_INDEX => Err(eyre::eyre!(
262                "store names local {index}, past the maximum index of {MAX_LOCAL_INDEX}"
263            )),
264            // A missing frame is upstream's own "no frame to pop", so only the
265            // case it gets wrong is intercepted.
266            C::Return(_) => match self.cpu.get_frame() {
267                Ok(frame) if self.cpu.stack().len() < frame.base => Err(eyre::eyre!(
268                    "ret with the stack {} deep, below its frame base {}: \
269                     the callee popped past its own frame",
270                    self.cpu.stack().len(),
271                    frame.base
272                )),
273                _ => Ok(()),
274            },
275            _ => Ok(()),
276        }
277    }

279    /// Where execution begins: the `@main` *symbol*, not byte zero.
280    ///
281    /// A lanes program is an executable, so it enters the way a linked binary
282    /// does — at a named entry the loader resolves, wherever it was laid out.
283    /// Starting at address 0 instead ran whichever function came first in the
284    /// source, so a module declaring `@helper` before `@main` executed the
285    /// wrong one and reported success.
286    ///
287    /// Pushes the entry point's arguments, checked against its declared
288    /// parameters, and returns the address to start at.
289    ///
290    /// Every run starts from a reset CPU holding only the entry frame, so the
291    /// arguments land at locals `0..n` however the machine was last left: a
292    /// halt mid-callee leaves frames and operands behind, and a `ret` from
293    /// `@main` pops the entry frame outright.
294    fn enter(&mut self, program: &Program, args: &[Value]) -> eyre::Result<usize> {
295        let function = super::program::entry_function(program)?;
296
297        let params = &function.signature.params;
298        if args.len() != params.len() {
299            eyre::bail!(
300                "the entry point declares {} parameter(s), but {} argument(s) were supplied",
301                params.len(),
302                args.len()
303            );
304        }
305        for (i, (arg, param)) in args.iter().zip(params).enumerate() {
306            if arg.type_of() != param.ty {
307                eyre::bail!(
308                    "entry argument {i} is {:?}, but the entry point declares {:?}",
309                    arg.type_of(),
310                    param.ty
311                );
312            }
313        }
314        vihaco::Reset::reset(&mut self.cpu);
315        self.cpu.push_frame(entry_frame());
316        for arg in args {
317            self.cpu.stack_push(*arg);
318        }
319
320        Ok(function.start_address as usize)
321    }

323    /// Pop the operands `inst` consumes and pack them into its message.
324    ///
325    /// Pops are in reverse push order throughout: the last value pushed is the
326    /// first popped, so a group read back from the stack is reversed to restore
327    /// program order.
328    fn resolve_lanes(&mut self, inst: &LanesInstruction) -> eyre::Result<LanesMessage> {
329        use LanesInstruction as I;
330        Ok(match inst {
331            // Constants carry their operand in the instruction word.
332            I::ConstLoc(_) | I::ConstLane(_) | I::ConstZone(_) => LanesMessage::None,
333
334            // The device cannot reach the stack, so the machine does the
335            // popping for it and the pushes come back as effects.
336            I::Pop => {
337                self.cpu.stack_pop()?;
338                LanesMessage::None
339            }
340            I::Swap => LanesMessage::Values(self.pop_values(2)?),
341
342            I::InitialFill(n) | I::Fill(n) => LanesMessage::Locations(self.pop_locations(*n)?),
343            I::Move(n) => LanesMessage::Lanes(self.pop_lanes(*n)?),
344
345            // Angles sit above the locations: `local_r` pops axis then
346            // rotation, then the location group beneath them.
347            I::LocalR(n) => LanesMessage::LocalRotation {
348                angles: self.pop_floats(2)?,
349                locations: self.pop_locations(*n)?,
350            },
351            I::LocalRz(n) => LanesMessage::LocalRotation {
352                angles: self.pop_floats(1)?,
353                locations: self.pop_locations(*n)?,
354            },
355            I::GlobalR => LanesMessage::GlobalRotation {
356                angles: self.pop_floats(2)?,
357            },
358            I::GlobalRz => LanesMessage::GlobalRotation {
359                angles: self.pop_floats(1)?,
360            },
361
362            I::Cz => LanesMessage::Zones(self.pop_zones(1)?),
363            I::Measure(n) => LanesMessage::Zones(self.pop_zones(*n)?),
364
365            // `new_array` consumes dim0×dim1 elements (dim1 = 0 means 1-D);
366            // `get_item` consumes the array reference and then its indices.
367            // Both are reported rather than executed — see
368            // [`LanesEffect::NotSimulated`] — so the operands travel with the
369            // message rather than being dropped on the floor.
370            //
371            // Both counts are computed in `u64`: `dim0 * dim1` overflows `u32`
372            // for operands a decoded program is free to carry, and `n + 1`
373            // overflows for `get_item(u32::MAX)`.
374            I::NewArray(_, dim0, dim1) => {
375                LanesMessage::Values(self.pop_values(array_element_count(*dim0, *dim1))?)
376            }
377            I::GetItem(n) => LanesMessage::Values(self.pop_values(*n as u64 + 1)?),
378
379            I::AwaitMeasure | I::SetDetector | I::SetObservable => {
380                LanesMessage::Values(self.pop_values(1)?)
381            }
382        })
383    }

385    /// Pop `n` values, restoring program order (the last pushed is popped
386    /// first).
387    ///
388    /// `n` comes straight out of an instruction word, so nothing is
389    /// pre-allocated against it: an implausible count runs out of stack within
390    /// a few pops and fails there. [`super::validate::validate_structure`]
391    /// rejects such a program up front; this makes the machine safe on its own
392    /// regardless.
393    fn pop_values(&mut self, n: u64) -> eyre::Result<Vec<Value>> {
394        let mut out = Vec::with_capacity(self.pop_capacity(n));
395        for _ in 0..n {
396            out.push(self.cpu.stack_pop()?);
397        }
398        out.reverse();
399        Ok(out)
400    }
401
402    /// Capacity to reserve for a pop of `n` values.
403    ///
404    /// `n` comes straight out of an instruction word, so it cannot be used as
405    /// a capacity: `initial_fill 4294967295` asks for 48 GiB of
406    /// `LocationAddr`. Linux's allocator refuses and Rust aborts the process
407    /// (SIGABRT, no unwinding, no test failure to catch); macOS commits
408    /// lazily and hands it back, which is why this only ever showed up in CI.
409    ///
410    /// A pop can never take more than the stack holds, so the stack depth is
411    /// both a safe bound and a sufficient one — every legitimate arity still
412    /// gets its single up-front allocation.
413    fn pop_capacity(&self, n: u64) -> usize {
414        n.min(self.cpu.stack().len() as u64) as usize
415    }

417    fn pop_u64(&mut self) -> eyre::Result<u64> {
418        match self.cpu.stack_pop()? {
419            Value::U64(v) => Ok(v),
420            v => Err(eyre::eyre!("expected a packed u64 address, got {v:?}")),
421        }
422    }

424    fn pop_floats(&mut self, n: u32) -> eyre::Result<Vec<f64>> {
425        let mut out = Vec::with_capacity(self.pop_capacity(n as u64));
426        for _ in 0..n {
427            match self.cpu.stack_pop()? {
428                Value::F64(v) => out.push(v),
429                v => return Err(eyre::eyre!("expected an angle (f64), got {v:?}")),
430            }
431        }
432        out.reverse();
433        Ok(out)
434    }

466    /// Apply a device effect. Only [`LanesEffect::Push`] touches the machine;
467    /// the rest are observations for the caller.
468    fn apply(&mut self, effects: &Effects<LanesEffect>) -> eyre::Result<()> {
469        let mut push = |effect: &LanesEffect| {
470            if let LanesEffect::Push(value) = effect {
471                self.cpu.stack_push(*value);
472            }
473        };
474        match effects {
475            Effects::None => {}
476            Effects::One(effect) => push(effect),
477            Effects::Many(effects) => effects.iter().for_each(push),
478        }
479        Ok(())
480    }

482    /// Run one lanes instruction end to end: pop its operands, execute, then
483    /// apply whatever comes back.
484    pub fn step_lanes(&mut self, inst: LanesInstruction) -> eyre::Result<Effects<LanesEffect>> {
485        let msg = self.resolve_lanes(&inst)?;
486        let effects = self.lanes.execute_generated(inst, msg)?;
487        self.apply(&effects)?;
488        Ok(effects)
489    }

508/// vihaco-cpu's own `Display` emits bare mnemonics (`halt`, `const.f64 1.5`)
509/// that its *parser* does not accept, so rendering is written here against the
510/// surface grammar instead. The round-trip tests pin the two together.
511pub(super) fn cpu_type_text(ty: Type) -> &'static str {
512    match ty {
513        Type::Undefined => "undef",
514        Type::String => "str",
515        Type::Bool => "bool",
516        Type::I64 => "i64",
517        Type::U32 => "u32",
518        Type::U64 => "u64",
519        Type::F64 => "f64",
520        Type::FunctionRef => "fn_ref",
521        Type::HeapRef => "heap_ref",
522    }
523}

525fn cpu_value_text(value: &Value) -> String {
526    match value {
527        // `{:?}` on f64 is round-trip exact; `{}` drops the `.0` on integral
528        // floats, which would then re-parse as an integer.
529        Value::F64(v) => format!("{v:?}"),
530        Value::Bool(v) => v.to_string(),
531        Value::I64(v) => v.to_string(),
532        Value::U32(v) => v.to_string(),
533        Value::U64(v) => v.to_string(),
534        Value::String(v) | Value::FunctionRef(v) | Value::HeapRef(v) => v.to_string(),
535        Value::Undefined => "undef".to_string(),
536    }
537}

539fn cpu_text(inst: &vihaco_cpu::RuntimeInstruction) -> String {
540    use vihaco_cpu::RuntimeInstruction as C;
541    let typed = |op: &str, ty: Type| format!("{op} {}", cpu_type_text(ty));
542    match inst {
543        C::Span(a, b, c) => format!("span {a} {b} {c}"),
544        C::Label(name) => format!("label @{}", name.as_str()),
545        C::FunctionStart => "func_start".into(),
546        C::FunctionEnd => "func_end".into(),
547        C::Breakpoint => "breakpoint".into(),
548        C::Branch(t) => format!("br @{t}"),
549        C::ConditionalBranch(t, f) => format!("cond_br @{t}, @{f}"),
550        C::Return(n) => format!("ret {n}"),
551        C::IndirectCall => "call_indirect".into(),
552        C::Call(arity, target) => format!("call {arity}, {target}"),
553        C::Halt => "halt".into(),
554        C::Print => "print".into(),
555        C::Load(ty, addr) => format!("load {}, {addr}", cpu_type_text(*ty)),
556        C::Store(ty, addr) => format!("store {}, {addr}", cpu_type_text(*ty)),
557        C::Dup => "dup".into(),
558        C::HeapAlloc(n) => format!("heap_alloc {n}"),
559        C::GetItem => "get_item".into(),
560        C::HeapDealloc => "heap_dealloc".into(),
561        C::Const(ty, v) => format!("const {}, {}", cpu_type_text(*ty), cpu_value_text(v)),
562        C::Add(t) => typed("add", *t),
563        C::Sub(t) => typed("sub", *t),
564        C::Mul(t) => typed("mul", *t),
565        C::Div(t) => typed("div", *t),
566        C::Rem(t) => typed("rem", *t),
567        C::Neg(t) => typed("neg", *t),
568        C::Shl(t) => typed("shl", *t),
569        C::Shr(t) => typed("shr", *t),
570        C::Rol(t) => typed("rol", *t),
571        C::Ror(t) => typed("ror", *t),
572        C::BitAnd(t) => typed("bitand", *t),
573        C::BitOr(t) => typed("bitor", *t),
574        C::BitXor(t) => typed("bitxor", *t),
575        C::Not => "not".into(),
576        C::And => "and".into(),

749/// Lower a parsed vihaco-cpu instruction to its runtime form.
750///
751/// The surface form is lexical — `SurfaceValue` is a raw token and branch
752/// targets are identifiers — so this is where a constant becomes a typed
753/// [`Value`].
754///
755/// What this accepts is the counterpart of what [`to_sst_text`] emits: a
756/// decoded binary may hold any of vihaco's nine `const` types, so all nine
757/// parse back. The two rejections both need something a per-instruction
758/// lowering does not have — a symbol table, for symbolic control flow, and a
759/// string table, for a quoted literal.
760fn lower_cpu(inst: CpuSurfaceInstruction) -> eyre::Result<vihaco_cpu::RuntimeInstruction> {
761    use CpuSurfaceInstruction as S;
762    use vihaco_cpu::RuntimeInstruction as R;
763
764    let ty = |t: vihaco_cpu::SurfaceType| -> Type {
765        match t {
766            vihaco_cpu::SurfaceType::Undefined => Type::Undefined,
767            vihaco_cpu::SurfaceType::String => Type::String,
768            vihaco_cpu::SurfaceType::Bool => Type::Bool,
769            vihaco_cpu::SurfaceType::I64 => Type::I64,
770            vihaco_cpu::SurfaceType::U32 => Type::U32,
771            vihaco_cpu::SurfaceType::U64 => Type::U64,
772            vihaco_cpu::SurfaceType::F64 => Type::F64,
773            vihaco_cpu::SurfaceType::FunctionRef => Type::FunctionRef,
774            vihaco_cpu::SurfaceType::HeapRef => Type::HeapRef,
775        }
776    };
777
778    Ok(match inst {
779        S::Const(t, v) => {
780            let ty = ty(t);
781            // A quoted literal would have to be interned to become a
782            // `Value::String`, and interning needs the module's string table —
783            // which a per-instruction lowering does not have. `const str` takes
784            // the interner index instead, which is what the value holds and
785            // what `cpu_value_text` renders.
786            let text = match &v {
787                vihaco_cpu::SurfaceValue::Bare(token) => token.0.clone(),
788                vihaco_cpu::SurfaceValue::Quoted(_) => {
789                    return Err(eyre::eyre!(
790                        "a quoted string literal needs the module's string table; \
791                         write the interner index instead"
792                    ));
793                }
794            };
795            // Every type the renderer can emit must parse back, or
796            // `disassemble` produces text `assemble` rejects. The three
797            // reference types and `String` are interner/heap indices, so they
798            // read as plain integers.
799            let value = match ty {
800                Type::F64 => Value::F64(text.parse()?),
801                Type::I64 => Value::I64(text.parse()?),
802                Type::U64 => Value::U64(text.parse()?),
803                Type::U32 => Value::U32(text.parse()?),
804                Type::Bool => Value::Bool(text.parse()?),
805                Type::String => Value::String(text.parse()?),
806                Type::FunctionRef => Value::FunctionRef(text.parse()?),
807                Type::HeapRef => Value::HeapRef(text.parse()?),
808                Type::Undefined if text == "undef" => Value::Undefined,
809                Type::Undefined => {
810                    return Err(eyre::eyre!("const undef takes no value but got '{text}'"));
811                }
812            };
813            R::Const(ty, value)
814        }
815        S::Span(a, b, c) => R::Span(a, b, c),
816        S::FunctionStart => R::FunctionStart,
817        S::FunctionEnd => R::FunctionEnd,
818        S::Breakpoint => R::Breakpoint,
819        S::Return(n) => R::Return(n),
820        S::IndirectCall => R::IndirectCall,
821        S::Halt => R::Halt,
822        S::Print => R::Print,
823        S::Load(t, addr) => R::Load(ty(t), addr),
824        S::Store(t, addr) => R::Store(ty(t), addr),
825        S::Dup => R::Dup,
826        S::HeapAlloc(n) => R::HeapAlloc(n),
827        S::GetItem => R::GetItem,
828        S::HeapDealloc => R::HeapDealloc,
829        S::Add(t) => R::Add(ty(t)),
830        S::Sub(t) => R::Sub(ty(t)),
831        S::Mul(t) => R::Mul(ty(t)),
832        S::Div(t) => R::Div(ty(t)),
833        S::Rem(t) => R::Rem(ty(t)),
834        S::Neg(t) => R::Neg(ty(t)),
835        S::Shl(t) => R::Shl(ty(t)),
836        S::Shr(t) => R::Shr(ty(t)),
837        S::Rol(t) => R::Rol(ty(t)),
838        S::Ror(t) => R::Ror(ty(t)),
839        S::BitAnd(t) => R::BitAnd(ty(t)),
840        S::BitOr(t) => R::BitOr(ty(t)),
841        S::BitXor(t) => R::BitXor(ty(t)),
842        S::Not => R::Not,
843        S::And => R::And,
844        S::Or => R::Or,
845        S::Xor => R::Xor,
846        S::Eq(t) => R::Eq(ty(t)),
847        S::Ne(t) => R::Ne(ty(t)),
848        S::Lt(t) => R::Lt(ty(t)),
849        S::Gt(t) => R::Gt(ty(t)),
850        S::Le(t) => R::Le(ty(t)),
851        S::Ge(t) => R::Ge(ty(t)),
852        S::Label(_) | S::Branch(_) | S::ConditionalBranch(_, _) | S::Call(_, _) => {
853            return Err(eyre::eyre!(
854                "symbolic control flow needs a label table and is not supported \
855                 in a lanes program"
856            ));
857        }
858    })

1772    /// A full text spelling is `<device>::<dialect>.<mnemonic>`. The narrower
1773    /// `cpu::add` form also appears in the docs — it is how a `DecodingError`
1774    /// names an instruction — but that is `op_name`, not a text mnemonic.
1775    fn is_text_spelling(token: &str) -> bool {
1776        token
1777            .split_once("::")
1778            .and_then(|(device, rest)| {
1779                let (dialect, mnemonic) = rest.split_once('.')?;
1780                Some(
1781                    matches!(device, "cpu" | "lanes")
1782                        && matches!(dialect, "cpu" | "lanes")
1783                        && !mnemonic.is_empty(),
1784                )
1785            })
1786            .unwrap_or(false)
1787    }

1789    /// The first text spelling in `line`, if any.
1790    fn mnemonic_in(line: &str) -> Option<&str> {
1791        line.split(|c: char| !(c.is_alphanumeric() || c == '_' || c == ':' || c == '.'))
1792            .map(|t| t.trim_end_matches('.'))
1793            .find(|t| is_text_spelling(t))
1794    }

1796    /// The first `0x`-prefixed hex literal in `line`, if any.
1797    fn hex_in(line: &str) -> Option<u16> {

1799            .find_map(|t| t.strip_prefix("0x"))

1804    ///
1805    /// `inst-spec.md` documented `cpu::cpu.const_int`, `cpu::cpu.const_float`
1806    /// and `cpu::cpu.return`, none of which parse: the constants are one typed
1807    /// `const` taking a comma, and `return` is spelled `ret`. Prose drifts

1810    fn every_documented_mnemonic_exists() {

1814                text.split(|c: char| !(c.is_alphanumeric() || c == '_' || c == ':' || c == '.'));
1815            for token in tokens {
1816                let token = token.trim_end_matches('.');
1817                if is_text_spelling(token) && !real_mnemonics().contains_key(token) {
1818                    bogus.push(format!("  {file}: `{token}`"));
1819                }
1820            }
1821        }
1822        bogus.sort();
1823        bogus.dedup();
1824        assert!(
1825            bogus.is_empty(),
1826            "the docs name {} mnemonic(s) the machine does not have:\n{}",
1827            bogus.len(),
1828            bogus.join("\n")
1829        );

1832    /// Every `Opcode` the docs tabulate must be the one the encoder produces.
1833    ///
1834    /// The spec prints a literal for each instruction, and nothing pinned
1835    /// them: they are assigned by declaration order, so they renumber whenever
1836    /// either instruction set gains a variant. The guide tells readers to
1837    /// compare on `op_name()` for exactly that reason — but a *format spec*
1838    /// that omits the byte values is not much of a format spec, so they stay
1839    /// and are checked here instead.
1840    #[test]
1841    fn every_documented_opcode_matches_the_encoding() {
1842        let real = real_mnemonics();
1843        let mut wrong = Vec::new();
1844        let mut checked = 0;
1845
1846        for (file, text) in DOCS {
1847            // `inst-spec.md` puts the mnemonic in a `#### ` heading and the
1848            // value in an `| Opcode | 0x.. |` row; `inst-quick-ref.md` puts
1849            // both in one row. Track the heading so either shape resolves.
1850            let mut heading: Option<&str> = None;
1851            for line in text.lines() {
1852                if let Some(rest) = line.strip_prefix("#### ") {
1853                    heading = mnemonic_in(rest);
1854                    continue;
1855                }
1856                if !line.starts_with('|') {
1857                    continue;
1858                }
1859                let Some(documented) = hex_in(line) else {
1860                    continue;
1861                };
1862                // Either the row names the instruction itself, or it is the
1863                // `Opcode` row belonging to the heading above it. Anything
1864                // else carrying a hex literal (the device-code table) is not
1865                // an opcode claim.
1866                let first_cell = line.split('|').nth(1).map(str::trim).unwrap_or("");
1867                let Some(name) = mnemonic_in(line).or(if first_cell == "Opcode" {
1868                    heading
1869                } else {
1870                    None
1871                }) else {
1872                    continue;
1873                };
1874
1875                match real.get(name) {
1876                    Some(actual) if *actual == documented => checked += 1,
1877                    Some(actual) => wrong.push(format!(
1878                        "  {file}: `{name}` documented as {documented:#06x}, encodes as {actual:#06x}"
1879                    )),
1880                    // The mnemonic test covers unknown names.
1881                    None => {}
1882                }
1883            }
1884        }
1885
1886        assert!(
1887            wrong.is_empty(),
1888            "{} documented opcode(s) disagree with the encoding:\n{}",
1889            wrong.len(),
1890            wrong.join("\n")
1891        );
1892        // A parser that silently matched nothing would make this vacuous:
1893        // 29 rows in the quick reference plus 23 in the spec.
1894        assert_eq!(
1895            checked, 52,
1896            "expected 52 documented opcodes, found {checked}"
1897        );
1898    }
