1//! The Bloqade Lanes machine: vihaco-cpu's `CPU` plus our atom-movement device.
2//!
3//! This is the idiomatic vihaco composition, and the same shape PPVM uses: each
4//! device owns its instruction set, and `#[composite]` generates the combined
5//! program type, the text parser (an alternation over the devices' parsers) and
6//! the section loaders.
7//!
8//! ```text
9//! cpu::cpu.const u64, 0      <- vihaco-cpu's device
10//! lanes::lanes.move 2        <- ours
11//! ```
12//!
13//! The `<field>::` prefix is the device field name; the second half is that
14//! device's own dialect head.
15//!
16//! ## Who does what
17//!
18//! Instruction operands live on the CPU stack, so a device never reads them
19//! directly. `resolve_lanes` pops them and packs them into a [`LanesMessage`];
20//! the device executes; the machine then applies any [`LanesEffect::Push`]
21//! back onto the stack. That round trip is the reason address constants can
22//! stay lanes instructions while still behaving like stack pushes.
23//!
24//! [`super::validate::simulate_stack`] is the static half of this: it models
25//! the same pops *and the same pushes*, so a program it accepts runs without
26//! underflowing. The ops the device does not interpret hold up their end by
27//! pushing [`Value::Undefined`] placeholders — the depth the simulator
28//! predicts, with a value nothing can mistake for a result.
29

51/// `#[composite]` derives only `Debug` and `Clone` on its runtime enum, but both
52/// device instruction sets *are* `PartialEq`, so comparing programs only needs
53/// the two arms spelled out. Round-trip tests and the Python `__eq__` both rely
54/// on it.
55impl PartialEq for MachineInstruction {
56    fn eq(&self, other: &Self) -> bool {
57        match (self, other) {
58            (Self::Cpu(a), Self::Cpu(b)) => a == b,
59            (Self::Lanes(a), Self::Lanes(b)) => a == b,
60            _ => false,
61        }
62    }
63}

81/// Why [`LanesMachine::run`] stopped.
82#[derive(Debug, Clone, Copy, PartialEq, Eq)]
83pub enum Stopped {
84    /// The program executed `halt`.
85    Halted,
86    /// `@main` returned.
87    Returned,
88    /// Execution ran off the end of the code without a terminator.
89    ///
90    /// [`super::validate::validate_structure`] rejects such a program, so this
91    /// only happens when running one that was never validated.
92    RanOff,
93    /// The step budget was exhausted. A program with a backward branch can
94    /// run forever, and a machine with no budget would hang rather than say
95    /// so.
96    OutOfSteps,
97}

99/// What running a program produced.
100#[derive(Debug, Clone)]
101pub struct Run {
102    /// Every effect the lanes device reported, in execution order. This is the
103    /// record of what the program asked the hardware to do, and the input an
104    /// observer for <https://github.com/QuEraComputing/bloqade-lanes/issues/1022>
105    /// would consume.
106    pub effects: Vec<LanesEffect>,
107    /// Why execution stopped.
108    pub stopped: Stopped,
109    /// How many instructions ran.
110    pub steps: u64,
111}

113impl LanesMachine {
114    /// A machine ready to run, with the entry frame already pushed.
115    ///
116    /// vihaco's locals are a window into the operand stack starting at the
117    /// current frame's `base`, so `load`/`store` and `ret` all fail with "no
118    /// current frame" until one exists. A `call` pushes its own; `@main` is
119    /// entered without one, so the machine establishes it.
120    pub fn new() -> Self {
121        let mut machine = Self::default();
122        machine.cpu.push_frame(Frame {
123            base: 0,
124            span: (0, 0, 0),
125            function: None,
126            ret_pc: 0,
127        });
128        machine
129    }

131    /// Point the machine at an architecture. `move` cannot resolve a lane into
132    /// (src, dst) endpoints without one, so it fails until this is set.
133    pub fn with_arch(mut self, arch: ArchSpec) -> Self {
134        self.lanes.arch = Some(arch);
135        self
136    }

138    /// Run a program: dispatch each instruction to the device that owns it,
139    /// until it halts, returns, or runs out of code or budget.
140    ///
141    /// This is the piece that makes the composite a machine rather than two
142    /// devices: [`step_lanes`](Self::step_lanes) handles one lanes
143    /// instruction, `CPU::execute_instruction` one CPU instruction, and
144    /// nothing until now walked a `MachineInstruction` stream across both.
145    ///
146    /// `max_steps` bounds execution. A program with a backward branch can run
147    /// forever, and the ISA carries vihaco-cpu's control flow whether or not
148    /// the lanes compiler emits it, so the budget is a parameter rather than
149    /// an assumption.
150    pub fn run(&mut self, program: &Program, max_steps: u64) -> eyre::Result<Run> {
151        use vihaco_cpu::StepOutcome;
152
153        let mut effects = Vec::new();
154        let mut steps = 0u64;
155        let mut pc = 0usize;
156
157        let stopped = loop {
158            let Some(inst) = program.code.get(pc) else {
159                break Stopped::RanOff;
160            };
161            if steps == max_steps {
162                break Stopped::OutOfSteps;
163            }
164            steps += 1;
165
166            // `op_call` reads this to work out where to return to, so it has
167            // to be set before the instruction runs, not after.
168            self.cpu.set_current_pc(pc as u32);
169
170            match inst {
171                MachineInstruction::Lanes(inst) => {
172                    collect(self.step_lanes(inst.clone())?, &mut effects);
173                }
174                MachineInstruction::Cpu(inst) => {
175                    match self.cpu.execute_instruction(inst.clone())? {
176                        StepOutcome::Halt => break Stopped::Halted,
177                        // `op_return` reports `Return` only when it pops the
178                        // last frame; an inner return sets the resume address
179                        // and reports `Continue`.
180                        StepOutcome::Return => break Stopped::Returned,
181                        StepOutcome::Continue | StepOutcome::Breakpoint => {}
182                    }
183                }
184            }
185
186            // A branch or call leaves its destination here; anything else
187            // falls through to the next instruction.
188            pc = match self.cpu.take_pending_pc() {
189                Some(target) => target as usize,
190                None => pc + 1,
191            };
192        };
193
194        Ok(Run {
195            effects,
196            stopped,
197            steps,
198        })
199    }

201    /// Pop the operands `inst` consumes and pack them into its message.
202    ///
203    /// Pops are in reverse push order throughout: the last value pushed is the
204    /// first popped, so a group read back from the stack is reversed to restore
205    /// program order.
206    fn resolve_lanes(&mut self, inst: &LanesInstruction) -> eyre::Result<LanesMessage> {
207        use LanesInstruction as I;
208        Ok(match inst {
209            // Constants carry their operand in the instruction word.
210            I::ConstLoc(_) | I::ConstLane(_) | I::ConstZone(_) => LanesMessage::None,
211
212            // The device cannot reach the stack, so the machine does the
213            // popping for it and the pushes come back as effects.
214            I::Pop => {
215                self.cpu.stack_pop()?;
216                LanesMessage::None
217            }
218            I::Swap => LanesMessage::Values(self.pop_values(2)?),
219
220            I::InitialFill(n) | I::Fill(n) => LanesMessage::Locations(self.pop_locations(*n)?),
221            I::Move(n) => LanesMessage::Lanes(self.pop_lanes(*n)?),
222
223            // Angles sit above the locations: `local_r` pops axis then
224            // rotation, then the location group beneath them.
225            I::LocalR(n) => LanesMessage::LocalRotation {
226                angles: self.pop_floats(2)?,
227                locations: self.pop_locations(*n)?,
228            },
229            I::LocalRz(n) => LanesMessage::LocalRotation {
230                angles: self.pop_floats(1)?,
231                locations: self.pop_locations(*n)?,
232            },
233            I::GlobalR => LanesMessage::GlobalRotation {
234                angles: self.pop_floats(2)?,
235            },
236            I::GlobalRz => LanesMessage::GlobalRotation {
237                angles: self.pop_floats(1)?,
238            },
239
240            I::Cz => LanesMessage::Zones(self.pop_zones(1)?),
241            I::Measure(n) => LanesMessage::Zones(self.pop_zones(*n)?),
242
243            // `new_array` consumes dim0×dim1 elements (dim1 = 0 means 1-D);
244            // `get_item` consumes the array reference and then its indices.
245            // Both are reported rather than executed — see
246            // [`LanesEffect::NotSimulated`] — so the operands travel with the
247            // message rather than being dropped on the floor.
248            //
249            // Both counts are computed in `u64`: `dim0 * dim1` overflows `u32`
250            // for operands a decoded program is free to carry, and `n + 1`
251            // overflows for `get_item(u32::MAX)`.
252            I::NewArray(_, dim0, dim1) => {
253                LanesMessage::Values(self.pop_values(array_element_count(*dim0, *dim1))?)
254            }
255            I::GetItem(n) => LanesMessage::Values(self.pop_values(*n as u64 + 1)?),
256
257            I::AwaitMeasure | I::SetDetector | I::SetObservable => {
258                LanesMessage::Values(self.pop_values(1)?)
259            }
260        })
261    }
262
263    /// Pop `n` values, restoring program order (the last pushed is popped
264    /// first).
265    ///
266    /// `n` comes straight out of an instruction word, so nothing is
267    /// pre-allocated against it: an implausible count runs out of stack within
268    /// a few pops and fails there. [`super::validate::validate_structure`]
269    /// rejects such a program up front; this makes the machine safe on its own
270    /// regardless.
271    fn pop_values(&mut self, n: u64) -> eyre::Result<Vec<Value>> {
272        let mut out = Vec::with_capacity(self.pop_capacity(n));
273        for _ in 0..n {
274            out.push(self.cpu.stack_pop()?);
275        }
276        out.reverse();
277        Ok(out)
278    }
279
280    /// Capacity to reserve for a pop of `n` values.
281    ///
282    /// `n` comes straight out of an instruction word, so it cannot be used as
283    /// a capacity: `initial_fill 4294967295` asks for 48 GiB of
284    /// `LocationAddr`. Linux's allocator refuses and Rust aborts the process
285    /// (SIGABRT, no unwinding, no test failure to catch); macOS commits
286    /// lazily and hands it back, which is why this only ever showed up in CI.
287    ///
288    /// A pop can never take more than the stack holds, so the stack depth is
289    /// both a safe bound and a sufficient one — every legitimate arity still
290    /// gets its single up-front allocation.
291    fn pop_capacity(&self, n: u64) -> usize {
292        n.min(self.cpu.stack().len() as u64) as usize
293    }
294
295    fn pop_u64(&mut self) -> eyre::Result<u64> {
296        match self.cpu.stack_pop()? {
297            Value::U64(v) => Ok(v),
298            v => Err(eyre::eyre!("expected a packed u64 address, got {v:?}")),
299        }
300    }
301
302    fn pop_floats(&mut self, n: u32) -> eyre::Result<Vec<f64>> {
303        let mut out = Vec::with_capacity(self.pop_capacity(n as u64));
304        for _ in 0..n {
305            match self.cpu.stack_pop()? {
306                Value::F64(v) => out.push(v),
307                v => return Err(eyre::eyre!("expected an angle (f64), got {v:?}")),
308            }
309        }
310        out.reverse();
311        Ok(out)
312    }
313
314    fn pop_locations(&mut self, n: u32) -> eyre::Result<Vec<LocationAddr>> {
315        let mut out = Vec::with_capacity(self.pop_capacity(n as u64));
316        for _ in 0..n {
317            out.push(LocationAddr::decode(self.pop_u64()?));
318        }
319        out.reverse();
320        Ok(out)
321    }
322
323    fn pop_lanes(&mut self, n: u32) -> eyre::Result<Vec<LaneAddr>> {
324        let mut out = Vec::with_capacity(self.pop_capacity(n as u64));
325        for _ in 0..n {
326            out.push(LaneAddr::decode_u64(self.pop_u64()?));
327        }
328        out.reverse();
329        Ok(out)
330    }
331
332    fn pop_zones(&mut self, n: u32) -> eyre::Result<Vec<ZoneAddr>> {
333        let mut out = Vec::with_capacity(self.pop_capacity(n as u64));
334        for _ in 0..n {
335            match self.cpu.stack_pop()? {
336                Value::U32(v) => out.push(ZoneAddr::decode(v)),
337                v => return Err(eyre::eyre!("expected a zone address (u32), got {v:?}")),
338            }
339        }
340        out.reverse();
341        Ok(out)
342    }

344    /// Apply a device effect. Only [`LanesEffect::Push`] touches the machine;
345    /// the rest are observations for the caller.
346    fn apply(&mut self, effects: &Effects<LanesEffect>) -> eyre::Result<()> {
347        let mut push = |effect: &LanesEffect| {
348            if let LanesEffect::Push(value) = effect {
349                self.cpu.stack_push(*value);
350            }
351        };
352        match effects {
353            Effects::None => {}
354            Effects::One(effect) => push(effect),
355            Effects::Many(effects) => effects.iter().for_each(push),
356        }
357        Ok(())
358    }
359
360    /// Run one lanes instruction end to end: pop its operands, execute, then
361    /// apply whatever comes back.
362    pub fn step_lanes(&mut self, inst: LanesInstruction) -> eyre::Result<Effects<LanesEffect>> {
363        let msg = self.resolve_lanes(&inst)?;
364        let effects = self.lanes.execute_generated(inst, msg)?;
365        self.apply(&effects)?;
366        Ok(effects)
367    }

371        &self.lanes.atoms
372    }

375/// Flatten one instruction's effects onto the run's record.
376fn collect(effects: Effects<LanesEffect>, into: &mut Vec<LanesEffect>) {
377    match effects {
378        Effects::None => {}
379        Effects::One(effect) => into.push(effect),
380        Effects::Many(effects) => into.extend(effects),
381    }

386/// vihaco-cpu's own `Display` emits bare mnemonics (`halt`, `const.f64 1.5`)
387/// that its *parser* does not accept, so rendering is written here against the
388/// surface grammar instead. The round-trip tests pin the two together.
389fn cpu_type_text(ty: Type) -> &'static str {
390    match ty {
391        Type::Undefined => "undef",
392        Type::String => "str",
393        Type::Bool => "bool",
394        Type::I64 => "i64",
395        Type::U32 => "u32",
396        Type::U64 => "u64",
397        Type::F64 => "f64",
398        Type::FunctionRef => "fn_ref",
399        Type::HeapRef => "heap_ref",
400    }
401}

403fn cpu_value_text(value: &Value) -> String {
404    match value {
405        // `{:?}` on f64 is round-trip exact; `{}` drops the `.0` on integral
406        // floats, which would then re-parse as an integer.
407        Value::F64(v) => format!("{v:?}"),
408        Value::Bool(v) => v.to_string(),
409        Value::I64(v) => v.to_string(),
410        Value::U32(v) => v.to_string(),
411        Value::U64(v) => v.to_string(),
412        Value::String(v) | Value::FunctionRef(v) | Value::HeapRef(v) => v.to_string(),
413        Value::Undefined => "undef".to_string(),
414    }
415}

509/// Canonical opcode name, without any prefix — the key the Python decoder
510/// dispatches on (`_visit_{op_name}`).
511///
512/// Two names deliberately differ from the text mnemonic, because the decoder
513/// depends on them:
514///
515/// - the constants stay `const_float` / `const_int` rather than vihaco-cpu's
516///   single typed `const`, since the decoder pushes a different value type for
517///   each and the mnemonic alone would not say which;
518/// - `return` keeps its spelling rather than vihaco-cpu's `ret`.
519///
520/// This is a `match` rather than the first token of [`to_sst_text`] because it
521/// runs once per instruction in the Python decoder's loop, and rendering to
522/// recover a name means formatting the operand that is then thrown away —
523/// `const_loc` cost a 20-character hex format of its `u64` plus two
524/// allocations, for one of about sixty fixed strings.
525/// `every_op_name_matches_its_mnemonic` keeps it in step with the renderer.
526pub fn op_name(inst: &MachineInstruction) -> &'static str {
527    match inst {
528        MachineInstruction::Cpu(i) => cpu_op_name(i),
529        MachineInstruction::Lanes(i) => lanes_op_name(i),
530    }
531}

616/// Lower a parsed instruction to its executable form.
617///
618/// `#[composite]` generates the surface and runtime enums independently and no
619/// conversion between them, so — as with the device — this is written out.
620pub fn lower(inst: MachineSurfaceInstruction) -> eyre::Result<MachineInstruction> {
621    Ok(match inst {
622        MachineSurfaceInstruction::Cpu(i) => MachineInstruction::Cpu(lower_cpu(i)?),
623        MachineSurfaceInstruction::Lanes(i) => MachineInstruction::Lanes(super::device::lower(i)),
624    })
625}

627/// Lower a parsed vihaco-cpu instruction to its runtime form.
628///
629/// The surface form is lexical — `SurfaceValue` is a raw token and branch
630/// targets are identifiers — so this is where a constant becomes a typed
631/// [`Value`].
632///
633/// What this accepts is the counterpart of what [`to_sst_text`] emits: a
634/// decoded binary may hold any of vihaco's nine `const` types, so all nine
635/// parse back. The two rejections both need something a per-instruction
636/// lowering does not have — a symbol table, for symbolic control flow, and a
637/// string table, for a quoted literal.
638fn lower_cpu(inst: CpuSurfaceInstruction) -> eyre::Result<vihaco_cpu::RuntimeInstruction> {
639    use CpuSurfaceInstruction as S;
640    use vihaco_cpu::RuntimeInstruction as R;
641
642    let ty = |t: vihaco_cpu::SurfaceType| -> Type {
643        match t {
644            vihaco_cpu::SurfaceType::Undefined => Type::Undefined,
645            vihaco_cpu::SurfaceType::String => Type::String,
646            vihaco_cpu::SurfaceType::Bool => Type::Bool,
647            vihaco_cpu::SurfaceType::I64 => Type::I64,
648            vihaco_cpu::SurfaceType::U32 => Type::U32,
649            vihaco_cpu::SurfaceType::U64 => Type::U64,
650            vihaco_cpu::SurfaceType::F64 => Type::F64,
651            vihaco_cpu::SurfaceType::FunctionRef => Type::FunctionRef,
652            vihaco_cpu::SurfaceType::HeapRef => Type::HeapRef,
653        }
654    };
655
656    Ok(match inst {
657        S::Const(t, v) => {
658            let ty = ty(t);
659            // A quoted literal would have to be interned to become a
660            // `Value::String`, and interning needs the module's string table —
661            // which a per-instruction lowering does not have. `const str` takes
662            // the interner index instead, which is what the value holds and
663            // what `cpu_value_text` renders.
664            let text = match &v {
665                vihaco_cpu::SurfaceValue::Bare(token) => token.0.clone(),
666                vihaco_cpu::SurfaceValue::Quoted(_) => {
667                    return Err(eyre::eyre!(
668                        "a quoted string literal needs the module's string table; \
669                         write the interner index instead"
670                    ));
671                }
672            };
673            // Every type the renderer can emit must parse back, or
674            // `disassemble` produces text `assemble` rejects. The three
675            // reference types and `String` are interner/heap indices, so they
676            // read as plain integers.
677            let value = match ty {
678                Type::F64 => Value::F64(text.parse()?),
679                Type::I64 => Value::I64(text.parse()?),
680                Type::U64 => Value::U64(text.parse()?),
681                Type::U32 => Value::U32(text.parse()?),
682                Type::Bool => Value::Bool(text.parse()?),
683                Type::String => Value::String(text.parse()?),
684                Type::FunctionRef => Value::FunctionRef(text.parse()?),
685                Type::HeapRef => Value::HeapRef(text.parse()?),
686                Type::Undefined if text == "undef" => Value::Undefined,
687                Type::Undefined => {
688                    return Err(eyre::eyre!("const undef takes no value but got '{text}'"));
689                }
690            };
691            R::Const(ty, value)
692        }
693        S::Span(a, b, c) => R::Span(a, b, c),
694        S::FunctionStart => R::FunctionStart,
695        S::FunctionEnd => R::FunctionEnd,
696        S::Breakpoint => R::Breakpoint,
697        S::Return(n) => R::Return(n),
698        S::IndirectCall => R::IndirectCall,
699        S::Halt => R::Halt,
700        S::Print => R::Print,
701        S::Load(t, addr) => R::Load(ty(t), addr),
702        S::Store(t, addr) => R::Store(ty(t), addr),
703        S::Dup => R::Dup,
704        S::HeapAlloc(n) => R::HeapAlloc(n),
705        S::GetItem => R::GetItem,
706        S::HeapDealloc => R::HeapDealloc,
707        S::Add(t) => R::Add(ty(t)),
708        S::Sub(t) => R::Sub(ty(t)),
709        S::Mul(t) => R::Mul(ty(t)),
710        S::Div(t) => R::Div(ty(t)),
711        S::Rem(t) => R::Rem(ty(t)),
712        S::Neg(t) => R::Neg(ty(t)),
713        S::Shl(t) => R::Shl(ty(t)),
714        S::Shr(t) => R::Shr(ty(t)),
715        S::Rol(t) => R::Rol(ty(t)),
716        S::Ror(t) => R::Ror(ty(t)),
717        S::BitAnd(t) => R::BitAnd(ty(t)),
718        S::BitOr(t) => R::BitOr(ty(t)),
719        S::BitXor(t) => R::BitXor(ty(t)),
720        S::Not => R::Not,
721        S::And => R::And,
722        S::Or => R::Or,
723        S::Xor => R::Xor,
724        S::Eq(t) => R::Eq(ty(t)),
725        S::Ne(t) => R::Ne(ty(t)),
726        S::Lt(t) => R::Lt(ty(t)),
727        S::Gt(t) => R::Gt(ty(t)),
728        S::Le(t) => R::Le(ty(t)),
729        S::Ge(t) => R::Ge(ty(t)),
730        S::Label(_) | S::Branch(_) | S::ConditionalBranch(_, _) | S::Call(_, _) => {
731            return Err(eyre::eyre!(
732                "symbolic control flow needs a label table and is not supported \
733                 in a lanes program"
734            ));
735        }
736    })
737}

1219    /// A full text spelling is `<device>::<dialect>.<mnemonic>`. The narrower
1220    /// `cpu::add` form also appears in the docs — it is how a `DecodingError`
1221    /// names an instruction — but that is `op_name`, not a text mnemonic.
1222    fn is_text_spelling(token: &str) -> bool {
1223        token
1224            .split_once("::")
1225            .and_then(|(device, rest)| {
1226                let (dialect, mnemonic) = rest.split_once('.')?;
1227                Some(
1228                    matches!(device, "cpu" | "lanes")
1229                        && matches!(dialect, "cpu" | "lanes")
1230                        && !mnemonic.is_empty(),
1231                )
1232            })
1233            .unwrap_or(false)
1234    }

1236    /// The first text spelling in `line`, if any.
1237    fn mnemonic_in(line: &str) -> Option<&str> {
1238        line.split(|c: char| !(c.is_alphanumeric() || c == '_' || c == ':' || c == '.'))
1239            .map(|t| t.trim_end_matches('.'))
1240            .find(|t| is_text_spelling(t))
1241    }
1242
1243    /// The first `0x`-prefixed hex literal in `line`, if any.
1244    fn hex_in(line: &str) -> Option<u16> {
1245        line.split(|c: char| !(c.is_alphanumeric() || c == 'x'))
1246            .find_map(|t| t.strip_prefix("0x"))
1247            .and_then(|t| u16::from_str_radix(t, 16).ok())
1248    }

1250    /// Every `<device>::<dialect>.<mnemonic>` the docs print must be real.
1251    ///
1252    /// `inst-spec.md` documented `cpu::cpu.const_int`, `cpu::cpu.const_float`
1253    /// and `cpu::cpu.return`, none of which parse: the constants are one typed
1254    /// `const` taking a comma, and `return` is spelled `ret`. Prose drifts
1255    /// silently, so the docs are read here rather than trusted.
1256    #[test]
1257    fn every_documented_mnemonic_exists() {
1258        let mut bogus = Vec::new();
1259        for (file, text) in DOCS {
1260            let tokens =
1261                text.split(|c: char| !(c.is_alphanumeric() || c == '_' || c == ':' || c == '.'));
1262            for token in tokens {
1263                let token = token.trim_end_matches('.');
1264                if is_text_spelling(token) && !real_mnemonics().contains_key(token) {
1265                    bogus.push(format!("  {file}: `{token}`"));
1266                }
1267            }
1268        }
1269        bogus.sort();
1270        bogus.dedup();
1271        assert!(
1272            bogus.is_empty(),
1273            "the docs name {} mnemonic(s) the machine does not have:\n{}",
1274            bogus.len(),
1275            bogus.join("\n")
1276        );
1277    }

1279    /// Every `Opcode` the docs tabulate must be the one the encoder produces.
1280    ///
1281    /// The spec prints a literal for each instruction, and nothing pinned
1282    /// them: they are assigned by declaration order, so they renumber whenever
1283    /// either instruction set gains a variant. The guide tells readers to
1284    /// compare on `op_name()` for exactly that reason — but a *format spec*
1285    /// that omits the byte values is not much of a format spec, so they stay
1286    /// and are checked here instead.
1287    #[test]
1288    fn every_documented_opcode_matches_the_encoding() {
1289        let real = real_mnemonics();
1290        let mut wrong = Vec::new();
1291        let mut checked = 0;
1292
1293        for (file, text) in DOCS {
1294            // `inst-spec.md` puts the mnemonic in a `#### ` heading and the
1295            // value in an `| Opcode | 0x.. |` row; `inst-quick-ref.md` puts
1296            // both in one row. Track the heading so either shape resolves.
1297            let mut heading: Option<&str> = None;
1298            for line in text.lines() {
1299                if let Some(rest) = line.strip_prefix("#### ") {
1300                    heading = mnemonic_in(rest);
1301                    continue;
1302                }
1303                if !line.starts_with('|') {
1304                    continue;
1305                }
1306                let Some(documented) = hex_in(line) else {
1307                    continue;
1308                };
1309                // Either the row names the instruction itself, or it is the
1310                // `Opcode` row belonging to the heading above it. Anything
1311                // else carrying a hex literal (the device-code table) is not
1312                // an opcode claim.
1313                let first_cell = line.split('|').nth(1).map(str::trim).unwrap_or("");
1314                let Some(name) = mnemonic_in(line).or(if first_cell == "Opcode" {
1315                    heading
1316                } else {
1317                    None
1318                }) else {
1319                    continue;
1320                };
1321
1322                match real.get(name) {
1323                    Some(actual) if *actual == documented => checked += 1,
1324                    Some(actual) => wrong.push(format!(
1325                        "  {file}: `{name}` documented as {documented:#06x}, encodes as {actual:#06x}"
1326                    )),
1327                    // The mnemonic test covers unknown names.
1328                    None => {}
1329                }
1330            }
1331        }
1332
1333        assert!(
1334            wrong.is_empty(),
1335            "{} documented opcode(s) disagree with the encoding:\n{}",
1336            wrong.len(),
1337            wrong.join("\n")
1338        );
1339        // A parser that silently matched nothing would make this vacuous:
1340        // 24 rows in the quick reference plus 23 in the spec.
1341        assert_eq!(
1342            checked, 47,
1343            "expected 47 documented opcodes, found {checked}"
1344        );
1345    }
