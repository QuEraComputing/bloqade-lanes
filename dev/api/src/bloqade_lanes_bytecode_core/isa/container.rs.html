1//! Writers for vihaco's section-based container formats.
2//!
3//! vihaco 0.4 ships **readers only** — [`vihaco::BytecodeFile::from_bytes`] and
4//! [`vihaco::SstFile::from_text`] parse its container, but nothing emits one. So
5//! we own the two emitters here, matching vihaco's layout byte for byte.
6//!
7//! The round-trip tests are what keep these honest: everything written here is
8//! read back through vihaco's own parser, so a layout mistake fails the build
9//! rather than producing a file only we can read.
10//!
11//! ## Shape of a lanes program
12//!
13//! Both emitters write one root section named `root`, whose header carries the
14//! [`LanesInfo`] version and whose body is the code stream. A program may
15//! declare any number of functions; they delimit themselves in that stream with
16//! `func_start`/`func_end`, so the root section needs no per-function framing.
17//!
18//! The two containers differ in how the symbol tables ride along. The binary
19//! form appends them as child sections ([`TABLE_SECTIONS`]), whose names resolve
20//! through the global context — which is why [`LanesContext`] is not empty and
21//! [`vihaco::NoContext`] will not do. The text form needs none of it: functions
22//! and labels are written syntactically (`fn @name`, `cpu::cpu.label @x`), so
23//! `.sst` has no child sections and an empty `.global:` block.
24//!
25//! ## Binary layout (vihaco `VHBC`)
26//!
27//! ```text
28//! magic                : 4 bytes = b"VHBC"
29//! version              : u16 LE  = 1
30//! flags                : u16 LE  = 0
31//! context_len          : u64 LE             (child-section name table)
32//! ── root section ──
33//! section_len          : u64 LE             (total, including this frame)
34//! composite_header_len : u64 LE
35//! composite header     : the LanesInfo header
36//! bytecode_len         : u64 LE
37//! bytecode             : N × INSTRUCTION_WIDTH bytes
38//! child_count          : u32 LE             (functions, labels, strings)
39//! ```
40//!
41//! ## Text layout (vihaco `sst v1`)
42//!
43//! ```text
44//! sst v1
45//!
46//! .section(root):
47//! .header(root):
48//! version 1.0
49//! .header(root).
50//! .text(root):
51//! fn @main() {
52//!   …
53//! }
54//! .text(root).
55//! .section(root).
56//! ```
57//!
58//! The global section is omitted entirely, which vihaco reads as an empty
59//! context.

76/// Names of the child sections carrying a program's symbol tables, in the order
77/// they are emitted.
78///
79/// The single source for all three uses — the context name table, the writer in
80/// [`to_binary`], and the reader in [`read_tables`] — so adding a table cannot
81/// leave one of them spelling a name the others do not know.
82/// `emitted_sections_are_exactly_the_declared_tables` pins that.
83pub const TABLE_SECTIONS: [&str; 3] = [FUNCTIONS_SECTION, LABELS_SECTION, STRINGS_SECTION];

85/// The global context of a lanes file: the child-section name table.
86///
87/// vihaco's binary child entries store a section's name as an *index*, resolved
88/// through the global context — so the moment a program has child sections, the
89/// context stops being empty. Ours carries exactly [`TABLE_SECTIONS`].
90///
91/// The text container needs none of this: functions and labels are written
92/// syntactically (`fn @name`, `cpu::cpu.label @x`), so `.sst` still has no child
93/// sections and an empty `.global:` block.
94///
95/// vihaco's own [`vihaco::NoContext`] cannot be used either way: it implements
96/// only the SST half of the pair, not [`vihaco::BytecodeGlobalContext`].
97#[derive(Debug, Clone, Default, PartialEq, Eq)]
98pub struct LanesContext {
99    names: Vec<String>,
100}
101
102impl LanesContext {
103    /// The context a binary program carries.
104    pub fn with_tables() -> Self {
105        Self {
106            names: TABLE_SECTIONS.iter().map(|s| (*s).to_owned()).collect(),
107        }
108    }
109
110    /// Serialized form: one name per line, matching the
111    /// [`BytecodeGlobalContext::from_bytes`](vihaco::BytecodeGlobalContext::from_bytes)
112    /// impl below.
113    pub fn to_bytes(&self) -> Vec<u8> {
114        self.names.join("\n").into_bytes()
115    }
116
117    /// The index a child-section name is stored under.
118    pub fn index_of(&self, name: &str) -> Option<u32> {
119        self.names.iter().position(|n| n == name).map(|i| i as u32)
120    }
121}

156/// Byte lengths of the fixed fields in vihaco's binary container. Named here so
157/// the emitter's arithmetic reads against the layout in the module docs.
158pub(super) mod len {
159    /// `magic` + `version` + `flags` + `context_len`.
160    ///
161    /// The reader's truncation guard in [`super::super::program::from_binary`]
162    /// uses this too: a buffer shorter than the file header is the one
163    /// truncation we measure ourselves, and it has to agree with what the
164    /// writer lays down.
165    pub const FILE_HEADER: usize = 4 + 2 + 2 + 8;
166    /// `section_len` + `composite_header_len`.
167    pub const SECTION_FRAME: usize = 8 + 8;
168    /// `bytecode_len`.
169    pub const BYTECODE_HEADER: usize = 8;
170    /// `child_count`.
171    pub const CHILD_TABLE_HEADER: usize = 4;
172    /// `local_name_string` + `section_offset`, per child.
173    pub const CHILD_TABLE_ENTRY: usize = 4 + 8;
174}

176/// A child section holding one symbol table.
177///
178/// Child sections are ordinary sections nested inside the root, so each needs
179/// the same framing. The table's bytes go in its bytecode region — vihaco only
180/// interprets that as instructions if you ask it to.
181struct ChildSection {
182    name: &'static str,
183    payload: Vec<u8>,
184}
185
186impl ChildSection {
187    /// Total encoded length: frame + empty header + payload + empty child table.
188    fn encoded_len(&self) -> usize {
189        len::SECTION_FRAME + len::BYTECODE_HEADER + self.payload.len() + len::CHILD_TABLE_HEADER
190    }
191
192    fn write_to(&self, out: &mut Vec<u8>) {
193        out.extend_from_slice(&(self.encoded_len() as u64).to_le_bytes());
194        out.extend_from_slice(&0u64.to_le_bytes()); // no composite header
195        out.extend_from_slice(&(self.payload.len() as u64).to_le_bytes());
196        out.extend_from_slice(&self.payload);
197        out.extend_from_slice(&0u32.to_le_bytes()); // no grandchildren
198    }
199}

201// ── Symbol tables ─────────────────────────────────────────────────────────────
202//
203// Each table is a `u32` count followed by fixed-size records, little-endian
204// throughout, matching the rest of the container. Strings are length-prefixed
205// bytes. These live in child sections because the root section has exactly one
206// payload slot and the code occupies it.
207
208fn encode_functions(program: &Program) -> Vec<u8> {
209    let mut out = Vec::new();
210    out.extend_from_slice(&(program.functions.len() as u32).to_le_bytes());
211    for f in &program.functions {
212        out.extend_from_slice(&f.name.to_le_bytes());
213        out.extend_from_slice(&f.local_count.to_le_bytes());
214        out.extend_from_slice(&f.start_address.to_le_bytes());
215        out.extend_from_slice(&f.end_address.to_le_bytes());
216        out.extend_from_slice(&f.file.to_le_bytes());
217
218        // The signature follows, length-prefixed. A function's arity is
219        // declared here and nowhere else: `call <arity>` carries the caller's
220        // claim, and this is what that claim is checked against.
221        out.extend_from_slice(&(f.signature.params.len() as u32).to_le_bytes());
222        for p in &f.signature.params {
223            out.extend_from_slice(&p.name.to_le_bytes());
224            out.extend_from_slice(&type_code(p.ty).to_le_bytes());
225        }
226        out.extend_from_slice(&(f.signature.ret.len() as u32).to_le_bytes());
227        for ty in &f.signature.ret {
228            out.extend_from_slice(&type_code(*ty).to_le_bytes());
229        }
230    }
231    out
232}

234/// Capacity to reserve for a table of `count` records of `record` bytes each.
235///
236/// `count` is the first word of the payload, so it is whatever the file says —
237/// and a capacity taken straight from it is a denial of service: `u32::MAX`
238/// `FunctionInfo`s is 288 GiB, which Linux's allocator refuses and Rust turns
239/// into an abort rather than a decode error. (macOS commits lazily and hands
240/// the reservation back, so this only shows up on one platform.)
241///
242/// After the 4-byte count the payload can hold at most `(len - 4) / record`
243/// entries, which is both a safe bound and an exact one: a well-formed table
244/// still gets its single up-front allocation, and a malformed count is caught
245/// by the per-field reads that follow.
246fn table_capacity(bytes: &[u8], count: usize, record: usize) -> usize {
247    count.min(bytes.len().saturating_sub(4) / record.max(1))
248}
249
250/// Smallest a function record can be: the five fixed words plus the two
251/// counts, with an empty signature. Only a *lower* bound now that records vary
252/// in length, which is all [`table_capacity`] needs it to be — it bounds the
253/// reservation, and the decode below still fails honestly on a short payload.
254const FUNCTION_RECORD_MIN: usize = 4 * 7;
255
256fn decode_functions(bytes: &[u8]) -> eyre::Result<Vec<vihaco::module::FunctionInfo<vihaco::Type>>> {
257    let count = read_u32(bytes, 0)? as usize;
258    let mut out = Vec::with_capacity(table_capacity(bytes, count, FUNCTION_RECORD_MIN));
259
260    // A signature is variable-length, so records are walked with a cursor
261    // rather than indexed — `4 + i * RECORD` stopped being an address the
262    // moment a function could declare parameters.
263    let mut at = 4;
264    let next = |at: &mut usize| -> eyre::Result<u32> {
265        let v = read_u32(bytes, *at)?;
266        *at += 4;
267        Ok(v)
268    };
269
270    for _ in 0..count {
271        let name = next(&mut at)?;
272        let local_count = next(&mut at)?;
273        let start_address = next(&mut at)?;
274        let end_address = next(&mut at)?;
275        let file = next(&mut at)?;
276
277        let param_count = next(&mut at)? as usize;
278        let mut params = Vec::with_capacity(table_capacity(bytes, param_count, 4 * 2));
279        for _ in 0..param_count {
280            let name = next(&mut at)?;
281            params.push(vihaco::module::Parameter {
282                name,
283                ty: decode_type_code(next(&mut at)?)?,
284            });
285        }
286
287        let ret_count = next(&mut at)? as usize;
288        let mut ret = Vec::with_capacity(table_capacity(bytes, ret_count, 4));
289        for _ in 0..ret_count {
290            ret.push(decode_type_code(next(&mut at)?)?);
291        }
292
293        out.push(vihaco::module::FunctionInfo {
294            name,
295            signature: vihaco::module::Signature { params, ret },
296            local_count,
297            start_address,
298            end_address,
299            file,
300        });
301    }
302    Ok(out)
303}

305/// Stable wire codes for [`vihaco::Type`].
306///
307/// Spelled out rather than taken from the enum's declaration order, which is
308/// what the *instruction* opcodes do — those are allowed to shift when the
309/// instruction set gains a variant, and a type written into a file is not.
310/// The match is exhaustive, so a new vihaco type is a compile error here
311/// rather than a silently mis-encoded file.
312fn type_code(ty: vihaco::Type) -> u32 {
313    use vihaco::Type as T;
314    match ty {
315        T::Undefined => 0,
316        T::String => 1,
317        T::Bool => 2,
318        T::I64 => 3,
319        T::U32 => 4,
320        T::U64 => 5,
321        T::F64 => 6,
322        T::FunctionRef => 7,
323        T::HeapRef => 8,
324    }
325}
326
327fn decode_type_code(code: u32) -> eyre::Result<vihaco::Type> {
328    use vihaco::Type as T;
329    Ok(match code {
330        0 => T::Undefined,
331        1 => T::String,
332        2 => T::Bool,
333        3 => T::I64,
334        4 => T::U32,
335        5 => T::U64,
336        6 => T::F64,
337        7 => T::FunctionRef,
338        8 => T::HeapRef,
339        other => eyre::bail!("unknown type code {other} in a function signature"),
340    })
341}

343fn encode_labels(program: &Program) -> Vec<u8> {
344    let mut out = Vec::new();
345    out.extend_from_slice(&(program.labels.len() as u32).to_le_bytes());
346    for l in &program.labels {
347        out.extend_from_slice(&l.address.to_le_bytes());
348        out.extend_from_slice(&l.name.to_le_bytes());
349    }
350    out
351}
352
353fn decode_labels(bytes: &[u8]) -> eyre::Result<Vec<vihaco::module::LabelInfo>> {
354    const RECORD: usize = 4 * 2;
355    let count = read_u32(bytes, 0)? as usize;
356    let mut out = Vec::with_capacity(table_capacity(bytes, count, RECORD));
357    for i in 0..count {
358        let at = 4 + i * RECORD;
359        out.push(vihaco::module::LabelInfo {
360            address: read_u32(bytes, at)?,
361            name: read_u32(bytes, at + 4)?,
362        });
363    }
364    Ok(out)
365}

367fn encode_strings(program: &Program) -> Vec<u8> {
368    let mut out = Vec::new();
369    out.extend_from_slice(&(program.strings.len() as u32).to_le_bytes());
370    for s in &program.strings {
371        out.extend_from_slice(&(s.len() as u32).to_le_bytes());
372        out.extend_from_slice(s.as_bytes());
373    }
374    out
375}

377fn decode_strings(bytes: &[u8]) -> eyre::Result<Vec<String>> {
378    // Entries are variable-length, but each carries a 4-byte length prefix,
379    // so four bytes is the smallest an entry can be.
380    let count = read_u32(bytes, 0)? as usize;
381    let mut out = Vec::with_capacity(table_capacity(bytes, count, 4));
382    let mut at = 4;
383    for _ in 0..count {
384        let len = read_u32(bytes, at)? as usize;
385        at += 4;
386        let end = at
387            .checked_add(len)
388            .filter(|end| *end <= bytes.len())
389            .ok_or_else(|| eyre::eyre!("string table entry runs past the section"))?;
390        out.push(String::from_utf8(bytes[at..end].to_vec())?);
391        at = end;
392    }
393    Ok(out)
394}
395
396fn read_u32(bytes: &[u8], at: usize) -> eyre::Result<u32> {
397    bytes
398        .get(at..at + 4)
399        .map(|b| u32::from_le_bytes([b[0], b[1], b[2], b[3]]))
400        .ok_or_else(|| eyre::eyre!("table section is truncated at byte {at}"))
401}

403/// Read the symbol tables back out of a parsed file's child sections.
404///
405/// An absent section leaves its table empty. That is not a compatibility
406/// path: a container predating the tables also predates the `func_start` /
407/// `func_end` markers, so it carries no functions to name and
408/// [`super::program::from_binary`] rejects it outright.
409pub fn read_tables(
410    root: &vihaco::BytecodeSectionView<'_, LanesContext>,
411    program: &mut Program,
412) -> eyre::Result<()> {
413    for child in root.children() {
414        match child.local_name() {
415            Some(FUNCTIONS_SECTION) => program.functions = decode_functions(child.bytecode())?,
416            Some(LABELS_SECTION) => program.labels = decode_labels(child.bytecode())?,
417            Some(STRINGS_SECTION) => program.strings = decode_strings(child.bytecode())?,
418            Some(other) => return Err(eyre::eyre!("unknown child section `{other}`")),
419            None => return Err(eyre::eyre!("child section has no name")),
420        }
421    }
422    Ok(())
423}

425/// Serialize a program into vihaco's `VHBC` container.
426///
427/// Instructions go through [`bytecode::encode`] because neither half of the
428/// composite carries a codec of its own. The symbol tables follow as child
429/// sections, whose offsets are relative to the start of the root section.
430pub fn to_binary(program: &Program) -> eyre::Result<Vec<u8>> {
431    // Two fields of `LocalModule` still have no place in the container, and
432    // neither is ever populated. Refusing to write a program that uses one
433    // turns a future silent drop into a loud failure.
434    //
435    // `FunctionInfo.signature` used to be the third, with a note that it was
436    // "the one to watch, since the moment the calling convention records
437    // parameters, `encode_functions` would discard them". That moment came:
438    // signatures are now declared, encoded below, and read back — so the
439    // refusal is gone rather than the field being dropped.
440    if !program.constants.is_empty() {
441        eyre::bail!("the container cannot carry a constant pool yet");
442    }
443    if !program.source_symbols.is_empty() {
444        eyre::bail!("the container cannot carry source symbols yet");
445    }
446    // vihaco's grammar spells a return type as `-> Ty` — at most one. A
447    // signature carrying more has no text form, and `to_text` would render the
448    // first and drop the rest, turning a round trip into a silent truncation.
449    // Nothing can build one through a supported path (the parser yields
450    // `Option<Ty>`), so this refuses rather than truncating, the same rule the
451    // loader applies to a container it cannot name.
452    if let Some(f) = program.functions.iter().find(|f| f.signature.ret.len() > 1) {
453        eyre::bail!(
454            "function signature declares {} return values; the text form has \
455             syntax for at most one",
456            f.signature.ret.len()
457        );
458    }
459
460    let context = LanesContext::with_tables();
461    let context_bytes = context.to_bytes();
462
463    let mut header = Vec::new();
464    program
465        .extra
466        .write_bytes(&mut header)
467        .expect("writing a header to a Vec cannot fail");
468
469    let mut code = Vec::new();
470    for inst in &program.code {
471        bytecode::encode(inst)?
472            .write_bytes(&mut code)
473            .expect("writing instruction bytes to a Vec cannot fail");
474    }
475
476    let children = [
477        ChildSection {
478            name: FUNCTIONS_SECTION,
479            payload: encode_functions(program),
480        },
481        ChildSection {
482            name: LABELS_SECTION,
483            payload: encode_labels(program),
484        },
485        ChildSection {
486            name: STRINGS_SECTION,
487            payload: encode_strings(program),
488        },
489    ];
490
491    // The root's own extent, before the children are appended.
492    let root_prefix = len::SECTION_FRAME
493        + header.len()
494        + len::BYTECODE_HEADER
495        + code.len()
496        + len::CHILD_TABLE_HEADER
497        + children.len() * len::CHILD_TABLE_ENTRY;
498    let section_len = root_prefix
499        + children
500            .iter()
501            .map(ChildSection::encoded_len)
502            .sum::<usize>();
503
504    let mut out = Vec::with_capacity(len::FILE_HEADER + context_bytes.len() + section_len);
505    // ── file header ──
506    out.extend_from_slice(vihaco::MAGIC);
507    out.extend_from_slice(&vihaco::VERSION.to_le_bytes());
508    out.extend_from_slice(&vihaco::FLAGS.to_le_bytes());
509    out.extend_from_slice(&(context_bytes.len() as u64).to_le_bytes());
510    out.extend_from_slice(&context_bytes);
511    // ── root section ──
512    out.extend_from_slice(&(section_len as u64).to_le_bytes());
513    out.extend_from_slice(&(header.len() as u64).to_le_bytes());
514    out.extend_from_slice(&header);
515    out.extend_from_slice(&(code.len() as u64).to_le_bytes());
516    out.extend_from_slice(&code);
517    // ── child table: offsets are relative to the root section's start ──
518    out.extend_from_slice(&(children.len() as u32).to_le_bytes());
519    let mut offset = root_prefix;
520    for child in &children {
521        let index = context.index_of(child.name).ok_or_else(|| {
522            eyre::eyre!("child section `{}` is missing from the context", child.name)
523        })?;
524        out.extend_from_slice(&index.to_le_bytes());
525        out.extend_from_slice(&(offset as u64).to_le_bytes());
526        offset += child.encoded_len();
527    }
528    for child in &children {
529        child.write_to(&mut out);
530    }
531    Ok(out)
532}

534/// Emit a program as vihaco's `sst v1` text container.
535///
536/// `body` is the already-rendered `fn @main() { … }` block.
537pub fn to_sst(info: &LanesInfo, body: &str) -> String {
538    let mut out = String::from("sst v1\n\n");
539    out.push_str(&format!(".section({ROOT_SECTION}):\n"));
540    out.push_str(&format!(".header({ROOT_SECTION}):\n"));
541    out.push_str(&info.to_string());
542    out.push('\n');
543    out.push_str(&format!(".header({ROOT_SECTION}).\n"));
544    out.push_str(&format!(".text({ROOT_SECTION}):\n"));
545    out.push_str(body);
546    out.push_str(&format!(".text({ROOT_SECTION}).\n"));
547    out.push_str(&format!(".section({ROOT_SECTION}).\n"));
548    out
549}

596        );

598        // `signature` used to be the third refusal, and was the one this test
599        // existed to guard: the comment on the bail called it "the one to
600        // watch, since the moment the calling convention records parameters,
601        // `encode_functions` would discard them". It now carries them, so the
602        // assertion is that the field survives rather than that it is refused.
603        let mut p = base;
604        p.functions[0].signature = Signature {
605            params: vec![Parameter {
606                name: 0,
607                ty: Type::I64,
608            }],
609            ret: vec![Type::Bool],
610        };
611        let bytes = to_binary(&p).expect("a declared signature is written, not refused");
612        let back = crate::isa::program::from_binary(&bytes).expect("and read back");
613        assert_eq!(
614            back.functions[0].signature, p.functions[0].signature,
615            "the signature should survive the container"
616        );
617    }
618
619    /// A table's declared count cannot be trusted as a capacity.
620    ///
621    /// It is the first word of the payload, so a malformed file can claim
622    /// `u32::MAX` records — 288 GiB of `FunctionInfo` — which Linux's
623    /// allocator refuses and Rust turns into a process abort rather than the
624    /// decode error the caller is waiting for. macOS commits lazily and hands
625    /// the reservation back, so the bound is asserted here rather than left to
626    /// an allocator to object on one platform and not the other.
627    #[test]
628    fn a_table_count_cannot_reserve_more_than_the_payload_holds() {
629        // A four-byte payload holding nothing but the count itself.
630        let mut bytes = u32::MAX.to_le_bytes().to_vec();
631        assert_eq!(table_capacity(&bytes, u32::MAX as usize, 20), 0);
632
633        // One complete 20-byte record: room for exactly one.
634        bytes.extend_from_slice(&[0u8; 20]);
635        assert_eq!(table_capacity(&bytes, u32::MAX as usize, 20), 1);
636
637        // A well-formed count still gets its exact capacity.
638        assert_eq!(table_capacity(&bytes, 1, 20), 1);
639    }
640
641    /// And the decoders reject the malformed count rather than allocating for
642    /// it — each reports the truncation the count implies.
643    #[test]
644    fn a_malformed_table_count_is_a_decode_error() {
645        let bytes = u32::MAX.to_le_bytes().to_vec();
646        assert!(decode_functions(&bytes).is_err());
647        assert!(decode_labels(&bytes).is_err());
648        assert!(decode_strings(&bytes).is_err());
649    }
