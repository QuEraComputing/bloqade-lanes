1//! Writers for vihaco's section-based container formats.
2//!
3//! vihaco 0.4 ships **readers only** — [`vihaco::BytecodeFile::from_bytes`] and
4//! [`vihaco::SstFile::from_text`] parse its container, but nothing emits one. So
5//! we own the two emitters here, matching vihaco's layout byte for byte.
6//!
7//! The round-trip tests are what keep these honest: everything written here is
8//! read back through vihaco's own parser, so a layout mistake fails the build
9//! rather than producing a file only we can read.
10//!
11//! ## Shape of a lanes program
12//!
13//! A lanes program is a single flat `@main`, so both emitters write the minimal
14//! well-formed tree: an empty global context ([`vihaco::NoContext`]), one root
15//! section named `root`, no child sections. The [`LanesInfo`] version lives in
16//! that section's header.
17//!
18//! ## Binary layout (vihaco `VHBC`)
19//!
20//! ```text
21//! magic                : 4 bytes = b"VHBC"
22//! version              : u16 LE  = 1
23//! flags                : u16 LE  = 0
24//! context_len          : u64 LE  = 0        (empty global context)
25//! ── root section ──
26//! section_len          : u64 LE             (total, including this frame)
27//! composite_header_len : u64 LE
28//! composite header     : the LanesInfo header
29//! bytecode_len         : u64 LE
30//! bytecode             : N × INSTRUCTION_WIDTH bytes
31//! child_count          : u32 LE  = 0        (no child sections)
32//! ```
33//!
34//! ## Text layout (vihaco `sst v1`)
35//!
36//! ```text
37//! sst v1
38//!
39//! .section(root):
40//! .header(root):
41//! version 1.0
42//! .header(root).
43//! .text(root):
44//! fn @main() {
45//!   …
46//! }
47//! .text(root).
48//! .section(root).
49//! ```
50//!
51//! The global section is omitted entirely, which vihaco reads as an empty
52//! context.

59/// The one section name vihaco accepts for a file's root.
60pub const ROOT_SECTION: &str = "root";

62/// The global context of a lanes file: empty.
63///
64/// The global context exists to resolve child-section *names*, and a lanes
65/// program has no child sections — so there is nothing to carry and nothing to
66/// resolve. vihaco's own [`vihaco::NoContext`] would say the same thing, but it
67/// only implements the SST half of the pair ([`vihaco::SstGlobalContext`],
68/// not [`vihaco::BytecodeGlobalContext`]), so it cannot be used on the
69/// binary path.
70#[derive(Debug, Clone, Copy, Default, PartialEq, Eq)]
71pub struct LanesContext;
72

79impl vihaco::BytecodeGlobalContext for LanesContext {
80    fn from_bytes(bytes: &[u8]) -> eyre::Result<Self> {
81        if bytes.is_empty() {
82            Ok(Self)
83        } else {
84            Err(eyre::eyre!(
85                "a lanes program carries no global context, found {} byte(s)",
86                bytes.len()
87            ))
88        }
89    }
90}
91
92impl vihaco::SstGlobalContext for LanesContext {
93    fn from_text(text: &str) -> eyre::Result<Self> {
94        if text.trim().is_empty() {
95            Ok(Self)
96        } else {
97            Err(eyre::eyre!(
98                "a lanes program carries no global context, found {:?}",
99                text.trim()
100            ))
101        }
102    }
103}

105/// Byte lengths of the fixed fields in vihaco's binary container. Named here so
106/// the emitter's arithmetic reads against the layout in the module docs.
107pub(super) mod len {
108    /// `magic` + `version` + `flags` + `context_len`.
109    ///
110    /// The reader's truncation guard in [`super::super::program::from_binary`]
111    /// uses this too: a buffer shorter than the file header is the one
112    /// truncation we measure ourselves, and it has to agree with what the
113    /// writer lays down.
114    pub const FILE_HEADER: usize = 4 + 2 + 2 + 8;
115    /// `section_len` + `composite_header_len`.
116    pub const SECTION_FRAME: usize = 8 + 8;
117    /// `bytecode_len`.
118    pub const BYTECODE_HEADER: usize = 8;
119    /// `child_count`.
120    pub const CHILD_TABLE_HEADER: usize = 4;
121}

123/// Serialize a program into vihaco's `VHBC` container.
124///
125/// Instructions go through [`bytecode::encode`] because neither half of the
126/// composite carries a codec of its own.
127pub fn to_binary(program: &Program) -> eyre::Result<Vec<u8>> {
128    let mut header = Vec::new();
129    program
130        .extra
131        .write_bytes(&mut header)
132        .expect("writing a header to a Vec cannot fail");
133
134    let mut code = Vec::new();
135    for inst in &program.code {
136        bytecode::encode(inst)?
137            .write_bytes(&mut code)
138            .expect("writing instruction bytes to a Vec cannot fail");
139    }
140
141    let section_len = len::SECTION_FRAME
142        + header.len()
143        + len::BYTECODE_HEADER
144        + code.len()
145        + len::CHILD_TABLE_HEADER;
146
147    let mut out = Vec::with_capacity(len::FILE_HEADER + section_len);
148    // ── file header ──
149    out.extend_from_slice(vihaco::MAGIC);
150    out.extend_from_slice(&vihaco::VERSION.to_le_bytes());
151    out.extend_from_slice(&vihaco::FLAGS.to_le_bytes());
152    out.extend_from_slice(&0u64.to_le_bytes()); // empty global context
153    // ── root section ──
154    out.extend_from_slice(&(section_len as u64).to_le_bytes());
155    out.extend_from_slice(&(header.len() as u64).to_le_bytes());
156    out.extend_from_slice(&header);
157    out.extend_from_slice(&(code.len() as u64).to_le_bytes());
158    out.extend_from_slice(&code);
159    out.extend_from_slice(&0u32.to_le_bytes()); // no child sections
160    Ok(out)
161}

163/// Emit a program as vihaco's `sst v1` text container.
164///
165/// `body` is the already-rendered `fn @main() { … }` block.
166pub fn to_sst(info: &LanesInfo, body: &str) -> String {
167    let mut out = String::from("sst v1\n\n");
168    out.push_str(&format!(".section({ROOT_SECTION}):\n"));
169    out.push_str(&format!(".header({ROOT_SECTION}):\n"));
170    out.push_str(&info.to_string());
171    out.push('\n');
172    out.push_str(&format!(".header({ROOT_SECTION}).\n"));
173    out.push_str(&format!(".text({ROOT_SECTION}):\n"));
174    out.push_str(body);
175    out.push_str(&format!(".text({ROOT_SECTION}).\n"));
176    out.push_str(&format!(".section({ROOT_SECTION}).\n"));
177    out
178}
