1//! Flat program container for the vihaco-backed ISA.
2//!
3//! A program is a [`Version`] plus a `Vec<`[`MachineInstruction`]`>` whose
4//! functions delimit themselves with `func_start`/`func_end`, alongside the
5//! symbol tables those markers index: functions, labels, and the string
6//! interner they name each other through.
7//!
8//! The markers are what make the layout self-describing. A function's extent
9//! comes from the instructions being read rather than from a span recorded
10//! beside them, so the two cannot disagree: `from_binary` re-derives the whole
11//! function table from the code stream on load rather than trusting the extents
12//! the container carried.
13//!
14//! vihaco's [`LocalModule`] / loader machinery carries the structured-language
15//! support, so we keep a thin container and delegate the per-instruction work
16//! to the mirror ISA's derived codec ([`WriteBytes`]/[`FromBytes`], see
17//! [`super::bytecode`]) and to the text parser in [`super::text`].
18//!
19//! ## Container
20//!
21//! Both serialized forms are vihaco's own section-based containers — binary
22//! `VHBC` and text `sst v1` — carrying one root section whose header is
23//! [`LanesInfo`]. Reading goes through [`vihaco::BytecodeFile`]; writing goes
24//! through [`super::container`], because vihaco ships no emitters. See that
25//! module for the byte layout.
26

37/// The section header of a lanes program: everything the container carries
38/// about the program besides its code.
39///
40/// vihaco's header traits are blanket impls over `FromBytes` / `WriteBytes` /
41/// `FromText`, so implementing those three makes this usable as both a
42/// [`vihaco::BytecodeHeader`] and a [`vihaco::SstHeader`].
43#[derive(Debug, Clone, PartialEq, Eq)]
44pub struct LanesInfo {
45    pub version: Version,
46}
47
48impl Default for LanesInfo {
49    fn default() -> Self {
50        Self {
51            version: Version::new(0, 0),
52        }
53    }
54}
55
56impl std::fmt::Display for LanesInfo {
57    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
58        write!(f, "version {}", self.version)
59    }
60}

62/// Binary header: the version packed into a `u32` LE.
63impl WriteBytes for LanesInfo {
64    fn write_bytes<W: std::io::Write>(&self, io: &mut W) -> eyre::Result<()> {
65        let packed: u32 = self.version.into();
66        io.write_all(&packed.to_le_bytes())?;
67        Ok(())
68    }
69}
70
71impl FromBytes for LanesInfo {
72    fn from_bytes<R: std::io::Read>(bytes: &mut R) -> eyre::Result<Self> {
73        let mut buf = [0u8; 4];
74        bytes.read_exact(&mut buf)?;
75        Ok(Self {
76            version: Version::from(u32::from_le_bytes(buf)),
77        })
78    }
79}

81/// Text header: the `version <major>.<minor>` directive.
82impl vihaco::traits::FromText for LanesInfo {
83    fn from_text(text: &str) -> eyre::Result<Self> {
84        let rest = text
85            .trim()
86            .strip_prefix("version")
87            .ok_or_else(|| eyre::eyre!("missing version header"))?;
88        let (major, minor) = rest
89            .trim()
90            .split_once('.')
91            .ok_or_else(|| eyre::eyre!("expected `version <major>.<minor>`"))?;
92        Ok(Self {
93            version: Version::new(major.trim().parse()?, minor.trim().parse()?),
94        })
95    }
96}

100/// A Bloqade Lanes program: a vihaco `LocalModule` specialised to our ISA. A
101/// single `@main` function's worth of flat code plus the version in `extra`.
102pub type Program = LocalModule<MachineInstruction, Value, Type, LanesInfo>;
103
104/// Build a `Program` from a version + flat instruction list, wrapped in a
105/// single `@main`.
106///
107/// This is the ONE constructor used by both binary and text loading, so all
108/// `Program`s built from the same (version, code) compare equal regardless of
109/// source. It declares the `@main` function table entry that rendering and
110/// execution both need — a `Program` with code but no functions would emit
111/// nothing and have no entry point.
112///
113/// Programs with several functions or with labels come from
114/// [`super::resolve::resolve`] instead; this is the flat case.
115#[allow(clippy::field_reassign_with_default)] // `LocalModule` is a foreign type; struct-literal init is not possible
116pub fn from_code(version: Version, code: Vec<MachineInstruction>) -> eyre::Result<Program> {
117    // The caller supplies a function *body*; the markers are this function's
118    // job to add. Accepting a list that already has them wrapped it twice —
119    // nested `fn` blocks with one closing brace, unparseable text, and a
120    // binary that does not compare equal to what produced it. The migration
121    // guide documents the markers as visible in `Program.instructions`, so
122    // `Program(v, list(p.instructions))` is the obvious thing to try; it now
123    // says what to do instead.
124    if code.iter().any(|i| {
125        matches!(
126            i,
127            MachineInstruction::Cpu(
128                vihaco_cpu::RuntimeInstruction::FunctionStart
129                    | vihaco_cpu::RuntimeInstruction::FunctionEnd
130            )
131        )
132    }) {
133        eyre::bail!(
134            "instruction list already carries func_start/func_end; these delimit a \
135             function and are added here. To copy a program, use from_binary/to_binary \
136             or from_text/to_text."
137        );
138    }
139
140    // Wrap the body in the function markers, so a program built from a bare
141    // instruction list is in the same shape as one resolved from text. The
142    // format has one layout, not two: a reader can rely on `func_start`
143    // delimiting every function because there is no way to build a program
144    // without it.
145    let mut wrapped = Vec::with_capacity(code.len() + 2);
146    wrapped.push(MachineInstruction::Cpu(
147        vihaco_cpu::RuntimeInstruction::FunctionStart,
148    ));
149    wrapped.extend(code);
150    wrapped.push(MachineInstruction::Cpu(
151        vihaco_cpu::RuntimeInstruction::FunctionEnd,
152    ));
153    let code = wrapped;
154
155    let end_address = code.len() as u32;
156    let mut m = Program::default();
157    m.code = code;
158    m.strings = vec!["main".to_owned()];
159    m.functions = vec![FunctionInfo {
160        name: 0,
161        signature: Signature {
162            params: Vec::new(),
163            ret: Vec::new(),
164        },
165        local_count: 0,
166        start_address: 0,
167        end_address,
168        file: 0,
169    }];
170    m.main_function = Some(0);
171    m.extra = LanesInfo { version };
172    Ok(m)
173}

175/// Error from binary (de)serialization.
176///
177/// The variants are unchanged from the pre-container format because each maps to
178/// a Python exception class; what changed is where they come from. vihaco's
179/// container parser reports failures as `eyre::Report`, so `classify` sorts
180/// those messages back into these variants.
181#[derive(Debug, Clone, PartialEq, Eq)]
182pub enum BinaryError {
183    /// The file did not start with vihaco's `VHBC` magic.
184    BadMagic,
185    /// Buffer ended before a complete header, section or instruction word.
186    Truncated { expected: usize, got: usize },
187    /// The code region length is not a multiple of the instruction width.
188    UnalignedCode { len: usize },
189    /// A word held an opcode/payload vihaco could not decode, or the container
190    /// was otherwise malformed.
191    Decode { pc: usize, message: String },
192    /// The program could not be *written*: it uses something the container has
193    /// no encoding for (see [`super::container::to_binary`]'s refusals).
194    ///
195    /// Distinct from [`BinaryError::Decode`] because nothing was being decoded
196    /// and there is no instruction to blame — reporting these as a decode
197    /// failure at `pc 0` named a program counter that had nothing to do with
198    /// the problem.
199    Encode { message: String },
200}
201
202impl std::fmt::Display for BinaryError {
203    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
204        match self {
205            BinaryError::BadMagic => write!(f, "bad magic bytes (expected VHBC)"),
206            BinaryError::Truncated { expected, got } => {
207                write!(f, "truncated: expected {expected} bytes, got {got}")
208            }
209            BinaryError::UnalignedCode { len } => write!(
210                f,
211                "code length {len} is not a multiple of {}",
212                bytecode::instruction_width()
213            ),
214            BinaryError::Decode { pc, message } => {
215                write!(f, "decode error at instruction {pc}: {message}")
216            }
217            BinaryError::Encode { message } => write!(f, "cannot encode program: {message}"),
218        }
219    }
220}

224/// Length of vihaco's fixed file header, taken from the writer's own layout so
225/// the reader's truncation guard cannot drift away from what is emitted.
226use super::container::len::FILE_HEADER as FILE_HEADER_LEN;
227
228/// Sort a vihaco container/decode error into a [`BinaryError`].
229///
230/// vihaco reports these as free-form `eyre` messages, so this matches on their
231/// text. A message we do not recognise still surfaces in full via
232/// [`BinaryError::Decode`] — the classification only decides which Python
233/// exception class is raised, never whether the error is reported.
234fn classify(err: &eyre::Report) -> BinaryError {
235    let msg = err.to_string();
236    if msg.contains("invalid bytecode magic") {
237        BinaryError::BadMagic
238    } else if let Some(len) = unaligned_len(&msg) {
239        BinaryError::UnalignedCode { len }
240    } else {
241        // vihaco's structural complaints (section extends past end, header out
242        // of bounds, length mismatch, …) carry no byte counts we could put in
243        // `Truncated`, so they surface with their message intact rather than
244        // with two invented zeroes. `Truncated` is reserved for the one case we
245        // measure ourselves: a buffer too short to hold the file header.
246        BinaryError::Decode {
247            pc: 0,
248            message: msg,
249        }
250    }
251}

253/// Recover the offending length from vihaco's
254/// "bytecode length {len} is not a multiple of instruction width {w}" message.
255fn unaligned_len(msg: &str) -> Option<usize> {
256    let rest = msg.strip_prefix("bytecode length ")?;
257    if !rest.contains("is not a multiple of instruction width") {
258        return None;
259    }
260    rest.split_whitespace().next()?.parse().ok()
261}

273/// Deserialize from vihaco's `VHBC` container.
274pub fn from_binary(bytes: &[u8]) -> Result<Program, BinaryError> {
275    // Check the magic up front: vihaco needs a full file header before it can
276    // report anything, so a short-but-wrong-magic buffer would otherwise come
277    // back as a truncation rather than the more useful `BadMagic`.
278    if bytes.len() >= vihaco::MAGIC.len() && &bytes[..vihaco::MAGIC.len()] != vihaco::MAGIC {
279        return Err(BinaryError::BadMagic);
280    }
281    if bytes.len() < FILE_HEADER_LEN {
282        return Err(BinaryError::Truncated {
283            expected: FILE_HEADER_LEN,
284            got: bytes.len(),
285        });
286    }
287
288    let file =
289        BytecodeFile::<LanesContext>::from_bytes(bytes.to_vec()).map_err(|e| classify(&e))?;
290    let root = file.root();
291    let info: LanesInfo = root.decode_header().map_err(|e| classify(&e))?;
292    let code = root
293        .decode_instructions::<BytecodeInstruction>()
294        .map_err(|e| classify(&e))?
295        .into_iter()
296        .map(bytecode::decode)
297        .collect();
298
299    // Built directly rather than through `from_code`, which wraps its input
300    // in the function markers — the decoded stream already carries them.
301    // `reconcile_function_spans` then derives the table from those markers.
302    // `LocalModule` is a foreign type, so it is built field by field.
303    let mut program = Program {
304        code,
305        extra: LanesInfo {
306            version: info.version,
307        },
308        ..Default::default()
309    };
310    super::container::read_tables(&root, &mut program).map_err(|e| classify(&e))?;
311    reconcile_function_spans(&mut program);
312    if program.functions.is_empty() {
313        return Err(BinaryError::Decode {
314            pc: 0,
315            message: "no func_start/func_end markers: this container predates the \
316                      function layout and cannot be loaded. Re-assemble it from source."
317                .to_owned(),
318        });
319    }
320    // Refused on the way in for the same reason `to_binary` refuses it on the
321    // way out: vihaco's grammar spells a return type as `-> Ty`, so a
322    // signature with more has no text form, and rendering one would drop all
323    // but the first without saying so. Rejecting keeps every loadable program
324    // renderable.
325    if let Some(f) = program.functions.iter().find(|f| f.signature.ret.len() > 1) {
326        return Err(BinaryError::Decode {
327            pc: 0,
328            message: format!(
329                "function signature declares {} return values; the text form \
330                 has syntax for at most one",
331                f.signature.ret.len()
332            ),
333        });
334    }
335    resolve_entry_point(&mut program)?;
336    Ok(program)
337}

339/// Re-derive every function's span from the `func_start`/`func_end` markers in
340/// the code, keeping only the names the table supplied.
341///
342/// The code stream is the authority on where a function begins and ends; the
343/// table is an index over it. Trusting the decoded spans instead let a
344/// well-formed container carry a table that disagreed with its own code — an
345/// `end_address` past the end of `code` panicked the disassembler, and a span
346/// that under-covered the code made instructions disappear from the rendered
347/// text with no error. Re-deriving makes both unrepresentable.
348///
349/// A stream with no markers is left alone: that is the flat single-`@main`
350/// shape [`from_code`] builds, and its span is derived from the code already.
351fn reconcile_function_spans(program: &mut Program) {
352    use super::machine::MachineInstruction as M;
353    use vihaco_cpu::RuntimeInstruction as C;
354
355    let mut spans: Vec<(u32, u32)> = Vec::new();
356    let mut open: Option<u32> = None;
357    for (address, inst) in program.code.iter().enumerate() {
358        match inst {
359            M::Cpu(C::FunctionStart) => open = Some(address as u32),
360            M::Cpu(C::FunctionEnd) => {
361                if let Some(start) = open.take() {
362                    spans.push((start, address as u32 + 1));
363                }
364            }
365            _ => {}
366        }
367    }
368    // An unterminated final function still owns the rest of the code.
369    if let Some(start) = open {
370        spans.push((start, program.code.len() as u32));
371    }
372
373    // Names come from the table, matched on `start_address` — the field the
374    // emitter already writes and the one thing the table and the code agree
375    // about. Matching positionally assumed the table was in layout order:
376    // a well-formed table listing `[helper@3, main@0]` silently renamed both,
377    // and `LanesMachine::run` then entered the wrong function reporting
378    // success. A short table left `name: u32::MAX`, which renders as `fn @?()`
379    // and re-parses as a duplicate once two functions have it.
380    let named = std::mem::take(&mut program.functions);
381    program.functions = spans
382        .into_iter()
383        .map(|(start_address, end_address)| {
384            let source = named.iter().find(|f| f.start_address == start_address);
385            FunctionInfo {
386                name: source.map_or(u32::MAX, |f| f.name),
387                // Carried over, not rebuilt. The *spans* come from the code
388                // because the markers are authoritative about extent; a
389                // signature has no representation in the code stream at all,
390                // so the table is its only source. Hardcoding it empty here
391                // was invisible while nothing populated it, and became a
392                // silent drop the moment functions could declare parameters.
393                signature: source.map_or_else(
394                    || Signature {
395                        params: Vec::new(),
396                        ret: Vec::new(),
397                    },
398                    |f| f.signature.clone(),
399                ),
400                local_count: source.map_or(0, |f| f.local_count),
401                start_address,
402                end_address,
403                file: source.map_or(0, |f| f.file),
404            }
405        })
406        .collect();
407}

409/// Point `main_function` at the function actually named `main`.
410///
411/// The container records the function table but not which entry is the entry
412/// point, so `from_code`'s default of `Some(0)` survived `read_tables` and was
413/// right only when `@main` happened to be laid out first. A program whose text
414/// declared `@helper` before `@main` came back reporting function 0 as the
415/// entry — the same program, unequal across a round-trip.
416///
417/// Recovered by name rather than recorded as a new field: the name is already
418/// in the string table, `super::resolve` derives it the same way from text,
419/// and deriving it in both places is what makes the two agree. (PPVM does
420/// serialize the index, because a PPVM module may legitimately have no `main`;
421/// `resolve` makes ours mandatory, so there is no absent case to encode.)
422fn resolve_entry_point(program: &mut Program) -> Result<(), BinaryError> {
423    let main = program.functions.iter().position(|f| {
424        program
425            .strings
426            .get(f.name as usize)
427            .is_some_and(|name| name == "main")
428    });
429    match main {
430        Some(index) => {
431            program.main_function = Some(index as u32);
432            Ok(())
433        }
434        None => Err(BinaryError::Decode {
435            pc: 0,
436            message: "function table declares no @main".to_owned(),
437        }),
438    }
439}

686    #[test]
687    fn unaligned_code_rejected() {
688        // Grow the bytecode region by one byte — and the section that holds it,
689        // so the container stays well-formed. The only fault is that the region
690        // is no longer a whole number of instruction words.
691        //
692        // Offsets are read out of the file rather than hard-coded, so this keeps
693        // working when the layout around them changes.
694        let mut bytes =
695            to_binary(&from_code(Version::new(1, 0), vec![M::Cpu(C::Halt)]).unwrap()).unwrap();
696
697        let read_u64 =
698            |b: &[u8], at: usize| u64::from_le_bytes(b[at..at + 8].try_into().unwrap()) as usize;
699        let context_len = read_u64(&bytes, 8);
700        let section = FILE_HEADER_LEN + context_len;
701        let composite_header_len = read_u64(&bytes, section + 8);
702        let bytecode_len_at = section + 16 + composite_header_len;
703        let bytecode_at = bytecode_len_at + 8;
704        let code_len = read_u64(&bytes, bytecode_len_at);
705
706        let bump = |buf: &mut Vec<u8>, at: usize| {
707            let v = read_u64(buf, at) + 1;
708            buf[at..at + 8].copy_from_slice(&(v as u64).to_le_bytes());
709        };
710        bump(&mut bytes, section); // section_len
711        bump(&mut bytes, bytecode_len_at);
712        bytes.insert(bytecode_at + code_len, 0);
713
714        // The inserted byte shifts everything after the code, so the child
715        // sections' recorded offsets move with it. Without this the container
716        // rejects the file for a structural reason and never reaches the
717        // alignment check.
718        let child_table = bytecode_at + code_len + 1;
719        let child_count =
720            u32::from_le_bytes(bytes[child_table..child_table + 4].try_into().unwrap()) as usize;
721        for i in 0..child_count {
722            bump(&mut bytes, child_table + 4 + i * 12 + 4);
723        }
724
725        assert!(
726            matches!(from_binary(&bytes), Err(BinaryError::UnalignedCode { len }) if len
727                == code_len + 1),
728            "got {:?}",
729            from_binary(&bytes)
730        );
731    }
