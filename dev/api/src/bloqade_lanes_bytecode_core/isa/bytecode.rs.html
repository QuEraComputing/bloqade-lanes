1//! Binary codec for [`MachineInstruction`].
2//!
3//! ## Why a mirror ISA
4//!
5//! Neither half of the composite can be encoded directly. vihaco-cpu's
6//! instruction enums implement `Parse` and carry runtime values but no
7//! `WriteBytes`/`FromBytes`/`OpCode`, and the enum `#[composite]` generates
8//! derives only `Debug`/`Clone`. So the encodable form is declared here as a
9//! parallel enum that *does* derive [`vihaco::Instruction`], with explicit
10//! conversions either way.
11//!
12//! This is the same shape PPVM uses (`BytecodeCpu` / `BytecodeCircuit` /
13//! `BytecodeInstruction`), and for the same reason. It is duplication, but it
14//! is duplication vihaco currently forces on every consumer that wants both a
15//! CPU component and a binary format.
16//!
17//! The round-trip test is what keeps the mirror honest: every variant of
18//! [`MachineInstruction`] must survive `encode` → bytes → `decode`.
19//!
20//! ## Layout
21//!
22//! vihaco assigns opcodes by declaration position, so [`BytecodeInstruction`]'s
23//! two variants tag the device (`0x00` cpu, `0x01` lanes) and the nested enum's
24//! opcode follows. Width is derived as the widest variant.
25

33/// Width, in bytes, of every encoded instruction word.
34///
35/// Derived, not chosen: vihaco computes it as `1 + Σ field widths` of the
36/// widest variant. Today that is 14 — a device byte, an opcode byte, and
37/// three `u32`s — reached by both `Cpu(Span(u32, u32, u32))` and
38/// `Lanes(NewArray(u32, u32, u32))`. (`Const(BytecodeType, BytecodeValue)` is
39/// only 12.) Every narrower instruction is zero-padded to match, so a program
40/// is N concatenated words.
41pub fn instruction_width() -> u32 {
42    BytecodeInstruction::width()
43}

75/// Generate the two-way mapping for a mirror whose variants match one-to-one.
76macro_rules! mirror {
77    ($source:ident, $encoded:ident, $encode:ident, $decode:ident;
78     $($variant:ident $(($binding:ident))?),+ $(,)?) => {
79        fn $encode(value: $source) -> $encoded {
80            match value {
81                $( $source::$variant $(($binding))? => $encoded::$variant $(($binding))?, )+
82            }
83        }
84
85        fn $decode(value: $encoded) -> $source {
86            match value {
87                $( $encoded::$variant $(($binding))? => $source::$variant $(($binding))?, )+
88            }
89        }
90    };
91}

187/// A 16-bit opcode that identifies an instruction across both devices:
188/// `(device_code << 8) | instruction_code`.
189///
190/// [`BytecodeInstruction`]'s own opcode is just the device tag — `0x00` cpu,
191/// `0x01` lanes — because the instruction's own code sits in the nested enum.
192/// On its own that does not distinguish `move` from `cz`, so the two are packed
193/// together here. This is what the Python `Instruction.opcode` reports.
194///
195/// Both halves shift when either instruction set gains a variant, so compare
196/// identity with `op_name()` rather than a literal.
197///
198/// Neither half is written down here. The device tag is read back off the
199/// constructed [`BytecodeInstruction`], so it is by construction the byte that
200/// goes on disk — literals would have drifted silently the first time either
201/// the variant order or a `#[device(..)]` attribute changed, and every test
202/// compares via `op_name`, so nothing would have caught it.
203///
204/// An instruction with no encodable form — today just a runtime label, whose
205/// identifier is meaningless outside its parse — reports
206/// [`NO_ENCODABLE_FORM`] rather than colliding with a real opcode.
207pub fn packed_opcode(inst: &MachineInstruction) -> u16 {
208    let Ok(encoded) = encode(inst) else {
209        return NO_ENCODABLE_FORM;
210    };
211    let device = OpCode::opcode(&encoded) as u16;
212    let code = match &encoded {
213        BytecodeInstruction::Cpu(i) => OpCode::opcode(i),
214        BytecodeInstruction::Lanes(i) => OpCode::opcode(i),
215    } as u16;
216    (device << 8) | code
217}
218
219/// Reported by [`packed_opcode`] for an instruction that cannot be encoded.
220///
221/// `0xFFFF` is outside the packed space — the device tag is a
222/// [`BytecodeInstruction`] variant index and the code an inner variant index,
223/// so neither half can reach `0xFF`. A plain `0` would have collided with the
224/// first CPU instruction's real opcode.
225pub const NO_ENCODABLE_FORM: u16 = 0xFFFF;

227/// Map a runtime instruction to its encodable mirror.
228pub fn encode(inst: &MachineInstruction) -> eyre::Result<BytecodeInstruction> {
229    Ok(match inst {
230        MachineInstruction::Cpu(inst) => BytecodeInstruction::Cpu(encode_cpu(inst)?),
231        MachineInstruction::Lanes(inst) => BytecodeInstruction::Lanes(encode_lanes(inst)),
232    })
233}
234
235/// Map an encodable mirror back to its runtime instruction.
236pub fn decode(inst: BytecodeInstruction) -> MachineInstruction {
237    match inst {
238        BytecodeInstruction::Cpu(inst) => MachineInstruction::Cpu(decode_cpu(inst)),
239        BytecodeInstruction::Lanes(inst) => MachineInstruction::Lanes(decode_lanes(inst)),
240    }
241}

243fn encode_cpu(inst: &CpuInstruction) -> eyre::Result<BytecodeCpu> {
244    use CpuInstruction as C;
245    Ok(match inst {
246        C::Span(a, b, c) => BytecodeCpu::Span(*a, *b, *c),
247        C::FunctionStart => BytecodeCpu::FunctionStart,
248        C::FunctionEnd => BytecodeCpu::FunctionEnd,
249        C::Breakpoint => BytecodeCpu::Breakpoint,
250        C::Branch(v) => BytecodeCpu::Branch(*v),
251        C::ConditionalBranch(a, b) => BytecodeCpu::ConditionalBranch(*a, *b),
252        C::Return(v) => BytecodeCpu::Return(*v),
253        C::IndirectCall => BytecodeCpu::IndirectCall,
254        C::Call(a, b) => BytecodeCpu::Call(*a, *b),
255        C::Halt => BytecodeCpu::Halt,
256        C::Print => BytecodeCpu::Print,
257        C::Load(t, v) => BytecodeCpu::Load(encode_type(*t), *v),
258        C::Store(t, v) => BytecodeCpu::Store(encode_type(*t), *v),
259        C::Dup => BytecodeCpu::Dup,
260        C::HeapAlloc(v) => BytecodeCpu::HeapAlloc(*v),
261        C::GetItem => BytecodeCpu::GetItem,
262        C::HeapDealloc => BytecodeCpu::HeapDealloc,
263        C::Const(t, v) => BytecodeCpu::Const(encode_type(*t), encode_value(*v)),
264        C::Add(t) => BytecodeCpu::Add(encode_type(*t)),
265        C::Sub(t) => BytecodeCpu::Sub(encode_type(*t)),
266        C::Mul(t) => BytecodeCpu::Mul(encode_type(*t)),
267        C::Div(t) => BytecodeCpu::Div(encode_type(*t)),
268        C::Rem(t) => BytecodeCpu::Rem(encode_type(*t)),
269        C::Neg(t) => BytecodeCpu::Neg(encode_type(*t)),
270        C::Shl(t) => BytecodeCpu::Shl(encode_type(*t)),
271        C::Shr(t) => BytecodeCpu::Shr(encode_type(*t)),
272        C::Rol(t) => BytecodeCpu::Rol(encode_type(*t)),
273        C::Ror(t) => BytecodeCpu::Ror(encode_type(*t)),
274        C::BitAnd(t) => BytecodeCpu::BitAnd(encode_type(*t)),
275        C::BitOr(t) => BytecodeCpu::BitOr(encode_type(*t)),
276        C::BitXor(t) => BytecodeCpu::BitXor(encode_type(*t)),
277        C::Not => BytecodeCpu::Not,
278        C::And => BytecodeCpu::And,
279        C::Or => BytecodeCpu::Or,
280        C::Xor => BytecodeCpu::Xor,
281        C::Eq(t) => BytecodeCpu::Eq(encode_type(*t)),
282        C::Ne(t) => BytecodeCpu::Ne(encode_type(*t)),
283        C::Lt(t) => BytecodeCpu::Lt(encode_type(*t)),
284        C::Gt(t) => BytecodeCpu::Gt(encode_type(*t)),
285        C::Le(t) => BytecodeCpu::Le(encode_type(*t)),
286        C::Ge(t) => BytecodeCpu::Ge(encode_type(*t)),
287        C::Label(_) => {
288            return Err(eyre::eyre!(
289                "a runtime label carries a parse-local identifier and cannot be encoded"
290            ));
291        }
292    })
293}

508#[cfg(test)]
509mod tests {
510    use super::tests_support::every_instruction as samples;
511    use super::*;
512    use vihaco::instruction::{FromBytes, WriteBytes};
513    use vihaco_parser::Ident;
514
515    #[test]
516    fn every_instruction_round_trips_through_bytes() {
517        for inst in samples() {
518            let encoded = encode(&inst).expect("sample should encode");
519            let mut buf = Vec::new();
520            encoded.write_bytes(&mut buf).unwrap();
521            assert_eq!(
522                buf.len(),
523                instruction_width() as usize,
524                "{inst:?} did not fill a full word"
525            );
526
527            let read = BytecodeInstruction::from_bytes(&mut std::io::Cursor::new(buf)).unwrap();
528            assert_eq!(read, encoded, "byte round-trip changed {inst:?}");
529            // And the mirror maps back to the same runtime instruction.
530            assert_eq!(format!("{:?}", decode(read)), format!("{inst:?}"));
531        }
532    }
533
534    #[test]
535    fn a_flat_stream_decodes_without_desync() {
536        let program = samples();
537        let mut bytes = Vec::new();
538        for inst in &program {
539            encode(inst).unwrap().write_bytes(&mut bytes).unwrap();
540        }
541        assert_eq!(bytes.len(), program.len() * instruction_width() as usize);
542
543        let mut cursor = std::io::Cursor::new(bytes);
544        for inst in &program {
545            let read = BytecodeInstruction::from_bytes(&mut cursor).unwrap();
546            assert_eq!(format!("{:?}", decode(read)), format!("{inst:?}"));
547        }
548    }
549
550    #[test]
551    fn a_runtime_label_cannot_be_encoded() {
552        // It carries an identifier that only means anything to the parse that
553        // produced it, so encoding is refused rather than silently lossy.
554        let label = MachineInstruction::Cpu(CpuInstruction::Label(Ident("loop".into())));
555        let err = encode(&label).unwrap_err().to_string();
556        assert!(err.contains("cannot be encoded"), "got {err}");
557        // And it reports a code no real instruction can occupy, rather than
558        // colliding with the first CPU opcode.
559        assert_eq!(packed_opcode(&label), NO_ENCODABLE_FORM);
560    }

562    /// The opcode Python reports must be the bytes that went to disk.
563    ///
564    /// Nothing else checks this: every other test compares instructions via
565    /// `op_name`, so a `packed_opcode` that disagreed with the encoding would
566    /// stay green. The first two bytes of a word are the device tag and the
567    /// instruction's own code, which is exactly what the packing claims to be.
568    #[test]
569    fn the_packed_opcode_is_the_first_two_bytes_on_disk() {
570        for inst in samples() {
571            let mut buf = Vec::new();
572            encode(&inst).unwrap().write_bytes(&mut buf).unwrap();
573            assert_eq!(
574                packed_opcode(&inst),
575                ((buf[0] as u16) << 8) | buf[1] as u16,
576                "{inst:?}: packed opcode disagrees with its encoding {:02x?}",
577                &buf[..2]
578            );
579        }
580    }

582    /// The word width is what the doc on `instruction_width` claims it is.
583    ///
584    /// 14 = 1 device byte + 1 opcode byte + three `u32`s, reached by the
585    /// widest variant on *either* device. It is derived, so it moves the
586    /// moment either gains a wider operand — which is worth noticing, since
587    /// it is the on-disk format.
588    #[test]
589    fn the_word_width_comes_from_the_widest_variant() {
590        assert_eq!(instruction_width(), 14);
591        for widest in [
592            BytecodeInstruction::Cpu(BytecodeCpu::Span(1, 2, 3)),
593            BytecodeInstruction::Lanes(BytecodeLanes::NewArray(1, 2, 3)),
594        ] {
595            let mut buf = Vec::new();
596            widest.write_bytes(&mut buf).unwrap();
597            assert_eq!(buf.len(), 14, "{widest:?}");
598        }
599        // The `const` word is narrower, and is padded out to match.
600        let mut buf = Vec::new();
601        BytecodeInstruction::Cpu(BytecodeCpu::Const(
602            BytecodeType::F64,
603            BytecodeValue::F64(1.5),
604        ))
605        .write_bytes(&mut buf)
606        .unwrap();
607        assert_eq!(buf.len(), 14);
608        assert_eq!(&buf[12..], &[0, 0], "the last two bytes should be padding");
609    }

611    /// Distinct instructions must get distinct opcodes — the defect the
612    /// packing exists to fix was every lanes op reporting the device tag.
613    #[test]
614    fn each_instruction_has_its_own_packed_opcode() {
615        let mut seen: std::collections::HashMap<u16, MachineInstruction> = Default::default();
616        for inst in samples() {
617            let code = packed_opcode(&inst);
618            // The type is an operand, not part of the opcode, so `add i64`
619            // and `add f64` — and all nine `const` spellings — legitimately
620            // share one. Key on the mnemonic, which is what an opcode names.
621            let mnemonic = |i: &MachineInstruction| {
622                super::super::machine::to_sst_text(i)
623                    .split_whitespace()
624                    .next()
625                    .unwrap()
626                    .to_owned()
627            };
628            if let Some(other) = seen.insert(code, inst.clone()) {
629                assert_eq!(
630                    mnemonic(&other),
631                    mnemonic(&inst),
632                    "{inst:?} and {other:?} share opcode {code:#06x}"
633                );
634            }
635        }
636        // Sanity: the two devices land in different high bytes.
637        assert_ne!(
638            packed_opcode(&MachineInstruction::Cpu(CpuInstruction::Halt)) >> 8,
639            packed_opcode(&MachineInstruction::Lanes(LanesInstruction::Cz)) >> 8,
640        );
641    }
