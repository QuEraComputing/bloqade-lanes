13/// Error categories for arch spec structural validation.
14///
15/// Each variant groups related validation checks. Multiple errors
16/// can be collected in a single validation pass.
17#[derive(Debug, Clone, PartialEq, Error)]
18pub enum ArchSpecError {
19    /// Structural error (grid dimensions, word consistency, minimum counts).
20    #[error("{0}")]
21    Structure(String),
22
23    /// Per-zone bus validation error (site/word buses, membership lists).
24    #[error("{0}")]
25    ZoneBus(String),
26
27    /// Inter-zone bus validation error (zone bus entries).
28    #[error("{0}")]
29    InterZoneBus(String),
30
31    /// Bus src→dst relation contains a cycle.
32    ///
33    /// A bus is by definition a set of explicit edge transports executed
34    /// simultaneously as one AOD operation — never a permutation. A chain
35    /// (`x→y, y→z` with the final destination outside the source set) shifts
36    /// atoms like a conveyor and is physically valid; a cycle
37    /// (`x→y, y→z, z→x`, including the self-loop `x→x`) would rotate a
38    /// fully-occupied set of atoms with no empty site, which AOD hardware
39    /// cannot do (tones in a group share a direction and cannot cross).
40    /// Cyclic buses are therefore rejected outright, while
41    /// overlapping-but-acyclic buses remain legal.
42    #[error("{0}")]
43    CyclicBus(String),
44
45    /// A bus describes no single AOD operation.
46    ///
47    /// An AOD traps at the Cartesian product of its x and y tones and sweeps
48    /// each tone independently, so the lanes a bus fires together must be
49    /// **separable and order-preserving**: an atom's landing x is decided by
50    /// its column alone and its landing y by its row alone, and the sorted
51    /// source tones land on the sorted destination tones, because two tones
52    /// cannot cross, merge or split. The sources must also form a complete
53    /// grid, since the AOD traps at every crossing of its tones. A bus that
54    /// would send one row onto a diagonal, reverse the order of two columns,
55    /// or start from a diagonal, is rejected.
56    ///
57    /// On a bus that passes, any rectangle of its sources lands on a
58    /// rectangle, which is what lets the router build shots from source
59    /// positions alone, and the full bus is one rectangle at each end.
60    #[error("{0}")]
61    AodTransport(String),
62
63    /// Entangling zone pair validation error.
64    #[error("{0}")]
65    EntanglingPair(String),
66
67    /// Mode validation error.
68    #[error("{0}")]
69    Mode(String),
70
71    /// Transport path validation error.
72    #[error("{0}")]
73    Path(String),
74}

76impl ArchSpec {
77    /// Validate the arch spec against all structural rules.
78    /// Collects all errors in one pass (not fail-fast).
79    pub fn validate(&self) -> Result<(), Vec<ArchSpecError>> {
80        let mut errors = Vec::new();
81
82        let num_words = self.words.len();
83        let num_zones = self.zones.len();
84        let sites_per_word = self.sites_per_word();
85
86        // Structural invariants
87        check_minimum_counts(self, &mut errors);
88        check_non_negative_grid_spacings(self, &mut errors);
89        check_uniform_grid_dimensions(self, &mut errors);
90        check_uniform_word_site_counts(self, &mut errors);
91        check_word_site_indices(self, &mut errors);
92
93        // Per-zone bus and entangling pair validation
94        for (zone_idx, zone) in self.zones.iter().enumerate() {
95            check_zone_words_with_site_buses(zone_idx, zone, num_words, &mut errors);
96            check_zone_sites_with_word_buses(zone_idx, zone, sites_per_word, &mut errors);
97            check_zone_site_buses(zone_idx, zone, sites_per_word, &mut errors);
98            check_zone_word_buses(zone_idx, zone, num_words, sites_per_word, &mut errors);
99            check_zone_entangling_pairs(zone_idx, zone, num_words, &mut errors);
100        }
101
102        // Inter-zone bus validation
103        check_zone_buses(self, num_zones, num_words, &mut errors);
104
105        // Every bus must be one AOD operation
106        check_aod_transports(self, sites_per_word, &mut errors);
107
108        // Mode validation
109        check_modes(self, num_zones, num_words, sites_per_word, &mut errors);
110
111        // Path validation
112        check_paths(self, num_zones, &mut errors);
113
114        // Zone physical-space overlap
115        check_zone_overlap(self, &mut errors);
116
117        if errors.is_empty() {
118            Ok(())
119        } else {
120            Err(errors)
121        }
122    }
123}

125/// At least one zone and one word must exist.
126fn check_minimum_counts(spec: &ArchSpec, errors: &mut Vec<ArchSpecError>) {
127    if spec.zones.is_empty() {
128        errors.push(ArchSpecError::Structure(
129            "at least one zone must exist".into(),
130        ));
131    }
132    if spec.words.is_empty() {
133        errors.push(ArchSpecError::Structure(
134            "at least one word must exist".into(),
135        ));
136    }

139/// All grid spacings must be non-negative. Negative spacings would produce
140/// non-monotonic positions, breaking the `Grid::bounding_box()` invariant
141/// and making position lookups ambiguous.
142fn check_non_negative_grid_spacings(spec: &ArchSpec, errors: &mut Vec<ArchSpecError>) {
143    for (idx, zone) in spec.zones.iter().enumerate() {
144        if let Some(pos) = zone.grid.x_spacing.iter().position(|&v| v < 0.0) {
145            errors.push(ArchSpecError::Structure(format!(
146                "zone {} grid x_spacing[{}] is negative ({:.6})",
147                idx, pos, zone.grid.x_spacing[pos]
148            )));
149        }
150        if let Some(pos) = zone.grid.y_spacing.iter().position(|&v| v < 0.0) {
151            errors.push(ArchSpecError::Structure(format!(
152                "zone {} grid y_spacing[{}] is negative ({:.6})",
153                idx, pos, zone.grid.y_spacing[pos]
154            )));
155        }
156    }
157}

159/// All zones must have the same grid dimensions (same num_x and num_y).
160fn check_uniform_grid_dimensions(spec: &ArchSpec, errors: &mut Vec<ArchSpecError>) {
161    if let Some(first_zone) = spec.zones.first() {
162        let ref_x = first_zone.grid.num_x();
163        let ref_y = first_zone.grid.num_y();
164        for (idx, zone) in spec.zones.iter().enumerate().skip(1) {
165            let zx = zone.grid.num_x();
166            let zy = zone.grid.num_y();
167            if zx != ref_x || zy != ref_y {
168                errors.push(ArchSpecError::Structure(format!(
169                    "zone {} grid dimensions ({}x{}) differ from zone 0 ({}x{})",
170                    idx, zx, zy, ref_x, ref_y
171                )));
172            }
173        }
174    }
175}

177/// All words must have the same number of sites.
178fn check_uniform_word_site_counts(spec: &ArchSpec, errors: &mut Vec<ArchSpecError>) {
179    if let Some(first_word) = spec.words.first() {
180        let ref_count = first_word.sites.len();
181        for (idx, word) in spec.words.iter().enumerate().skip(1) {
182            if word.sites.len() != ref_count {
183                errors.push(ArchSpecError::Structure(format!(
184                    "word {} has {} sites, expected {} (same as word 0)",
185                    idx,
186                    word.sites.len(),
187                    ref_count
188                )));
189            }
190        }
191    }
192}

194/// Word site indices must be within the grid dimensions of every zone.
195/// For each site [x, y]: x < grid.num_x() and y < grid.num_y().
196fn check_word_site_indices(spec: &ArchSpec, errors: &mut Vec<ArchSpecError>) {
197    // Use zone 0's grid as the reference (uniform dimensions already checked).
198    let (grid_x, grid_y) = match spec.zones.first() {
199        Some(z) => (z.grid.num_x(), z.grid.num_y()),
200        None => return, // no zones → nothing to check
201    };
202
203    for (word_idx, word) in spec.words.iter().enumerate() {
204        for (site_idx, site) in word.sites.iter().enumerate() {
205            let x = site[0] as usize;
206            let y = site[1] as usize;
207            if x >= grid_x {
208                errors.push(ArchSpecError::Structure(format!(
209                    "word {}, site {}: x index {} out of range (grid has {} x-positions)",
210                    word_idx, site_idx, site[0], grid_x
211                )));
212            }
213            if y >= grid_y {
214                errors.push(ArchSpecError::Structure(format!(
215                    "word {}, site {}: y index {} out of range (grid has {} y-positions)",
216                    word_idx, site_idx, site[1], grid_y
217                )));
218            }
219        }
220    }
221}

261/// Site bus src/dst must have same length and SiteRef values < sites_per_word.
262fn check_zone_site_buses(
263    zone_idx: usize,
264    zone: &Zone,
265    sites_per_word: usize,
266    errors: &mut Vec<ArchSpecError>,
267) {
268    for (bus_idx, bus) in zone.site_buses.iter().enumerate() {
269        if bus.src.len() != bus.dst.len() {
270            errors.push(ArchSpecError::ZoneBus(format!(
271                "zone {}, site_bus {}: src length ({}) != dst length ({})",
272                zone_idx,
273                bus_idx,
274                bus.src.len(),
275                bus.dst.len()
276            )));
277        }
278        for (i, sref) in bus.src.iter().enumerate() {
279            if sref.0 as usize >= sites_per_word {
280                errors.push(ArchSpecError::ZoneBus(format!(
281                    "zone {}, site_bus {}: src[{}] SiteRef({}) >= sites_per_word ({})",
282                    zone_idx, bus_idx, i, sref.0, sites_per_word
283                )));
284            }
285        }
286        for (i, sref) in bus.dst.iter().enumerate() {
287            if sref.0 as usize >= sites_per_word {
288                errors.push(ArchSpecError::ZoneBus(format!(
289                    "zone {}, site_bus {}: dst[{}] SiteRef({}) >= sites_per_word ({})",
290                    zone_idx, bus_idx, i, sref.0, sites_per_word
291                )));
292            }
293        }
294        check_bus_relation(
295            &bus.src,
296            &bus.dst,
297            &format!("zone {zone_idx}, site_bus {bus_idx}"),
298            |s| s.0.to_string(),
299            ArchSpecError::ZoneBus,
300            errors,
301        );
302    }
303}

305/// Word bus src/dst must have same length and WordRef values < number of words,
306/// and an optional `site_map` must be well formed.
307fn check_zone_word_buses(
308    zone_idx: usize,
309    zone: &Zone,
310    num_words: usize,
311    sites_per_word: usize,
312    errors: &mut Vec<ArchSpecError>,
313) {

315        if bus.src.len() != bus.dst.len() {

318                zone_idx,
319                bus_idx,
320                bus.src.len(),
321                bus.dst.len()
322            )));
323        }
324        for (i, wref) in bus.src.iter().enumerate() {
325            if wref.0 as usize >= num_words {
326                errors.push(ArchSpecError::ZoneBus(format!(
327                    "zone {}, word_bus {}: src[{}] WordRef({}) >= num_words ({})",
328                    zone_idx, bus_idx, i, wref.0, num_words
329                )));
330            }
331        }
332        for (i, wref) in bus.dst.iter().enumerate() {
333            if wref.0 as usize >= num_words {
334                errors.push(ArchSpecError::ZoneBus(format!(
335                    "zone {}, word_bus {}: dst[{}] WordRef({}) >= num_words ({})",
336                    zone_idx, bus_idx, i, wref.0, num_words
337                )));
338            }
339        }
340        check_bus_relation(
341            &bus.src,
342            &bus.dst,
343            &format!("zone {zone_idx}, word_bus {bus_idx}"),
344            |w| w.0.to_string(),
345            ArchSpecError::ZoneBus,
346            errors,
347        );
348        if let Some(map) = &bus.site_map {
349            check_word_bus_site_map(zone_idx, bus_idx, zone, map, sites_per_word, errors);
350        }
351    }

354/// A word bus's `site_map` obeys the same rules as a site bus, plus a
355/// membership rule.
356///
357/// `src`/`dst` must have equal length, every entry must be `< sites_per_word`
358/// and listed in the zone's `sites_with_word_buses` (the word-bus landing
359/// pads), and the src→dst relation must be well formed exactly like a site
360/// bus's ([`check_bus_relation`]): unique sources, unique destinations, and no
361/// cycles — the site map is part of the same AOD operation, so a rotation of
362/// sites is rejected even though the word relation already prevents a
363/// `(word, site)` location from mapping onto itself.
364fn check_word_bus_site_map(
365    zone_idx: usize,
366    bus_idx: usize,
367    zone: &Zone,
368    map: &SiteMap,
369    sites_per_word: usize,
370    errors: &mut Vec<ArchSpecError>,
371) {
372    let label = format!("zone {zone_idx}, word_bus {bus_idx} site_map");
373    if map.src.len() != map.dst.len() {
374        errors.push(ArchSpecError::ZoneBus(format!(
375            "{label}: src length ({}) != dst length ({})",
376            map.src.len(),
377            map.dst.len()
378        )));
379    }
380    for (side, refs) in [("src", &map.src), ("dst", &map.dst)] {
381        for (i, sref) in refs.iter().enumerate() {
382            if sref.0 as usize >= sites_per_word {
383                errors.push(ArchSpecError::ZoneBus(format!(
384                    "{label}: {side}[{i}] SiteRef({}) >= sites_per_word ({sites_per_word})",
385                    sref.0
386                )));
387            } else if !zone.sites_with_word_buses.contains(&(sref.0 as u32)) {
388                errors.push(ArchSpecError::ZoneBus(format!(
389                    "{label}: {side}[{i}] site {} is not in zone {zone_idx} sites_with_word_buses",
390                    sref.0
391                )));
392            }
393        }
394    }
395    check_bus_relation(
396        &map.src,
397        &map.dst,
398        &label,
399        |s| s.0.to_string(),
400        ArchSpecError::ZoneBus,
401        errors,
402    );
403}

407/// Zone bus entries must have valid zone_id and word_id, src/dst same length,
408/// and every pair must cross a zone boundary.
409fn check_zone_buses(
410    spec: &ArchSpec,
411    num_zones: usize,
412    num_words: usize,
413    errors: &mut Vec<ArchSpecError>,
414) {
415    for (bus_idx, bus) in spec.zone_buses.iter().enumerate() {
416        if bus.src.len() != bus.dst.len() {
417            errors.push(ArchSpecError::InterZoneBus(format!(
418                "zone_bus {}: src length ({}) != dst length ({})",
419                bus_idx,
420                bus.src.len(),
421                bus.dst.len()
422            )));
423        }
424
425        // Validate all ZonedWordRef entries
426        for (i, zwr) in bus.src.iter().enumerate() {
427            if zwr.zone_id as usize >= num_zones {
428                errors.push(ArchSpecError::InterZoneBus(format!(
429                    "zone_bus {}: src[{}] zone_id {} >= num_zones ({})",
430                    bus_idx, i, zwr.zone_id, num_zones
431                )));
432            }
433            if zwr.word_id as usize >= num_words {
434                errors.push(ArchSpecError::InterZoneBus(format!(
435                    "zone_bus {}: src[{}] word_id {} >= num_words ({})",
436                    bus_idx, i, zwr.word_id, num_words
437                )));
438            }
439        }
440        for (i, zwr) in bus.dst.iter().enumerate() {
441            if zwr.zone_id as usize >= num_zones {
442                errors.push(ArchSpecError::InterZoneBus(format!(
443                    "zone_bus {}: dst[{}] zone_id {} >= num_zones ({})",
444                    bus_idx, i, zwr.zone_id, num_zones
445                )));
446            }
447            if zwr.word_id as usize >= num_words {
448                errors.push(ArchSpecError::InterZoneBus(format!(
449                    "zone_bus {}: dst[{}] word_id {} >= num_words ({})",
450                    bus_idx, i, zwr.word_id, num_words
451                )));
452            }
453        }
454
455        // Every (src[i], dst[i]) pair must cross a zone boundary
456        let pair_count = bus.src.len().min(bus.dst.len());
457        for i in 0..pair_count {
458            if bus.src[i].zone_id == bus.dst[i].zone_id {
459                errors.push(ArchSpecError::InterZoneBus(format!(
460                    "zone_bus {}: pair {} does not cross a zone boundary \
461                     (src zone_id={}, dst zone_id={})",
462                    bus_idx, i, bus.src[i].zone_id, bus.dst[i].zone_id
463                )));
464            }
465        }
466
467        check_bus_relation(
468            &bus.src,
469            &bus.dst,
470            &format!("zone_bus {bus_idx}"),
471            |z| format!("z{}w{}", z.zone_id, z.word_id),
472            ArchSpecError::InterZoneBus,
473            errors,
474        );
475    }
476}

480/// One lane's `(source, destination)` physical positions.
481type TransportPair = ((f64, f64), (f64, f64));
482
483/// One bus transport: one [`TransportPair`] per lane fired together.
484type Transport = Vec<TransportPair>;
485
486/// Every bus must be one AOD operation — see [`ArchSpecError::AodTransport`].
487///
488/// A transport is every lane that can fire in one shot, which is the key
489/// `check_lane_group_consistency` groups on: a site or word bus is one
490/// transport in its zone, and a zone bus is one transport per forward source
491/// zone, since its lanes carry that zone's id. A bus with an endpoint whose
492/// position does not resolve was already reported by the structural checks,
493/// so it is skipped here rather than reported twice.
494///
495/// This is the rule the Python architecture builder enforced in
496/// `check_aod_transport` for the specs it built, made stricter: the builder
497/// let a bus's sources skip a tone crossing that held no atom, and here they
498/// must cover every crossing. Specs loaded from JSON never went through the
499/// builder, and a word bus `site_map` cannot be built there at all.
500fn check_aod_transports(spec: &ArchSpec, sites_per_word: usize, errors: &mut Vec<ArchSpecError>) {
501    let position = |zone_id: u32, word_id: u32, site_id: u32| {
502        spec.location_position(&LocationAddr {
503            zone_id,
504            word_id,
505            site_id,
506        })
507    };
508
509    for (zone_idx, zone) in spec.zones.iter().enumerate() {
510        let z = zone_idx as u32;
511        for (bus_idx, bus) in zone.site_buses.iter().enumerate() {
512            let transport: Option<Transport> = zone
513                .words_with_site_buses
514                .iter()
515                .flat_map(|&w| {
516                    bus.src
517                        .iter()
518                        .zip(&bus.dst)
519                        .map(move |(s, d)| (w, u32::from(s.0), u32::from(d.0)))
520                })
521                .map(|(w, s, d)| Some((position(z, w, s)?, position(z, w, d)?)))
522                .collect();
523            report_aod_transport(
524                transport,
525                &format!("zone {zone_idx}, site_bus {bus_idx}"),
526                errors,
527            );
528        }
529        for (bus_idx, bus) in zone.word_buses.iter().enumerate() {
530            let sites = zone.word_bus_site_pairs(bus_idx);
531            let transport: Option<Transport> = bus
532                .src
533                .iter()
534                .zip(&bus.dst)
535                .flat_map(|(sw, dw)| {
536                    sites
537                        .iter()
538                        .map(move |&(s, d)| (u32::from(sw.0), u32::from(dw.0), s, d))
539                })
540                .map(|(sw, dw, s, d)| Some((position(z, sw, s)?, position(z, dw, d)?)))
541                .collect();
542            report_aod_transport(
543                transport,
544                &format!("zone {zone_idx}, word_bus {bus_idx}"),
545                errors,
546            );
547        }
548    }
549
550    for (bus_idx, bus) in spec.zone_buses.iter().enumerate() {
551        let mut by_source_zone: BTreeMap<u8, Option<Transport>> = BTreeMap::new();
552        for (s, d) in bus.src.iter().zip(&bus.dst) {
553            let entry = by_source_zone
554                .entry(s.zone_id)
555                .or_insert_with(|| Some(Vec::new()));
556            for site in 0..sites_per_word as u32 {
557                let pair = position(s.zone_id.into(), s.word_id.into(), site).zip(position(
558                    d.zone_id.into(),
559                    d.word_id.into(),
560                    site,
561                ));
562                match pair {
563                    Some(p) => {
564                        if let Some(transport) = entry {
565                            transport.push(p);
566                        }
567                    }
568                    None => *entry = None,
569                }
570            }
571        }
572        for (zone_id, transport) in by_source_zone {
573            report_aod_transport(
574                transport,
575                &format!("zone_bus {bus_idx} (lanes from zone {zone_id})"),
576                errors,
577            );
578        }
579    }
580}

582fn report_aod_transport(
583    transport: Option<Transport>,
584    label: &str,
585    errors: &mut Vec<ArchSpecError>,
586) {
587    if let Some(pairs) = transport
588        && let Some(msg) = aod_transport_violation(&pairs)
589    {
590        errors.push(ArchSpecError::AodTransport(format!("{label}: {msg}")));
591    }
592}

594/// `None` when `pairs` is one AOD operation: the sorted source tones map onto
595/// the sorted destination tones, every atom lands where its own column and
596/// row land, and the sources cover every crossing of their tones. Otherwise
597/// a description of the first violation.
598fn aod_transport_violation(pairs: &[TransportPair]) -> Option<String> {
599    let tones = |pick: fn(&TransportPair) -> f64| -> Vec<f64> {
600        let mut tones: Vec<f64> = pairs.iter().map(pick).collect();
601        tones.sort_by(f64::total_cmp);
602        tones.dedup_by(|a, b| a.to_bits() == b.to_bits());
603        tones
604    };
605    let src_x = tones(|p| p.0.0);
606    let src_y = tones(|p| p.0.1);
607    let dst_x = tones(|p| p.1.0);
608    let dst_y = tones(|p| p.1.1);
609
610    if src_x.len() != dst_x.len() || src_y.len() != dst_y.len() {
611        return Some(format!(
612            "the bus spans a {}x{} grid of tones at its source but {}x{} at its \
613             destination; an AOD cannot add, drop or merge a tone mid-transport",
614            src_x.len(),
615            src_y.len(),
616            dst_x.len(),
617            dst_y.len()
618        ));
619    }
620
621    let tone_index = |tones: &[f64], v: f64| {
622        tones
623            .binary_search_by(|t| t.total_cmp(&v))
624            .expect("every coordinate is one of its own side's tones")
625    };
626    for &((sx, sy), (dx, dy)) in pairs {
627        let wx = dst_x[tone_index(&src_x, sx)];
628        let wy = dst_y[tone_index(&src_y, sy)];
629        if dx.to_bits() != wx.to_bits() || dy.to_bits() != wy.to_bits() {
630            return Some(format!(
631                "the atom at ({sx}, {sy}) is sent to ({dx}, {dy}), but its column and \
632                 row land at ({wx}, {wy}); an AOD sweeps whole x and y tones, which \
633                 cannot cross or trade places, so every atom must stay on its own tones"
634            ));
635        }
636    }
637
638    // Completeness: the AOD traps at every crossing of its tones, so every
639    // crossing must be one of the bus's sources. (On a separable,
640    // order-preserving bus the destinations are then a complete grid too.)
641    let sources: HashSet<(u64, u64)> = pairs
642        .iter()
643        .map(|((x, y), _)| (x.to_bits(), y.to_bits()))
644        .collect();
645    if sources.len() != src_x.len() * src_y.len() {
646        let missing = src_y
647            .iter()
648            .flat_map(|&y| src_x.iter().map(move |&x| (x, y)))
649            .find(|(x, y)| !sources.contains(&(x.to_bits(), y.to_bits())))
650            .expect("fewer sources than crossings leaves a crossing uncovered");
651        return Some(format!(
652            "the bus sources span {}x{} tones but cover only {} of their {} crossings \
653             ({:?} is not a source); an AOD traps at every crossing of its tones, so \
654             the sources of a bus must form a complete grid",
655            src_x.len(),
656            src_y.len(),
657            sources.len(),
658            src_x.len() * src_y.len(),
659            missing
660        ));
661    }
662    None
663}

665// --- Bus src→dst relation well-formedness ---
666
667/// Check that a bus's src→dst relation is well-formed: unique sources,
668/// unique destinations, and no cycles.
669///
670/// Duplicate sources are rejected because endpoint resolution is positional
671/// first-match ([`super::types::Bus::resolve_forward`]), so a duplicated
672/// source silently shadows every later pair. Duplicate destinations are
673/// rejected because two simultaneous transports into one site cannot both
674/// complete. Cycles (including self-loops) are rejected because a bus is a
675/// set of explicit transports, never a permutation — see
676/// [`ArchSpecError::CyclicBus`]. Overlapping-but-acyclic relations
677/// (conveyor chains such as `0→1, 1→2`) are legal.
678///
679/// `dup_err` selects the error category for duplicate-endpoint errors
680/// (`ZoneBus` for per-zone buses, `InterZoneBus` for zone buses); cycles
681/// always report as [`ArchSpecError::CyclicBus`]. Pairs are zipped, so a
682/// src/dst length mismatch (reported separately) truncates to the shorter
683/// side here.
684fn check_bus_relation<T>(
685    src: &[T],
686    dst: &[T],
687    label: &str,
688    fmt: impl Fn(&T) -> String,
689    dup_err: impl Fn(String) -> ArchSpecError,
690    errors: &mut Vec<ArchSpecError>,
691) where
692    T: Copy + Eq + std::hash::Hash,
693{
694    let mut first_src: HashMap<T, usize> = HashMap::new();
695    for (i, s) in src.iter().enumerate() {
696        if let Some(&j) = first_src.get(s) {
697            errors.push(dup_err(format!(
698                "{label}: duplicate src {} at indices {j} and {i} \
699                 (first-match resolution silently shadows the later pair)",
700                fmt(s)
701            )));
702        } else {
703            first_src.insert(*s, i);
704        }
705    }
706
707    let mut first_dst: HashMap<T, usize> = HashMap::new();
708    for (i, d) in dst.iter().enumerate() {
709        if let Some(&j) = first_dst.get(d) {
710            errors.push(dup_err(format!(
711                "{label}: duplicate dst {} at indices {j} and {i} \
712                 (two simultaneous transports into one site cannot both complete)",
713                fmt(d)
714            )));
715        } else {
716            first_dst.insert(*d, i);
717        }
718    }
719
720    // Cycle detection on the src→dst functional graph. With unique sources
721    // each node has at most one successor; with duplicates the first pair
722    // wins, matching first-match endpoint resolution.
723    let mut next: HashMap<T, T> = HashMap::new();
724    for (s, d) in src.iter().zip(dst.iter()) {
725        next.entry(*s).or_insert(*d);
726    }
727
728    // 0 = unvisited (absent), 1 = on the current walk, 2 = finished.
729    let mut state: HashMap<T, u8> = HashMap::new();
730    for start in src {
731        if state.get(start).copied() == Some(2) {
732            continue;
733        }
734        let mut walk: Vec<T> = Vec::new();
735        let mut cur = *start;
736        loop {
737            match state.get(&cur).copied() {
738                Some(2) => break,
739                Some(1) => {
740                    let pos = walk
741                        .iter()
742                        .position(|n| *n == cur)
743                        .expect("a node in state 1 is always on the current walk stack");
744                    let cycle: Vec<String> = walk[pos..]
745                        .iter()
746                        .chain(std::iter::once(&cur))
747                        .map(&fmt)
748                        .collect();
749                    errors.push(ArchSpecError::CyclicBus(format!(
750                        "{label}: src→dst relation contains a cycle: {} \
751                         (a bus is a set of explicit transports, never a rotation)",
752                        cycle.join(" -> ")
753                    )));
754                    break;
755                }
756                _ => {
757                    state.insert(cur, 1);
758                    walk.push(cur);
759                    match next.get(&cur) {
760                        Some(n) => cur = *n,
761                        None => break,
762                    }
763                }
764            }
765        }
766        for n in walk {
767            state.insert(n, 2);
768        }
769    }
770}

772// --- Per-zone entangling pair validation ---
773
774/// Word indices in entangling_pairs must be valid, distinct, and no duplicate pairs.
775fn check_zone_entangling_pairs(
776    zone_idx: usize,
777    zone: &super::types::Zone,
778    num_words: usize,
779    errors: &mut Vec<ArchSpecError>,
780) {
781    let mut seen: HashSet<[u32; 2]> = HashSet::new();
782    // A word has at most one CZ partner per zone. The same word may still be
783    // paired in other zones: the word template is spec-wide, pairs are per zone.
784    let mut pair_of_word: HashMap<u32, usize> = HashMap::new();
785    for (idx, pair) in zone.entangling_pairs.iter().enumerate() {
786        let [a, b] = *pair;
787        if a as usize >= num_words {
788            errors.push(ArchSpecError::EntanglingPair(format!(
789                "zone[{}].entangling_pairs[{}]: word ID {} >= num_words ({})",
790                zone_idx, idx, a, num_words
791            )));
792        }
793        if b as usize >= num_words {
794            errors.push(ArchSpecError::EntanglingPair(format!(
795                "zone[{}].entangling_pairs[{}]: word ID {} >= num_words ({})",
796                zone_idx, idx, b, num_words
797            )));
798        }
799        if a == b {
800            errors.push(ArchSpecError::EntanglingPair(format!(
801                "zone[{}].entangling_pairs[{}]: word paired with itself ({})",
802                zone_idx, idx, a
803            )));
804        }
805        let normalized = if a <= b { [a, b] } else { [b, a] };
806        if !seen.insert(normalized) {
807            errors.push(ArchSpecError::EntanglingPair(format!(
808                "zone[{}].entangling_pairs[{}]: duplicate pair [{}, {}]",
809                zone_idx, idx, a, b
810            )));
811            // Already reported as a duplicate; don't also flag its words.
812            continue;
813        }
814        if a == b {
815            continue;
816        }
817        for word in [a, b] {
818            if let Some(&first) = pair_of_word.get(&word) {
819                errors.push(ArchSpecError::EntanglingPair(format!(
820                    "zone[{}].entangling_pairs[{}]: word {} is already paired in \
821                     entangling_pairs[{}] (a word has at most one partner per zone)",
822                    zone_idx, idx, word, first
823                )));
824            } else {
825                pair_of_word.insert(word, idx);
826            }
827        }
828    }
829}

831// --- Mode validation ---
832
833/// Zone indices and bitstring_order entries must be valid.
834fn check_modes(
835    spec: &ArchSpec,
836    num_zones: usize,
837    num_words: usize,
838    sites_per_word: usize,
839    errors: &mut Vec<ArchSpecError>,
840) {
841    for (mode_idx, mode) in spec.modes.iter().enumerate() {
842        for &zone_id in &mode.zones {

844                errors.push(ArchSpecError::Mode(format!(
845                    "mode '{}' (index {}): zone ID {} >= num_zones ({})",

847                )));
848            }
849        }
850        for (loc_idx, loc) in mode.bitstring_order.iter().enumerate() {
851            if loc.zone_id as usize >= num_zones {
852                errors.push(ArchSpecError::Mode(format!(
853                    "mode '{}' (index {}): bitstring_order[{}] zone_id {} >= num_zones ({})",
854                    mode.name, mode_idx, loc_idx, loc.zone_id, num_zones
855                )));
856            }
857            if loc.word_id as usize >= num_words {
858                errors.push(ArchSpecError::Mode(format!(
859                    "mode '{}' (index {}): bitstring_order[{}] word_id {} >= num_words ({})",
860                    mode.name, mode_idx, loc_idx, loc.word_id, num_words
861                )));
862            }
863            if loc.site_id as usize >= sites_per_word {
864                errors.push(ArchSpecError::Mode(format!(
865                    "mode '{}' (index {}): bitstring_order[{}] site_id {} >= sites_per_word ({})",
866                    mode.name, mode_idx, loc_idx, loc.site_id, sites_per_word
867                )));
868            }
869        }
870    }

892                )));
893            }
894
895            if path.waypoints.len() < 2 {
896                errors.push(ArchSpecError::Path(format!(
897                    "paths[{}]: lane 0x{:016X} has {} waypoint(s), minimum is 2",
898                    idx,
899                    path.lane,
900                    path.waypoints.len()
901                )));
902            }
903        }
904    }

907/// Verify that no two zones have overlapping bounding boxes in physical
908/// (x, y) space. Overlap would produce ambiguous site positions and make
909/// inter-zone path generation impossible (#463).
910fn check_zone_overlap(spec: &ArchSpec, errors: &mut Vec<ArchSpecError>) {
911    let boxes: Vec<(usize, (f64, f64, f64, f64))> = spec
912        .zones
913        .iter()
914        .enumerate()
915        .map(|(i, z)| (i, z.grid.bounding_box()))
916        .collect();
917
918    for i in 0..boxes.len() {
919        for j in (i + 1)..boxes.len() {
920            let (zi, (ax_min, ax_max, ay_min, ay_max)) = boxes[i];
921            let (zj, (bx_min, bx_max, by_min, by_max)) = boxes[j];
922
923            // Two axis-aligned rectangles overlap iff they overlap on
924            // both axes independently.
925            let x_overlap = ax_min < bx_max && bx_min < ax_max;
926            let y_overlap = ay_min < by_max && by_min < ay_max;
927
928            if x_overlap && y_overlap {
929                errors.push(ArchSpecError::Structure(format!(
930                    "zone {} and zone {} have overlapping bounding boxes: \
931                     zone {} spans x=[{:.6}, {:.6}] y=[{:.6}, {:.6}], \
932                     zone {} spans x=[{:.6}, {:.6}] y=[{:.6}, {:.6}]",
933                    zi, zj, zi, ax_min, ax_max, ay_min, ay_max, zj, bx_min, bx_max, by_min, by_max,
934                )));
935            }
936        }
937    }
938}

1362    /// One zone on a 5 µm × 3 µm grid with `words.len()` words, each
1363    /// `nx × ny` sites laid out row-major (site `i` at column `i % nx`, row
1364    /// `i / nx`), placed side by side along x. Every site is a word-bus
1365    /// landing pad and every word has site buses; the buses start empty.
1366    fn aod_spec(num_words: u32, nx: u32, ny: u32) -> ArchSpec {
1367        let xs: Vec<f64> = (0..num_words * nx).map(|i| f64::from(i) * 5.0).collect();
1368        let ys: Vec<f64> = (0..ny).map(|j| f64::from(j) * 3.0).collect();
1369        let words = (0..num_words)
1370            .map(|w| Word {
1371                sites: (0..nx * ny).map(|i| [w * nx + i % nx, i / nx]).collect(),
1372            })
1373            .collect();
1374        ArchSpec {
1375            version: Version::new(2, 0),
1376            words,
1377            zones: vec![Zone {
1378                name: String::new(),
1379                grid: Grid::from_positions(&xs, &ys),
1380                site_buses: vec![],
1381                word_buses: vec![],
1382                words_with_site_buses: (0..num_words).collect(),
1383                sites_with_word_buses: (0..nx * ny).collect(),
1384                entangling_pairs: vec![],
1385            }],
1386            zone_buses: vec![],
1387            modes: vec![Mode {
1388                name: "all".to_string(),
1389                zones: vec![0],
1390                bitstring_order: vec![],
1391            }],
1392            paths: None,
1393            feed_forward: false,
1394            atom_reloading: false,
1395            blockade_radius: None,
1396            aod_capacity: None,
1397        }
1398    }
